import React, { useState } from 'react';
import { Button } from '@/components/ui/button';
import { Input } from '@/components/ui/input';
import { Textarea } from '@/components/ui/textarea';
import { Label } from '@/components/ui/label';
import { Switch } from '@/components/ui/switch';
import {
  Select,
  SelectContent,
  SelectItem,
  SelectTrigger,
  SelectValue,
} from '@/components/ui/select';
import { X, Plus, ChevronDown, ChevronUp } from 'lucide-react';
import CountryCombobox from '../shared/CountryCombobox';
import CityCombobox from '../shared/CityCombobox';
import RoleMultiSelect from './RoleMultiSelect';
import { personKey, rolesOf } from '@/lib/contactRoles';

const CONTRACTING_TYPES = ['promoter', 'venue', 'club'];

export default function ContactForm({ contact, onSubmit, onCancel, isNew = false }) {
  const [formData, setFormData] = useState(
    contact || {
      name: '',
      type: 'promoter',
      geography: '',
      company: '',
      email: '',
      phone: '',
      country: '',
      city: '',
      website: '',
      social_media: {},
      notes: '',
      tags: [],
      rating: '',
      default_deal_terms: {},
    }
  );

  const [tagInput, setTagInput] = useState('');
  const [contractingOpen, setContractingOpen] = useState(false);

  const setDealTerms = (key, val) => setFormData(prev => ({
    ...prev,
    default_deal_terms: { ...(prev.default_deal_terms || {}), [key]: val },
  }));

  const setDealTermsAddress = (key, val) => setFormData(prev => ({
    ...prev,
    default_deal_terms: {
      ...(prev.default_deal_terms || {}),
      billing_address: { ...(prev.default_deal_terms?.billing_address || {}), [key]: val },
    },
  }));

  // Initialize contact_persons from existing data or seed with the primary contact fields
  const [contactPersons, setContactPersons] = useState(() => {
    if (contact?.contact_persons?.length > 0) return contact.contact_persons;
    // Seed from legacy fields if present
    return [{ name: contact?.contact_person || '', email: contact?.email || '', phone: contact?.phone || '' }];
  });

  // Role tags are kept per row and saved into person_roles keyed by personKey.
  const [personRoles, setPersonRoles] = useState(() => contactPersons.map(p => rolesOf(contact, p)));
  const setRolesAt = (i, roles) => setPersonRoles(prev => { const next = [...prev]; next[i] = roles; return next; });

  const addPerson = () => { setContactPersons([...contactPersons, { name: '', email: '', phone: '' }]); setPersonRoles(prev => [...prev, []]); };
  const removePerson = (i) => { setContactPersons(contactPersons.filter((_, idx) => idx !== i)); setPersonRoles(prev => prev.filter((_, idx) => idx !== i)); };
  const updatePerson = (i, field, value) => {
    const updated = contactPersons.map((p, idx) => idx === i ? { ...p, [field]: value } : p);
    setContactPersons(updated);
  };

  const handleSubmit = (e) => {
    e.preventDefault();
    const kept = contactPersons.map((p, i) => ({ p, roles: personRoles[i] || [] })).filter(({ p }) => p.name || p.email || p.phone);
    const cleaned = kept.map(({ p }) => p);
    const person_roles = {};
    kept.forEach(({ p, roles }) => { if (roles.length) person_roles[personKey(p)] = roles; });
    const primary = cleaned[0] || {};
    onSubmit({
      ...formData,
      email: primary.email || '',
      phone: primary.phone || '',
      contact_persons: cleaned,
      person_roles,
      rating: formData.rating ? parseInt(formData.rating) : undefined,
    });
  };

  const addTag = () => {
    if (tagInput.trim() && !formData.tags.includes(tagInput.trim())) {
      setFormData({
        ...formData,
        tags: [...formData.tags, tagInput.trim()],
      });
      setTagInput('');
    }
  };

  const removeTag = (index) => {
    setFormData({
      ...formData,
      tags: formData.tags.filter((_, i) => i !== index),
    });
  };

  return (
    <div>
      <form onSubmit={handleSubmit} className="space-y-6">
        <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
          <div>
            <Label htmlFor="name">Name *</Label>
            <Input
              id="name"
              value={formData.name}
              onChange={(e) => setFormData({ ...formData, name: e.target.value })}
              required
              className="mt-1"
            />
          </div>

          <div>
            <Label htmlFor="type">Type *</Label>
            <Select
              value={formData.type}
              onValueChange={(value) => setFormData({ ...formData, type: value })}
            >
              <SelectTrigger className="mt-1">
                <SelectValue />
              </SelectTrigger>
              <SelectContent>
                <SelectItem value="promoter">Promoter</SelectItem>
                <SelectItem value="venue">Venue</SelectItem>
                <SelectItem value="club">Club</SelectItem>
                <SelectItem value="festival">Festival</SelectItem>
                <SelectItem value="brand">Brand</SelectItem>
                <SelectItem value="label">Label</SelectItem>
                <SelectItem value="agent">Agent</SelectItem>
                <SelectItem value="a_and_r">A&R</SelectItem>
                <SelectItem value="media">Media</SelectItem>
                <SelectItem value="other">Other</SelectItem>
              </SelectContent>
            </Select>
          </div>
        </div>

        {/* Agency is deprecated — hidden from the UI, still stored on legacy records. */}
        <div>
          <Label htmlFor="role">Role / Title</Label>
          <Input
            id="role"
            value={formData.role || ''}
            onChange={(e) => setFormData({ ...formData, role: e.target.value })}
            placeholder="e.g. A&R Manager, Booking Agent"
            className="mt-1"
          />
        </div>

        <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
          <div>
            <Label htmlFor="geography">Geography</Label>
            <Select
              value={formData.geography || ''}
              onValueChange={(value) => setFormData({ ...formData, geography: value })}
            >
              <SelectTrigger className="mt-1">
                <SelectValue placeholder="Select geography" />
              </SelectTrigger>
              <SelectContent>
                <SelectItem value="NA">NA</SelectItem>
                <SelectItem value="LATAM">LATAM</SelectItem>
                <SelectItem value="Europe">Europe</SelectItem>
                <SelectItem value="MENA">MENA</SelectItem>
                <SelectItem value="APAC">APAC</SelectItem>
              </SelectContent>
            </Select>
          </div>

          <div>
            <Label htmlFor="rating">Rating (1-5)</Label>
            <Select
              value={formData.rating?.toString() || ''}
              onValueChange={(value) => setFormData({ ...formData, rating: value })}
            >
              <SelectTrigger className="mt-1">
                <SelectValue placeholder="Select rating" />
              </SelectTrigger>
              <SelectContent>
                <SelectItem value="1">1 Star</SelectItem>
                <SelectItem value="2">2 Stars</SelectItem>
                <SelectItem value="3">3 Stars</SelectItem>
                <SelectItem value="4">4 Stars</SelectItem>
                <SelectItem value="5">5 Stars</SelectItem>
              </SelectContent>
            </Select>
          </div>
        </div>

        <div>
          <Label htmlFor="company">Company</Label>
          <Input
            id="company"
            value={formData.company || ''}
            onChange={(e) => setFormData({ ...formData, company: e.target.value })}
            className="mt-1"
          />
        </div>

        {/* Contact Persons */}
        <div>
          <div className="flex items-center justify-between mb-2">
            <Label>Contact People</Label>
            <button
              type="button"
              onClick={addPerson}
              className="flex items-center gap-1 text-xs font-semibold text-black bg-yellow-400 hover:bg-yellow-500 px-2 py-1 rounded-lg transition"
            >
              <Plus className="w-3.5 h-3.5" /> Add Person
            </button>
          </div>
          <div className="space-y-3">
            {contactPersons.map((person, i) => (
              <div key={i} className="border-2 border-gray-100 rounded-xl p-3 space-y-2 relative bg-gray-50">
                {contactPersons.length > 1 && (
                  <button
                    type="button"
                    onClick={() => removePerson(i)}
                    className="absolute top-2 right-2 text-gray-400 hover:text-red-500 transition"
                  >
                    <X className="w-4 h-4" />
                  </button>
                )}
                <div className="grid grid-cols-1 md:grid-cols-3 gap-2">
                  <Input
                    placeholder="Name"
                    value={person.name}
                    onChange={(e) => updatePerson(i, 'name', e.target.value)}
                    className="bg-white"
                  />
                  <Input
                    placeholder="Email"
                    type="email"
                    value={person.email}
                    onChange={(e) => updatePerson(i, 'email', e.target.value)}
                    className="bg-white"
                  />
                  <Input
                    placeholder="Phone"
                    value={person.phone}
                    onChange={(e) => updatePerson(i, 'phone', e.target.value)}
                    className="bg-white"
                  />
                </div>
                <RoleMultiSelect value={personRoles[i] || []} onChange={(roles) => setRolesAt(i, roles)} />
              </div>
            ))}
          </div>
        </div>

        <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
          <div>
            <Label htmlFor="country">Country</Label>
            <CountryCombobox
              value={formData.country || ''}
              onChange={(country) => setFormData({ ...formData, country, city: country === formData.country ? formData.city : '' })}
            />
          </div>

          <div>
            <Label htmlFor="city">City</Label>
            <CityCombobox
              value={formData.city || ''}
              country={formData.country}
              onChange={(city) => setFormData({ ...formData, city })}
            />
          </div>
        </div>

        <div>
          <Label htmlFor="website">Website</Label>
          <Input
            id="website"
            value={formData.website || ''}
            onChange={(e) => setFormData({ ...formData, website: e.target.value })}
            className="mt-1"
          />
        </div>

        <div>
          <Label>Tags</Label>
          <div className="flex gap-2 mt-1">
            <Input
              value={tagInput}
              onChange={(e) => setTagInput(e.target.value)}
              placeholder="Add tag..."
              onKeyPress={(e) => {
                if (e.key === 'Enter') {
                  e.preventDefault();
                  addTag();
                }
              }}
            />
            <Button type="button" onClick={addTag} size="icon">
              <Plus className="w-4 h-4" />
            </Button>
          </div>
          <div className="flex flex-wrap gap-2 mt-2">
            {formData.tags?.map((tag, idx) => (
              <span
                key={idx}
                className="bg-yellow-400 text-black px-3 py-1 rounded-full text-sm font-medium flex items-center gap-2"
              >
                {tag}
                <button type="button" onClick={() => removeTag(idx)}>
                  <X className="w-3 h-3" />
                </button>
              </span>
            ))}
          </div>
        </div>

        <div>
          <Label htmlFor="notes">Notes</Label>
          <Textarea
            id="notes"
            value={formData.notes || ''}
            onChange={(e) => setFormData({ ...formData, notes: e.target.value })}
            rows={4}
            className="mt-1"
          />
        </div>

        {/* Contracting Defaults — shown only for promoter / venue / club */}
        {CONTRACTING_TYPES.includes(formData.type) && (
          <div className="border border-gray-200 rounded-xl overflow-hidden">
            <button
              type="button"
              onClick={() => setContractingOpen(p => !p)}
              className="w-full flex items-center justify-between px-4 py-3 bg-gray-50 hover:bg-gray-100 transition-colors"
            >
              <span className="text-sm font-semibold text-gray-700">Contracting Defaults</span>
              {contractingOpen ? <ChevronUp className="w-4 h-4 text-gray-500" /> : <ChevronDown className="w-4 h-4 text-gray-500" />}
            </button>

            {contractingOpen && (
              <div className="p-4 space-y-4">
                <p className="text-[11px] text-gray-400 italic">
                  These values are copied as a one-time snapshot when this contact is linked to a show — edits here do not affect any existing show deals.
                </p>

                <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
                  <div>
                    <Label className="text-xs">Deal Type</Label>
                    <Select
                      value={formData.default_deal_terms?.default_deal_type || ''}
                      onValueChange={v => setDealTerms('default_deal_type', v)}
                    >
                      <SelectTrigger className="mt-1">
                        <SelectValue placeholder="Select..." />
                      </SelectTrigger>
                      <SelectContent>
                        <SelectItem value="flat_fee">Flat Fee</SelectItem>
                        <SelectItem value="vs_door">Vs Door</SelectItem>
                        <SelectItem value="vs_door_guarantee">Vs Door + Guarantee</SelectItem>
                        <SelectItem value="percentage">Percentage</SelectItem>
                      </SelectContent>
                    </Select>
                  </div>

                  <div>
                    <Label className="text-xs">WHT Base</Label>
                    <Select
                      value={formData.default_deal_terms?.default_wht_base || ''}
                      onValueChange={v => setDealTerms('default_wht_base', v)}
                    >
                      <SelectTrigger className="mt-1">
                        <SelectValue placeholder="Select..." />
                      </SelectTrigger>
                      <SelectContent>
                        <SelectItem value="gross_fee">Gross Fee</SelectItem>
                        <SelectItem value="gross_fee_minus_costs">Gross Fee − Costs</SelectItem>
                        <SelectItem value="gross_fee_minus_costs_minus_booking_fee">Gross Fee − Costs − Booking Fee</SelectItem>
                      </SelectContent>
                    </Select>
                  </div>
                </div>

                <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
                  <div>
                    <Label className="text-xs">Default WHT %</Label>
                    <Input
                      type="number"
                      className="mt-1"
                      value={formData.default_deal_terms?.default_wht_percentage ?? ''}
                      onChange={e => setDealTerms('default_wht_percentage', parseFloat(e.target.value) || 0)}
                      placeholder="e.g. 24"
                    />
                  </div>
                  <div>
                    <Label className="text-xs">Default VAT %</Label>
                    <Input
                      type="number"
                      className="mt-1"
                      value={formData.default_deal_terms?.default_vat_percentage ?? ''}
                      onChange={e => setDealTerms('default_vat_percentage', parseFloat(e.target.value) || 0)}
                      placeholder="e.g. 21"
                    />
                  </div>
                </div>

                <div className="flex items-center gap-3">
                  <Switch
                    checked={!!formData.default_deal_terms?.default_vat_applicable}
                    onCheckedChange={v => setDealTerms('default_vat_applicable', v)}
                  />
                  <Label className="text-xs">VAT Applicable by default</Label>
                </div>

                <div className="border-t border-gray-100 pt-4 space-y-3">
                  <p className="text-xs font-semibold text-gray-600">Legal Details</p>
                  <div>
                    <Label className="text-xs">Legal Company Name</Label>
                    <Input
                      className="mt-1"
                      value={formData.default_deal_terms?.legal_company_name || ''}
                      onChange={e => setDealTerms('legal_company_name', e.target.value)}
                      placeholder="Full registered company name"
                    />
                  </div>
                  <div>
                    <Label className="text-xs">Company Registration Number</Label>
                    <Input
                      className="mt-1"
                      value={formData.default_deal_terms?.company_registration_number || ''}
                      onChange={e => setDealTerms('company_registration_number', e.target.value)}
                      placeholder="CIF / VAT / Reg number"
                    />
                  </div>
                </div>

                <div className="border-t border-gray-100 pt-4 space-y-3">
                  <p className="text-xs font-semibold text-gray-600">Billing Address</p>
                  <Input
                    placeholder="Street"
                    value={formData.default_deal_terms?.billing_address?.street || ''}
                    onChange={e => setDealTermsAddress('street', e.target.value)}
                  />
                  <div className="grid grid-cols-2 gap-3">
                    <CityCombobox
                      placeholder="City"
                      className=""
                      value={formData.default_deal_terms?.billing_address?.city || ''}
                      country={formData.default_deal_terms?.billing_address?.country}
                      onChange={city => setDealTermsAddress('city', city)}
                    />
                    <Input
                      placeholder="Postal Code"
                      value={formData.default_deal_terms?.billing_address?.postal_code || ''}
                      onChange={e => setDealTermsAddress('postal_code', e.target.value)}
                    />
                  </div>
                  <CountryCombobox
                    placeholder="Country"
                    className=""
                    value={formData.default_deal_terms?.billing_address?.country || ''}
                    onChange={country => setDealTermsAddress('country', country)}
                  />
                </div>
              </div>
            )}
          </div>
        )}

        <div className="flex justify-end gap-3 pt-4 border-t">
          <Button type="button" variant="outline" onClick={onCancel}>
            Cancel
          </Button>
          <Button
            type="submit"
            className="bg-yellow-400 text-black hover:bg-yellow-500 font-semibold"
          >
            {contact && !isNew ? 'Update' : 'Create'}
          </Button>
        </div>
      </form>
    </div>
  );
}
