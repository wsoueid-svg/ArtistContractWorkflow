import React from 'react';
import { Outlet, useLocation, useSearchParams } from 'react-router-dom';
import { CloudOff, Loader2, AlertTriangle } from 'lucide-react';
import { PortalProvider, usePortal } from '@/lib/portal/PortalContext';
import PortalTopBar from './PortalTopBar';
import PortalBottomNav from './PortalBottomNav';

function titleFor(pathname) {
  if (pathname.startsWith('/portal/shows')) return 'Show Detail';
  if (pathname.startsWith('/portal/tours')) return 'Tours';
  if (pathname.startsWith('/portal/profile')) return 'My Profile';
  return 'Itinerary';
}

function PortalShell() {
  const { pathname } = useLocation();
  const [searchParams] = useSearchParams();
  const { online, isLoading, error, hasData, lastSyncedAt } = usePortal();

  const artist = searchParams.get('artist');
  const search = artist ? `?artist=${artist}` : '';

  const syncedLabel = lastSyncedAt
    ? new Date(lastSyncedAt).toLocaleString('en-GB', { day: '2-digit', month: 'short', hour: '2-digit', minute: '2-digit' })
    : null;

  return (
    <div className="min-h-screen bg-slate-50">
      <PortalTopBar title={titleFor(pathname)} />

      {/* Offline banner — slim, non-blocking, cached data still renders below */}
      {!online && (
        <div className="sticky top-0 z-30 flex items-center justify-center gap-2 bg-slate-800 px-4 py-2 text-[11px] font-semibold text-slate-200">
          <CloudOff className="h-3.5 w-3.5 text-yellow-400" aria-hidden="true" />
          Offline — showing cached itinerary
          {syncedLabel && <span className="font-normal text-slate-400">· {syncedLabel}</span>}
        </div>
      )}

      <main className="mx-auto max-w-lg px-4 pb-28 pt-4">
        {isLoading ? (
          <div className="flex flex-col items-center gap-3 py-20 text-slate-400">
            <Loader2 className="h-6 w-6 animate-spin" />
            <p className="text-sm">Loading your itinerary…</p>
          </div>
        ) : error && !hasData ? (
          <div className="rounded-2xl border border-slate-200 bg-white p-6 text-center">
            <AlertTriangle className="mx-auto h-6 w-6 text-yellow-500" aria-hidden="true" />
            <p className="mt-3 text-sm font-semibold text-slate-900">
              {error.error === 'no_artist_linked' ? 'No artist linked yet' : 'Couldn’t load your itinerary'}
            </p>
            <p className="mt-1 text-xs text-slate-500">
              {error.message || 'Pull to refresh once you have signal, or contact your agent.'}
            </p>
          </div>
        ) : (
          <Outlet />
        )}
      </main>

      <PortalBottomNav search={search} />
    </div>
  );
}

/**
 * Standalone artist-portal shell — deliberately NOT the staff Layout.
 * No sidebar, no org switcher, no command palette.
 */
export default function PortalLayout() {
  return (
    <PortalProvider>
      <PortalShell />
    </PortalProvider>
  );
}
