import React from 'react';
import { AlertTriangle, Clock, RefreshCw } from 'lucide-react';

export default function ContractAlerts({ contracts, artists }) {
  const now = new Date();
  const in30 = new Date(now.getTime() + 30 * 24 * 60 * 60 * 1000);
  const in60 = new Date(now.getTime() + 60 * 24 * 60 * 60 * 1000);

  const expired = contracts.filter(c => c.end_date && new Date(c.end_date) < now && c.status !== 'cancelled');
  const expiringSoon = contracts.filter(c => {
    if (!c.end_date || c.status === 'cancelled') return false;
    const d = new Date(c.end_date);
    return d >= now && d <= in30;
  });
  const renewalSoon = contracts.filter(c => {
    if (!c.renewal_date || c.status === 'cancelled') return false;
    const d = new Date(c.renewal_date);
    return d >= now && d <= in60;
  });

  if (expired.length === 0 && expiringSoon.length === 0 && renewalSoon.length === 0) return null;

  const getArtistName = (id) => artists.find(a => a.id === id)?.name || 'Unknown';

  return (
    <div className="space-y-2 mb-6">
      {expired.length > 0 && (
        <div className="bg-red-50 border-2 border-red-200 rounded-xl p-4">
          <div className="flex items-center gap-2 mb-2">
            <AlertTriangle className="w-4 h-4 text-red-600" />
            <span className="font-semibold text-sm text-red-800">Expired Contracts ({expired.length})</span>
          </div>
          <div className="space-y-1">
            {expired.slice(0, 5).map(c => (
              <p key={c.id} className="text-xs text-red-700">
                <strong>{c.title}</strong> — {getArtistName(c.artist_id)} — expired {new Date(c.end_date).toLocaleDateString()}
              </p>
            ))}
            {expired.length > 5 && <p className="text-xs text-red-500 italic">+{expired.length - 5} more</p>}
          </div>
        </div>
      )}
      {expiringSoon.length > 0 && (
        <div className="bg-yellow-50 border-2 border-yellow-200 rounded-xl p-4">
          <div className="flex items-center gap-2 mb-2">
            <Clock className="w-4 h-4 text-yellow-700" />
            <span className="font-semibold text-sm text-yellow-800">Expiring Within 30 Days ({expiringSoon.length})</span>
          </div>
          <div className="space-y-1">
            {expiringSoon.map(c => (
              <p key={c.id} className="text-xs text-yellow-700">
                <strong>{c.title}</strong> — {getArtistName(c.artist_id)} — expires {new Date(c.end_date).toLocaleDateString()}
              </p>
            ))}
          </div>
        </div>
      )}
      {renewalSoon.length > 0 && (
        <div className="bg-blue-50 border-2 border-blue-200 rounded-xl p-4">
          <div className="flex items-center gap-2 mb-2">
            <RefreshCw className="w-4 h-4 text-blue-600" />
            <span className="font-semibold text-sm text-blue-800">Upcoming Renewals ({renewalSoon.length})</span>
          </div>
          <div className="space-y-1">
            {renewalSoon.map(c => (
              <p key={c.id} className="text-xs text-blue-700">
                <strong>{c.title}</strong> — {getArtistName(c.artist_id)} — renewal due {new Date(c.renewal_date).toLocaleDateString()}
              </p>
            ))}
          </div>
        </div>
      )}
    </div>
  );
}
