import React, { useRef, useState } from 'react';
import { useQuery } from '@tanstack/react-query';
import { base44 } from '@/api/base44Client';
import { Paperclip, Loader2, X } from 'lucide-react';

const SOURCE_LABEL = { vault: 'Vault', show: 'Show', upload: 'Upload' };

export default function RiderAttachmentPicker({ show, value, onChange }) {
  const fileRef = useRef(null);
  const [uploading, setUploading] = useState(false);
  const { data: vault = [] } = useQuery({ queryKey: ['artist-vault', show?.artist_id], queryFn: () => base44.entities.ArtistDocument.filter({ artist_id: show.artist_id }), enabled: !!show?.artist_id });
  const { data: showDocs = [] } = useQuery({ queryKey: ['show-documents', show?.id], queryFn: () => base44.entities.ShowDocument.filter({ show_id: show.id }), enabled: !!show?.id });

  const has = (source, id) => value.some(r => r.source === source && r.id === id);
  const add = (rider) => value.length < 5 && onChange([...value, rider]);
  const remove = (index) => onChange(value.filter((_, i) => i !== index));

  const pick = (e) => {
    const [source, id] = e.target.value.split(':');
    e.target.value = '';
    const doc = (source === 'vault' ? vault : showDocs).find(d => d.id === id);
    if (doc && !has(source, id)) add({ source, id, filename: doc.file_name });
  };

  const upload = async (e) => {
    const file = e.target.files?.[0];
    e.target.value = '';
    if (!file) return;
    setUploading(true);
    const { file_uri } = await base44.integrations.Core.UploadPrivateFile({ file });
    add({ source: 'upload', file_uri, filename: file.name });
    setUploading(false);
  };

  return (
    <div className="space-y-2">
      <div className="flex gap-2">
        <select onChange={pick} defaultValue="" disabled={value.length >= 5} className="h-11 min-w-0 flex-1 rounded-lg border border-slate-200 bg-white px-2 text-xs text-slate-700 outline-none">
          <option value="" disabled>Add rider from documents…</option>
          {vault.length > 0 && <optgroup label="Artist vault">{vault.map(d => <option key={d.id} value={`vault:${d.id}`}>{d.file_name}</option>)}</optgroup>}
          {showDocs.length > 0 && <optgroup label="Show documents">{showDocs.map(d => <option key={d.id} value={`show:${d.id}`}>{d.file_name}</option>)}</optgroup>}
        </select>
        <button type="button" onClick={() => fileRef.current?.click()} disabled={uploading || value.length >= 5} className="flex h-11 items-center gap-1.5 rounded-lg border border-slate-200 bg-white px-3 text-xs font-semibold text-slate-700 hover:bg-slate-100 disabled:opacity-50">
          {uploading ? <Loader2 className="h-3.5 w-3.5 animate-spin" /> : <Paperclip className="h-3.5 w-3.5" />} Upload
        </button>
        <input ref={fileRef} type="file" className="hidden" onChange={upload} />
      </div>
      {value.length > 0 && (
        <div className="flex flex-wrap gap-1.5">
          {value.map((r, i) => (
            <span key={`${r.source}-${r.id || r.file_uri}`} className="inline-flex items-center gap-1.5 rounded-full border border-slate-200 bg-white py-1 pl-2.5 pr-1 text-[11px] text-slate-700">
              <span className="font-semibold uppercase tracking-wide text-slate-400">{SOURCE_LABEL[r.source]}</span>
              <span className="max-w-[160px] truncate">{r.filename}</span>
              <button type="button" onClick={() => remove(i)} aria-label={`Remove ${r.filename}`} className="rounded-full p-1 hover:bg-slate-100"><X className="h-3 w-3" /></button>
            </span>
          ))}
        </div>
      )}
    </div>
  );
}
