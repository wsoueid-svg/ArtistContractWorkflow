import React, { useState } from 'react';
import { useMutation, useQueryClient } from '@tanstack/react-query';
import { base44 } from '@/api/base44Client';
import { Button } from '@/components/ui/button';
import { Input } from '@/components/ui/input';
import { Label } from '@/components/ui/label';
import { Textarea } from '@/components/ui/textarea';
import { Badge } from '@/components/ui/badge';
import { AlertDialog, AlertDialogAction, AlertDialogCancel, AlertDialogContent, AlertDialogDescription, AlertDialogFooter, AlertDialogHeader, AlertDialogTitle } from '@/components/ui/alert-dialog';
import { Plus, Pencil, Trash2, Clock, GripVertical, Save, X, AlertTriangle } from 'lucide-react';
import { toast } from 'sonner';
import { useAuth } from '@/lib/AuthContext';
import { useOrg } from '@/lib/OrgContext';
import { auditStamp, deepClone, invalidateShowQueries } from '@/lib/showEdit';

const EVENT_TYPES = [
  { value: 'load_in', label: 'Load In', color: 'bg-gray-100 text-gray-700 border-gray-300' },
  { value: 'sound_check', label: 'Sound Check', color: 'bg-blue-100 text-blue-700 border-blue-300' },
  { value: 'doors_open', label: 'Doors Open', color: 'bg-yellow-100 text-yellow-700 border-yellow-300' },
  { value: 'support_act', label: 'Support Act', color: 'bg-purple-100 text-purple-700 border-purple-300' },
  { value: 'artist_on_stage', label: 'Artist On Stage', color: 'bg-green-100 text-green-700 border-green-300' },
  { value: 'set_end', label: 'Set End', color: 'bg-orange-100 text-orange-700 border-orange-300' },
  { value: 'curfew', label: 'Curfew', color: 'bg-red-100 text-red-700 border-red-300' },
  { value: 'travel', label: 'Travel', color: 'bg-indigo-100 text-indigo-700 border-indigo-300' },
  { value: 'other', label: 'Other', color: 'bg-gray-100 text-gray-600 border-gray-300' },
];

const getTypeConfig = (type) => EVENT_TYPES.find(t => t.value === type) || EVENT_TYPES[EVENT_TYPES.length - 1];

function toMinutes(timeStr) {
  if (!timeStr) return -1;
  const [h, m] = timeStr.split(':').map(Number);
  return h * 60 + m;
}

function EventRow({ event, idx, onEdit, onDelete }) {
  const cfg = getTypeConfig(event.type);
  return (
    <div className="flex items-start gap-3 group">
      {/* Timeline */}
      <div className="flex flex-col items-center flex-shrink-0 pt-1">
        <div className={`w-3 h-3 rounded-full border-2 ${event.type === 'artist_on_stage' ? 'bg-yellow-400 border-yellow-500' : 'bg-white border-gray-400'}`} />
        {idx !== undefined && <div className="w-0.5 h-8 bg-gray-200 mt-0.5" />}
      </div>

      {/* Content */}
      <div className="flex-1 flex items-start justify-between gap-2 pb-3">
        <div className="flex items-start gap-2 flex-wrap flex-1 min-w-0">
          <span className="text-sm font-mono font-semibold text-black w-12 flex-shrink-0">
            {event.time || '--:--'}
          </span>
          <Badge className={`border text-xs py-0 ${cfg.color} flex-shrink-0`}>{cfg.label}</Badge>
          {event.notes && (
            <span className="text-xs text-gray-500 truncate flex-1">{event.notes}</span>
          )}
        </div>
        <div className="flex gap-1 opacity-0 group-hover:opacity-100 transition-opacity flex-shrink-0">
          <Button size="sm" variant="ghost" className="h-6 w-6 p-0" onClick={() => onEdit(event)}>
            <Pencil className="w-3 h-3" />
          </Button>
          <Button size="sm" variant="ghost" className="h-6 w-6 p-0 text-red-400 hover:text-red-600" onClick={() => onDelete(event)}>
            <Trash2 className="w-3 h-3" />
          </Button>
        </div>
      </div>
    </div>
  );
}

function EventForm({ initial, onSave, onCancel }) {
  const [data, setData] = useState(initial || { type: 'other', time: '', notes: '' });

  return (
    <div className="bg-gray-50 border border-gray-200 rounded-xl p-4 space-y-3">
      <div className="grid grid-cols-2 gap-3">
        <div>
          <Label className="text-xs">Event Type</Label>
          <select
            value={data.type}
            onChange={e => setData({ ...data, type: e.target.value })}
            className="w-full mt-1 px-2 py-1.5 text-sm border border-gray-200 rounded-md focus:border-yellow-400 focus:outline-none"
          >
            {EVENT_TYPES.map(t => (
              <option key={t.value} value={t.value}>{t.label}</option>
            ))}
          </select>
        </div>
        <div>
          <Label className="text-xs">Time</Label>
          <Input
            type="time"
            value={data.time}
            onChange={e => setData({ ...data, time: e.target.value })}
            className="mt-1 h-8 text-sm"
          />
        </div>
      </div>
      <div>
        <Label className="text-xs">Notes (optional)</Label>
        <Input
          value={data.notes || ''}
          onChange={e => setData({ ...data, notes: e.target.value })}
          placeholder="Duration, special requirements..."
          className="mt-1 h-8 text-sm"
        />
      </div>
      <div className="flex justify-end gap-2 pt-1">
        <Button size="sm" variant="ghost" onClick={onCancel} className="h-7 text-xs">
          <X className="w-3 h-3 mr-1" /> Cancel
        </Button>
        <Button
          size="sm"
          onClick={() => onSave(data)}
          disabled={!data.type}
          className="h-7 text-xs bg-yellow-400 text-black hover:bg-yellow-500"
        >
          <Save className="w-3 h-3 mr-1" /> Save
        </Button>
      </div>
    </div>
  );
}

export default function ShowDayTimeline({ show }) {
  const queryClient = useQueryClient();
  const { user } = useAuth();
  const { activeOrg } = useOrg();
  const [adding, setAdding] = useState(false);
  const [editing, setEditing] = useState(null); // event object
  const [pendingDelete, setPendingDelete] = useState(null); // Bug 6 — AlertDialog target

  // timeline_events stored on show as array
  const events = [...(show.timeline_events || [])].sort((a, b) => toMinutes(a.time) - toMinutes(b.time));

  const updateMutation = useMutation({
    mutationFn: (events) => base44.entities.Show.update(show.id, { timeline_events: deepClone(events), ...auditStamp(user, activeOrg) }),
    onSuccess: async () => {
      await invalidateShowQueries(queryClient, show.id);
      setAdding(false);
      setEditing(null);
      toast.success('Timeline updated');
    },
    onError: () => toast.error('Failed to save timeline'),
  });

  const handleAdd = (newEvent) => {
    const id = `evt_${Date.now()}`;
    updateMutation.mutate([...events, { ...newEvent, id }]);
  };

  const handleEdit = (updated) => {
    updateMutation.mutate(events.map(e => e.id === updated.id ? updated : e));
  };

  const confirmDelete = () => {
    if (!pendingDelete) return;
    updateMutation.mutate(events.filter(e => e.id !== pendingDelete.id));
    setPendingDelete(null);
  };

  // Detect conflicts: same type appearing twice
  const stageEvents = events.filter(e => e.type === 'artist_on_stage');
  const hasConflict = stageEvents.length > 1;
  const legacyStageTime = show.logistics?.stage_time;
  const showLegacyWarning = Boolean(legacyStageTime) && stageEvents.length === 0;

  return (
    <div className="space-y-3">
      <div className="flex items-center justify-between">
        <div className="flex items-center gap-2">
          <Clock className="w-4 h-4 text-gray-500" />
          <h4 className="font-semibold text-sm text-gray-800">Day-of Timeline</h4>
          {hasConflict && (
            <span className="flex items-center gap-1 text-xs text-orange-600 bg-orange-50 border border-orange-200 rounded px-2 py-0.5">
              <AlertTriangle className="w-3 h-3" /> Multiple stage times — check schedule
            </span>
          )}
        </div>
        {!adding && (
          <Button
            size="sm"
            variant="outline"
            onClick={() => setAdding(true)}
            className="h-7 text-xs gap-1"
          >
            <Plus className="w-3 h-3" /> Add Event
          </Button>
        )}
      </div>

      {showLegacyWarning && (
        <div role="alert" className="flex items-start gap-2 rounded-xl border border-amber-300 bg-amber-50 p-3 text-sm text-amber-900">
          <AlertTriangle className="mt-0.5 h-4 w-4 flex-shrink-0" />
          <span><strong>Legacy Stage Time: {legacyStageTime}.</strong> Please add this to the timeline.</span>
        </div>
      )}

      {events.length === 0 && !adding && (
        <p className="text-sm text-gray-400 italic pl-1">
          No timeline events yet. Add load-in, soundcheck, stage times, etc.
        </p>
      )}

      <div>
        {events.map((event, idx) => (
          editing?.id === event.id ? (
            <div key={event.id} className="mb-2">
              <EventForm
                initial={editing}
                onSave={(data) => handleEdit({ ...data, id: event.id })}
                onCancel={() => setEditing(null)}
              />
            </div>
          ) : (
            <EventRow
              key={event.id}
              event={event}
              idx={idx < events.length - 1 ? idx : undefined}
              onEdit={(event) => setEditing(deepClone(event))}
              onDelete={setPendingDelete}
            />
          )
        ))}

        {adding && (
          <EventForm
            onSave={handleAdd}
            onCancel={() => setAdding(false)}
          />
        )}
      </div>

      <AlertDialog open={!!pendingDelete} onOpenChange={open => !open && setPendingDelete(null)}>
        <AlertDialogContent>
          <AlertDialogHeader>
            <AlertDialogTitle>Remove this timeline event?</AlertDialogTitle>
            <AlertDialogDescription>
              {pendingDelete && `"${getTypeConfig(pendingDelete.type).label}" at ${pendingDelete.time || '--:--'} will be removed from the day-of timeline. This cannot be undone.`}
            </AlertDialogDescription>
          </AlertDialogHeader>
          <AlertDialogFooter>
            <AlertDialogCancel className="min-h-[44px]">Cancel</AlertDialogCancel>
            <AlertDialogAction onClick={confirmDelete} className="min-h-[44px] bg-red-600 hover:bg-red-700">Remove event</AlertDialogAction>
          </AlertDialogFooter>
        </AlertDialogContent>
      </AlertDialog>
    </div>
  );
}
