import React, { useState, useMemo } from 'react';
import { useQuery } from '@tanstack/react-query';
import { base44 } from '@/api/base44Client';
import { Input } from '@/components/ui/input';
import { Loader2 } from 'lucide-react';

/**
 * CityCombobox — searchable city picker whose options depend on the selected country.
 * Cities load from the standardized geo dataset; unlisted cities can still be
 * added via "Use '…'" so incomplete data never blocks entry.
 */
export default function CityCombobox({ value, onChange, country, className = 'mt-1', placeholder = 'Search city…' }) {
  const [query, setQuery] = useState('');
  const [open, setOpen] = useState(false);

  const { data: cities = [], isFetching } = useQuery({
    queryKey: ['cities', country],
    queryFn: () => base44.functions.invoke('getCitiesForCountry', { country }).then(r => r.data?.cities || []),
    enabled: !!country && open,
    staleTime: 24 * 60 * 60 * 1000,
  });

  const filtered = useMemo(() => {
    const q = query.toLowerCase();
    const list = q ? cities.filter(c => c.toLowerCase().includes(q)) : cities;
    return list.slice(0, 50);
  }, [cities, query]);

  const select = (city) => { onChange(city); setOpen(false); setQuery(''); };

  return (
    <div className="relative">
      <Input
        value={open ? query : (value || '')}
        onChange={e => { setQuery(e.target.value); setOpen(true); }}
        onFocus={() => setOpen(true)}
        onBlur={() => setTimeout(() => setOpen(false), 200)}
        placeholder={country ? (value && !open ? value : placeholder) : 'Select country first'}
        className={className}
        autoComplete="off"
      />
      {open && (
        <div className="absolute z-50 top-full left-0 right-0 mt-1 bg-white border border-gray-200 rounded-lg shadow-lg max-h-48 overflow-auto">
          {!country && (
            <p className="px-3 py-2 text-xs text-gray-400">Select a country first to see its cities.</p>
          )}
          {country && isFetching && (
            <p className="px-3 py-2 text-xs text-gray-400 flex items-center gap-1.5">
              <Loader2 className="w-3 h-3 animate-spin" /> Loading cities…
            </p>
          )}
          {country && !isFetching && value && (
            <button type="button" onMouseDown={() => select('')}
              className="w-full text-left px-3 py-1.5 text-xs text-red-500 hover:bg-red-50 border-b border-gray-100">
              ✕ Clear selection
            </button>
          )}
          {country && !isFetching && filtered.map(c => (
            <button key={c} type="button" onMouseDown={() => select(c)}
              className={`w-full text-left px-3 py-1.5 text-sm hover:bg-gray-50 ${c === value ? 'font-semibold bg-gray-50' : ''}`}>
              {c}
            </button>
          ))}
          {country && !isFetching && query && !cities.some(c => c.toLowerCase() === query.toLowerCase()) && (
            <button type="button" onMouseDown={() => select(query)}
              className="w-full text-left px-3 py-2 text-sm text-blue-600 hover:bg-blue-50 border-t border-gray-100 font-medium">
              + Use "{query}"
            </button>
          )}
          {country && !isFetching && filtered.length === 0 && !query && (
            <p className="px-3 py-2 text-xs text-gray-400">No cities found — type to add one.</p>
          )}
        </div>
      )}
    </div>
  );
}
