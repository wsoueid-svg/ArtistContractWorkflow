import React from 'react';

/**
 * Component-level skeleton loader that mimics the shape of content.
 * Usage: <SkeletonBlock rows={3} /> or <SkeletonBlock type="card" />
 */
export default function SkeletonBlock({ rows = 3, type = 'lines', className = '' }) {
  if (type === 'card') {
    return (
      <div className={`bg-white rounded-xl border border-gray-100 p-5 space-y-3 ${className}`}>
        <div className="skeleton h-4 w-1/3" />
        <div className="skeleton h-8 w-2/3" />
        <div className="skeleton h-3 w-1/2" />
      </div>
    );
  }

  if (type === 'stat') {
    return (
      <div className={`bg-white rounded-xl border border-gray-100 p-4 ${className}`}>
        <div className="skeleton h-3 w-16 mb-2" />
        <div className="skeleton h-7 w-24" />
      </div>
    );
  }

  if (type === 'row') {
    return (
      <div className={`flex items-center gap-3 p-3 ${className}`}>
        <div className="skeleton h-9 w-9 rounded-lg flex-shrink-0" />
        <div className="flex-1 space-y-2">
          <div className="skeleton h-4 w-3/4" />
          <div className="skeleton h-3 w-1/2" />
        </div>
        <div className="skeleton h-5 w-16 rounded-full" />
      </div>
    );
  }

  return (
    <div className={`space-y-2.5 ${className}`}>
      {Array.from({ length: rows }).map((_, i) => (
        <div
          key={i}
          className="skeleton h-4"
          style={{ width: `${85 - i * 12}%` }}
        />
      ))}
    </div>
  );
}
