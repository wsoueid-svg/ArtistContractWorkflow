import React from 'react';
import { useOrg } from '@/lib/OrgContext';

/**
 * PageHeader — the single page-title treatment for the whole app.
 *
 * One heading scale everywhere (text-2xl → sm:text-3xl) so no two modules
 * disagree, and one visual language per org:
 *   • analog  → black banner, yellow tick, white title
 *   • elevate → light header with the yellow rule underneath
 *
 * `actions` renders on the right of the title row.
 */
export default function PageHeader({ eyebrow, title, subtitle, actions }) {
  const { activeOrg } = useOrg();
  const isAnalog = activeOrg === 'analog';

  if (isAnalog) {
    return (
      <div className="studio-page-header bg-black rounded-2xl px-4 sm:px-5 py-4 flex flex-wrap items-center justify-between gap-3">
        <div className="min-w-0">
          {eyebrow && (
            <p className="text-[10px] font-bold uppercase tracking-widest text-yellow-400 mb-1 ml-4">{eyebrow}</p>
          )}
          <h1 className="text-2xl sm:text-3xl font-bold text-white tracking-tight flex items-center gap-2.5">
            <span className="w-1.5 h-6 bg-yellow-400 rounded-full flex-shrink-0" />
            <span className="truncate">{title}</span>
          </h1>
          {subtitle && <p className="text-xs text-gray-400 mt-1 ml-4">{subtitle}</p>}
        </div>
        {actions && <div className="flex items-center gap-2 flex-wrap min-w-0">{actions}</div>}
      </div>
    );
  }

  return (
    <div className="studio-page-header flex flex-wrap items-center justify-between gap-3 border-b-4 border-yellow-400 pb-4">
      <div className="min-w-0">
        {eyebrow && (
          <p className="text-[10px] font-bold uppercase tracking-widest text-yellow-600 mb-1">{eyebrow}</p>
        )}
        <h1 className="text-2xl sm:text-3xl font-bold text-black tracking-tight truncate">{title}</h1>
        {subtitle && <p className="text-sm text-gray-500 mt-0.5">{subtitle}</p>}
      </div>
      {actions && <div className="flex items-center gap-2 flex-wrap min-w-0">{actions}</div>}
    </div>
  );
}
