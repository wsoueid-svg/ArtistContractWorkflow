import React from 'react';
import { base44 } from '@/api/base44Client';
import { useQuery } from '@tanstack/react-query';
import { CheckCircle2, Circle, Clock, AlertCircle } from 'lucide-react';

export default function MarketingPackChecklist({ show }) {
  const { data: assets = [] } = useQuery({
    queryKey: ['marketing-assets', show.id],
    queryFn: () => base44.entities.MarketingAsset.filter({ show_id: show.id }),
  });

  const { data: approvals = [] } = useQuery({
    queryKey: ['approvals', show.id],
    queryFn: () => base44.entities.Approval.filter({ show_id: show.id }),
  });

  const artworkApproved = show.marketing?.artwork_approved || false;
  const artworkProposals = show.marketing?.artwork_proposal_count || 0;
  const ticketLink = show.marketing?.ticket_link;
  const announcementReady = show.marketing?.announcement_ready || false;
  const copyApproved = show.marketing?.copy_status === 'approved';
  const promoMaterialsSubmitted = show.marketing?.promo_materials_submitted || false;

  const steps = [
    {
      id: 'artwork',
      title: 'Artwork Submission',
      description: `${artworkProposals}/2 proposals submitted`,
      status: artworkApproved ? 'completed' : artworkProposals >= 2 ? 'pending_approval' : 'pending',
      required: true,
    },
    {
      id: 'copy',
      title: 'Copy Approval',
      description: 'Announcement copy text',
      status: copyApproved ? 'completed' : show.marketing?.copy_text ? 'pending_approval' : 'pending',
      required: true,
    },
    {
      id: 'ticket_link',
      title: 'Ticket Link',
      description: 'Event ticket URL',
      status: ticketLink ? 'completed' : 'pending',
      required: true,
    },
    {
      id: 'announcement',
      title: 'Announcement Ready',
      description: 'Ready to announce publicly',
      status: announcementReady ? 'completed' : 'pending',
      required: true,
      gated: !artworkApproved || !ticketLink,
    },
    {
      id: 'promo',
      title: 'Promo Materials',
      description: 'Videos/tracks for promotion',
      status: promoMaterialsSubmitted ? 'completed' : 'pending',
      required: false,
    },
    {
      id: 'content',
      title: 'Post-Event Content',
      description: 'Photos/videos delivery',
      status: show.marketing?.post_event_content_received ? 'completed' : 'pending',
      required: false,
    },
  ];

  const completedSteps = steps.filter(s => s.status === 'completed').length;
  const progress = Math.round((completedSteps / steps.length) * 100);

  return (
    <div className="bg-white border border-gray-200 rounded p-4">
      <div className="flex justify-between items-center mb-4">
        <h3 className="font-bold text-black">Marketing Pack Progress</h3>
        <span className="text-sm font-medium text-gray-600">
          {completedSteps}/{steps.length} Complete
        </span>
      </div>

      <div className="mb-4">
        <div className="h-2 bg-gray-200 rounded-full overflow-hidden">
          <div
            className="h-full bg-yellow-400 transition-all"
            style={{ width: `${progress}%` }}
          />
        </div>
      </div>

      <div className="space-y-3">
        {steps.map((step) => {
          const Icon =
            step.status === 'completed'
              ? CheckCircle2
              : step.status === 'pending_approval'
              ? Clock
              : step.gated
              ? AlertCircle
              : Circle;

          return (
            <div key={step.id} className="flex items-start gap-3">
              <Icon
                className={`w-5 h-5 mt-0.5 ${
                  step.status === 'completed'
                    ? 'text-green-600'
                    : step.status === 'pending_approval'
                    ? 'text-yellow-600'
                    : step.gated
                    ? 'text-red-500'
                    : 'text-gray-400'
                }`}
              />
              <div className="flex-1">
                <div className="flex items-center gap-2">
                  <h4 className="font-medium text-sm text-black">{step.title}</h4>
                  {step.required && (
                    <span className="text-xs text-red-600">*</span>
                  )}
                  {step.gated && (
                    <span className="text-xs bg-red-100 text-red-700 px-2 py-0.5 rounded">
                      Blocked
                    </span>
                  )}
                </div>
                <p className="text-xs text-gray-600 mt-0.5">{step.description}</p>
                {step.gated && (
                  <p className="text-xs text-red-600 mt-1">
                    Requires artwork approval and ticket link
                  </p>
                )}
              </div>
            </div>
          );
        })}
      </div>
    </div>
  );
}
