import React, { useState } from 'react';
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
import { base44 } from '@/api/base44Client';
import { Card, CardContent } from '@/components/ui/card';
import { Badge } from '@/components/ui/badge';
import { Button } from '@/components/ui/button';
import { Dialog, DialogContent, DialogHeader, DialogTitle, DialogFooter } from '@/components/ui/dialog';
import { Textarea } from '@/components/ui/textarea';
import { CheckCircle2, Clock, Send, AlertTriangle, Eye, RefreshCw, Trash2 } from 'lucide-react';
import FormSender from './FormSender';

export default function FormSubmissionsPanel({ show, artist }) {
  const [sendDialogOpen, setSendDialogOpen] = useState(false);
  const [viewingInstance, setViewingInstance] = useState(null);
  const [remindInstance, setRemindInstance] = useState(null);
  const [remindEmailBody, setRemindEmailBody] = useState('');
  const queryClient = useQueryClient();

  const { data: instances = [], isLoading } = useQuery({
    queryKey: ['form-instances', show.id],
    queryFn: () => base44.entities.FormInstance.filter({ show_id: show.id }, '-sent_at'),
  });

  const { data: templates = [] } = useQuery({
    queryKey: ['form-templates'],
    queryFn: () => base44.entities.FormTemplate.filter({ is_active: true }),
  });

  const { data: submissions = [] } = useQuery({
    queryKey: ['form-submissions', show.id],
    queryFn: () => base44.entities.FormSubmission.filter({ show_id: show.id }),
  });

  const removeMutation = useMutation({
    mutationFn: async (instance) => {
      await base44.entities.FormInstance.delete(instance.id);
    },
    onSuccess: () => queryClient.invalidateQueries({ queryKey: ['form-instances', show.id] }),
  });

  const buildDefaultEmailBody = (instance) => {
    const formUrl = `${window.location.origin}/PublicForm?token=${instance.public_token}`;
    return `Hi ${instance.recipient_name || 'there'},

This is a friendly reminder — we still need your input for the upcoming show:

${show.event_name}
${show.venue ? `${show.venue}, ` : ''}${show.city || ''}
${new Date(show.date).toLocaleDateString('en-GB', { weekday: 'long', year: 'numeric', month: 'long', day: 'numeric' })}

Please fill in the form here: ${formUrl}

No account needed. If you've already submitted, you can ignore this email.`;
  };

  const openRemindDialog = (instance) => {
    setRemindInstance(instance);
    setRemindEmailBody(buildDefaultEmailBody(instance));
  };

  const resendMutation = useMutation({
    mutationFn: async ({ instance, body }) => {
      const user = await base44.auth.me();
      const formUrl = `${window.location.origin}/PublicForm?token=${instance.public_token}`;
      const htmlBody = `
<div style="font-family: Arial, sans-serif; max-width: 600px; margin: 0 auto;">
  <div style="background: #000; padding: 16px 20px; border-radius: 12px 12px 0 0;">
    <span style="background: #facc15; width: 28px; height: 28px; border-radius: 8px; display: inline-block; text-align: center; line-height: 28px; font-weight: bold; color: #000;">E</span>
    <span style="color: #fff; font-weight: bold; font-size: 16px; margin-left: 8px;">Elevate</span>
  </div>
  <div style="background: #fff; border: 1px solid #eee; border-top: none; border-radius: 0 0 12px 12px; padding: 28px 24px;">
    <div style="color: #333; white-space: pre-line; line-height: 1.6;">${body.replace(formUrl, `<a href="${formUrl}" style="background: #facc15; color: #000; padding: 10px 24px; border-radius: 8px; text-decoration: none; font-weight: bold; display: inline-block; margin: 8px 0;">Fill in the Form →</a>`)}</div>
  </div>
</div>`;
      await base44.functions.invoke('sendFormEmail', {
        to: instance.recipient_email,
        subject: `Reminder: Please fill in the form for ${show.event_name}`,
        body: htmlBody,
        from_name: user.full_name,
      });
    },
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ['form-instances', show.id] });
      setRemindInstance(null);
    },
  });

  const getSubmission = (instanceId) => submissions.find(s => s.form_instance_id === instanceId);
  const getTemplate = (templateId) => templates.find(t => t.id === templateId);

  const getMissingFields = (instance) => {
    const template = getTemplate(instance.template_id);
    const submission = getSubmission(instance.id);
    if (!template || !submission) return template?.fields?.filter(f => f.required) || [];
    const answered = submission.normalized_answers || {};
    return (template.fields || []).filter(f => f.required && !answered[f.field_key]);
  };

  if (isLoading) return <div className="text-center py-8 text-gray-400">Loading forms...</div>;

  return (
    <div className="space-y-4">
      <div className="flex items-center justify-between">
        <div>
          <h4 className="font-semibold text-black">Forms Sent to Promoter</h4>
          <p className="text-sm text-gray-500">{instances.length} form{instances.length !== 1 ? 's' : ''} sent</p>
        </div>
        <Button size="sm" onClick={() => setSendDialogOpen(true)} className="bg-yellow-400 text-black hover:bg-yellow-500">
          <Send className="w-4 h-4 mr-1.5" />
          Send New Form
        </Button>
      </div>

      {instances.length === 0 ? (
        <Card className="border-dashed">
          <CardContent className="py-10 text-center">
            <Send className="w-8 h-8 text-gray-300 mx-auto mb-3" />
            <p className="text-gray-500 font-medium">No forms sent yet</p>
            <p className="text-gray-400 text-sm mt-1">Send a form to collect materials from the promoter</p>
            <Button size="sm" className="mt-4 bg-yellow-400 text-black hover:bg-yellow-500" onClick={() => setSendDialogOpen(true)}>
              Send First Form
            </Button>
          </CardContent>
        </Card>
      ) : (
        <div className="rounded-lg border border-gray-200 overflow-hidden">
          <table className="w-full text-sm">
            <thead className="bg-gray-50 border-b border-gray-200">
              <tr>
                <th className="text-left px-4 py-2.5 text-xs font-semibold text-gray-500 uppercase tracking-wider">Form</th>
                <th className="text-left px-4 py-2.5 text-xs font-semibold text-gray-500 uppercase tracking-wider">Recipient</th>
                <th className="text-left px-4 py-2.5 text-xs font-semibold text-gray-500 uppercase tracking-wider">Sent</th>
                <th className="text-left px-4 py-2.5 text-xs font-semibold text-gray-500 uppercase tracking-wider">Status</th>
                <th className="text-left px-4 py-2.5 text-xs font-semibold text-gray-500 uppercase tracking-wider">Notes</th>
                <th className="px-4 py-2.5"></th>
              </tr>
            </thead>
            <tbody className="divide-y divide-gray-100">
              {instances.map(instance => {
                const submission = getSubmission(instance.id);
                const template = getTemplate(instance.template_id);
                const missing = getMissingFields(instance);
                const isCompleted = instance.status === 'completed';

                return (
                  <tr key={instance.id} className="bg-white hover:bg-gray-50 transition-colors">
                    <td className="px-4 py-3 font-medium text-black">{template?.name || 'Unknown'}</td>
                    <td className="px-4 py-3 text-gray-600">
                      <div>{instance.recipient_name || '—'}</div>
                      <div className="text-xs text-gray-400">{instance.recipient_email}</div>
                    </td>
                    <td className="px-4 py-3 text-gray-500 text-xs whitespace-nowrap">
                      {instance.sent_at ? new Date(instance.sent_at).toLocaleDateString() : '—'}
                    </td>
                    <td className="px-4 py-3">
                      <Badge className={isCompleted
                        ? 'bg-green-100 text-green-700 border-green-200'
                        : instance.status === 'expired'
                        ? 'bg-gray-100 text-gray-500'
                        : 'bg-yellow-100 text-yellow-700 border-yellow-200'
                      }>
                        {isCompleted
                          ? <><CheckCircle2 className="w-3 h-3 mr-1" />Completed</>
                          : instance.status === 'expired' ? 'Expired'
                          : <><Clock className="w-3 h-3 mr-1" />Awaiting</>}
                      </Badge>
                    </td>
                    <td className="px-4 py-3 max-w-[200px]">
                      {isCompleted && missing.length > 0 && (
                        <div className="flex items-start gap-1 text-xs text-amber-700">
                          <AlertTriangle className="w-3.5 h-3.5 flex-shrink-0 mt-0.5" />
                          <span>{missing.length} field{missing.length > 1 ? 's' : ''} missing: {missing.map(f => f.label).join(', ')}</span>
                        </div>
                      )}
                      {isCompleted && missing.length === 0 && (
                        <span className="text-xs text-green-600 flex items-center gap-1">
                          <CheckCircle2 className="w-3.5 h-3.5" /> All fields filled
                        </span>
                      )}
                    </td>
                    <td className="px-4 py-3">
                      <div className="flex items-center gap-1 justify-end">
                        {submission && (
                          <Button size="sm" variant="ghost" className="h-7 text-xs px-2" onClick={() => setViewingInstance(instance)}>
                            <Eye className="w-3.5 h-3.5 mr-1" /> View
                          </Button>
                        )}
                        {(!isCompleted || (isCompleted && missing.length > 0)) && (
                          <Button
                            size="sm"
                            variant="ghost"
                            className={`h-7 text-xs px-2 ${isCompleted && missing.length > 0 ? 'text-amber-600 hover:bg-amber-50' : ''}`}
                            onClick={() => openRemindDialog(instance)}
                          >
                            <RefreshCw className="w-3.5 h-3.5 mr-1" />
                            {isCompleted ? 'Follow up' : 'Remind'}
                          </Button>
                        )}
                        <Button
                          size="sm"
                          variant="ghost"
                          className="h-7 text-xs px-2 text-red-500 hover:bg-red-50 hover:text-red-600"
                          disabled={removeMutation.isPending}
                          onClick={() => removeMutation.mutate(instance)}
                        >
                          <Trash2 className="w-3.5 h-3.5" />
                        </Button>
                      </div>
                    </td>
                  </tr>
                );
              })}
            </tbody>
          </table>
        </div>
      )}

      {/* View Submission Modal */}
      {viewingInstance && (
        <Dialog open={!!viewingInstance} onOpenChange={() => setViewingInstance(null)}>
          <DialogContent className="max-w-xl max-h-[80vh] overflow-y-auto">
            <DialogHeader>
              <DialogTitle>Submission from {viewingInstance.recipient_name || viewingInstance.recipient_email}</DialogTitle>
            </DialogHeader>
            <SubmissionDetails instance={viewingInstance} submission={getSubmission(viewingInstance.id)} template={getTemplate(viewingInstance.template_id)} />
          </DialogContent>
        </Dialog>
      )}

      {/* Remind Dialog */}
      {remindInstance && (
        <Dialog open={!!remindInstance} onOpenChange={() => setRemindInstance(null)}>
          <DialogContent className="max-w-lg">
            <DialogHeader>
              <DialogTitle>Send Reminder to {remindInstance.recipient_name || remindInstance.recipient_email}</DialogTitle>
            </DialogHeader>
            <div className="space-y-3">
              <p className="text-sm text-gray-500">Edit the message below before sending:</p>
              <Textarea
                value={remindEmailBody}
                onChange={(e) => setRemindEmailBody(e.target.value)}
                rows={12}
                className="text-sm font-mono"
              />
            </div>
            <DialogFooter className="mt-4">
              <Button variant="outline" onClick={() => setRemindInstance(null)}>Cancel</Button>
              <Button
                className="bg-yellow-400 text-black hover:bg-yellow-500"
                disabled={resendMutation.isPending}
                onClick={() => resendMutation.mutate({ instance: remindInstance, body: remindEmailBody })}
              >
                <Send className="w-4 h-4 mr-1.5" />
                {resendMutation.isPending ? 'Sending...' : 'Send Reminder'}
              </Button>
            </DialogFooter>
          </DialogContent>
        </Dialog>
      )}

      {/* Send Form Dialog */}
      <Dialog open={sendDialogOpen} onOpenChange={setSendDialogOpen}>
        <DialogContent className="max-w-lg">
          <DialogHeader>
            <DialogTitle>Send Form</DialogTitle>
          </DialogHeader>
          <FormSender show={show} artist={artist} onClose={() => {
            setSendDialogOpen(false);
            queryClient.invalidateQueries({ queryKey: ['form-instances', show.id] });
          }} />
        </DialogContent>
      </Dialog>
    </div>
  );
}

function SubmissionDetails({ instance, submission, template }) {
  if (!submission) return <p className="text-gray-500">No submission found.</p>;

  const answers = submission.normalized_answers || submission.raw_answers || {};
  const fields = template?.fields || [];

  return (
    <div className="space-y-4">
      <div className="text-xs text-gray-500">
        Submitted {submission.submitted_at ? new Date(submission.submitted_at).toLocaleString() : 'Unknown'}
      </div>
      {fields.map(field => {
        const val = answers[field.field_key];
        return (
          <div key={field.field_key} className={`pb-3 border-b last:border-0 ${!val ? 'opacity-40' : ''}`}>
            <p className="text-xs font-semibold text-gray-500 uppercase tracking-wider">{field.label}</p>
            {val ? (
              field.type === 'url' || field.type === 'file_link' ? (
                <a href={val} target="_blank" rel="noreferrer" className="text-sm text-blue-600 hover:underline break-all">{val}</a>
              ) : (
                <p className="text-sm text-black mt-0.5 break-words">{val}</p>
              )
            ) : (
              <p className="text-sm text-gray-400 italic mt-0.5">{field.required ? '⚠ Missing (required)' : 'Not provided'}</p>
            )}
          </div>
        );
      })}
    </div>
  );
}
