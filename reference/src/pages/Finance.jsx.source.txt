import React, { useState, useEffect } from 'react';
import { base44 } from '@/api/base44Client';
import { useLanguage } from '@/lib/LanguageContext';
import { Routes, Route, Navigate, NavLink, useLocation } from 'react-router-dom';
import {
  LayoutDashboard, Receipt, CreditCard, Plug, Mail,
  ListChecks, Settings2, ShieldAlert, BarChart3, Landmark, ShieldCheck
} from 'lucide-react';

import FinanceDashboard from './finance/FinanceDashboard';
import InvoiceManagement from './finance/InvoiceManagement';
import PaymentTracking from './finance/PaymentTracking';
import IntegrationHub from './finance/IntegrationHub';
import FinanceSettings from './finance/FinanceSettings';
import EmailTemplates from './finance/EmailTemplates';
import ActivityLogPage from './finance/ActivityLogPage';
import BankReconciliation from './finance/BankReconciliation';
import AuditHealthDashboard from './finance/AuditHealthDashboard';
import { FinanceOrgFilterProvider } from '@/lib/FinanceOrgFilter';
import FinanceOrgFilterBar from '@/components/finance/FinanceOrgFilterBar';

const NAV_ITEMS = [
  { nameKey: 'finance_dashboard',       path: 'dashboard',      icon: LayoutDashboard, roles: ['Finance Viewer', 'Finance Manager'] },
  { nameKey: 'finance_invoices',        path: 'invoices',        icon: Receipt,         roles: ['Finance Viewer', 'Finance Manager'] },
  { nameKey: 'finance_payments',        path: 'payments',        icon: CreditCard,      roles: ['Finance Viewer', 'Finance Manager'] },
  { nameKey: 'nav_tasks',               path: 'activity',        icon: ListChecks,      roles: ['Finance Viewer', 'Finance Manager'] },
  { nameKey: 'finance_integrations',    path: 'integrations',    icon: Plug,            roles: ['Finance Manager'] },
  { nameKey: 'finance_email_templates', path: 'email-templates', icon: Mail,            roles: ['Finance Manager'] },
  { nameKey: 'finance_reconciliation',  path: 'reconciliation',  icon: Landmark,        roles: ['Finance Manager'] },
  { nameKey: 'finance_audit',           path: 'audit',           icon: ShieldCheck,     roles: ['Finance Viewer', 'Finance Manager'] },
  { nameKey: 'nav_settings',            path: 'settings',        icon: Settings2,       roles: ['Finance Manager'] },
];

const REPORTS_LINK = '/reports';

export default function Finance() {
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(true);
  const { t } = useLanguage();
  const location = useLocation();

  useEffect(() => {
    base44.auth.me().then(u => { setUser(u); setLoading(false); }).catch(() => setLoading(false));
  }, []);

  if (loading) {
    return (
      <div className="flex items-center justify-center min-h-[60vh]">
        <div className="w-8 h-8 border-4 border-gray-200 border-t-black rounded-full animate-spin" />
      </div>
    );
  }

  const isAdmin = user?.role === 'admin';
  const financeRole = user?.finance_role;

  if (!isAdmin && !financeRole) {
    return (
      <div className="flex flex-col items-center justify-center min-h-[60vh] gap-4">
        <ShieldAlert className="w-16 h-16 text-gray-300" />
        <h2 className="text-2xl font-bold text-gray-800">{t('finance_no_access')}</h2>
        <p className="text-gray-500 text-center max-w-sm">
          {t('finance_contact_admin')}
        </p>
      </div>
    );
  }

  const effectiveRole = isAdmin ? 'Finance Manager' : financeRole;
  const visibleNav = NAV_ITEMS.filter(item => item.roles.includes(effectiveRole));

  return (
    <FinanceOrgFilterProvider>
    <div className="flex flex-col lg:flex-row min-h-[calc(100vh-120px)] -m-4 bg-slate-50 lg:-m-6">
      {/* Finance Sidebar */}
      <aside className="lg:w-56 bg-white border-b lg:border-b-0 lg:border-r border-slate-200 flex-shrink-0">
        <div className="p-4 border-b border-gray-100">
          <h2 className="text-base font-bold text-black tracking-tight">{t('finance_hub')}</h2>
          {effectiveRole && (
            <span className="text-[10px] font-medium text-yellow-700 bg-yellow-100 px-2 py-0.5 rounded-full mt-1 inline-block">
              {effectiveRole}
            </span>
          )}
        </div>
        <nav className="p-2 flex lg:flex-col flex-row gap-0.5 overflow-x-auto scrollbar-thin">
          {visibleNav.map(item => {
            const Icon = item.icon;
            const isActive = location.pathname.includes(`/finance/${item.path}`);
            return (
              <NavLink
                key={item.path}
                to={item.path}
                className={`flex min-h-11 items-center gap-2.5 px-3 py-2 rounded-lg text-sm font-medium transition-colors whitespace-nowrap ${
                  isActive
                    ? 'bg-yellow-400 text-black'
                    : 'text-gray-600 hover:bg-gray-100 hover:text-black'
                }`}
              >
                <Icon className="w-4 h-4 flex-shrink-0" />
                {t(item.nameKey)}
              </NavLink>
            );
          })}
          {/* Separator + Reports shortcut */}
          <div className="hidden lg:block border-t border-gray-100 my-1" />
          <NavLink
            to={REPORTS_LINK}
            className="flex items-center gap-2.5 px-3 py-2 rounded-lg text-sm font-medium transition-colors whitespace-nowrap text-gray-500 hover:bg-gray-100 hover:text-black"
          >
            <BarChart3 className="w-4 h-4 flex-shrink-0" />
            {t('nav_reports')} ↗
          </NavLink>
        </nav>
      </aside>

      {/* Main Content */}
      <main className="flex-1 min-w-0 overflow-auto bg-slate-50">
        <div className="sticky top-0 z-20 border-b border-slate-200 bg-white/95 px-4 py-3 backdrop-blur lg:px-6">
          <FinanceOrgFilterBar />
        </div>
        <div className="p-4 lg:p-6">
        <Routes>
          <Route index element={<Navigate to="dashboard" replace />} />
          <Route path="dashboard"      element={<FinanceDashboard financeRole={effectiveRole} />} />
          <Route path="invoices/*"     element={<InvoiceManagement financeRole={effectiveRole} />} />
          <Route path="payments"       element={<PaymentTracking financeRole={effectiveRole} />} />
          <Route path="activity"       element={<ActivityLogPage />} />
          <Route path="audit"          element={<AuditHealthDashboard />} />
          {effectiveRole === 'Finance Manager' && <>
            <Route path="integrations"    element={<IntegrationHub />} />
            <Route path="email-templates" element={<EmailTemplates />} />
            <Route path="settings"        element={<FinanceSettings />} />
            <Route path="reconciliation"  element={<BankReconciliation />} />
          </>}
          <Route path="*" element={<Navigate to="dashboard" replace />} />
        </Routes>
        </div>
      </main>
    </div>
    </FinanceOrgFilterProvider>
  );
}
