import React, { useState, useEffect, useMemo } from 'react';
import PullToRefresh from '../components/mobile/PullToRefresh';
import { base44 } from '@/api/base44Client';
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
import OrgAttributionBadge from '../components/shows/OrgAttributionBadge';
import { Button } from '@/components/ui/button';
import { Input } from '@/components/ui/input';
import { Card, CardContent } from '@/components/ui/card';
import { Badge } from '@/components/ui/badge';
import { Plus, Search, Calendar, MapPin, Eye, List, Table, Loader2, ChevronLeft, ChevronRight, Pencil, Lock, Kanban } from 'lucide-react';
import ShowsPipelineBoard from '../components/shows/ShowsPipelineBoard';
import ShowStatusChip, { canConfirmShow } from '../components/shows/ShowStatusChip';
import { auditStamp } from '../lib/showEdit';
import { toast } from 'sonner';
import { Link, useNavigate } from 'react-router-dom';
import { createPageUrl } from '../utils';
import ShowForm from '../components/shows/ShowForm';
import useArtistLookup from '../hooks/useArtistLookup';
import { useOrg } from '../lib/OrgContext';
import { filterShowsForOrg, buildStubArtistIds } from '../lib/orgUtils';
import {
  Dialog,
  DialogContent,
  DialogHeader,
  DialogTitle,
} from '@/components/ui/dialog';
import { Label } from '@/components/ui/label';
import { Textarea } from '@/components/ui/textarea';
import {
  Select,
  SelectContent,
  SelectItem,
  SelectTrigger,
  SelectValue,
} from '@/components/ui/select';

export default function Shows() {
  const [searchQuery, setSearchQuery] = useState('');
  const [showForm, setShowForm] = useState(false);
  const [editingShow, setEditingShow] = useState(null);
  const [view, setView] = useState('list');
  const [filterArtist, setFilterArtist] = useState('all');
  const [filterStatus, setFilterStatus] = useState('all');
  const [selectedDate, setSelectedDate] = useState(null);
  const [currentCalendarDate, setCurrentCalendarDate] = useState(new Date());
  const [taskDialogOpen, setTaskDialogOpen] = useState(false);
  const [newTask, setNewTask] = useState({ title: '', description: '', assigned_to: '', priority: 'medium' });
  const [selectedShow, setSelectedShow] = useState(null);
  const queryClient = useQueryClient();
  const navigate = useNavigate();
  const [currentUser, setCurrentUser] = useState(null);
  const canViewFinances = currentUser?.role === 'admin' ||
    currentUser?.finance_role === 'Finance Manager' ||
    currentUser?.finance_role === 'Finance Viewer';

  useEffect(() => {
    let mounted = true;
    base44.auth.me().then(u => { if (mounted) setCurrentUser(u); }).catch(() => {});
    return () => { mounted = false; };
  }, []);

  const { data: users = [] } = useQuery({
    queryKey: ['users'],
    queryFn: () => base44.entities.User.list(),
  });

  const { activeOrg } = useOrg();
  const { data: rawShows = [], isLoading } = useQuery({
    queryKey: ['shows'],
    queryFn: () => base44.entities.Show.list('-date', 300),
  });

  // Load visibility records to filter hidden shows (Elevate unlink)
  const { data: hiddenShows = [] } = useQuery({
    queryKey: ['show-org-visibility'],
    queryFn: () => base44.entities.ShowOrgVisibility.filter({ org_id: 'elevate', is_hidden: true }),
  });
  const hiddenShowIds = new Set((hiddenShows || []).map(v => v.show_id));

  const { artists, allArtists } = useArtistLookup();
  const abossStubArtistIds = useMemo(() => buildStubArtistIds(allArtists), [allArtists]);
  const shows = useMemo(() => filterShowsForOrg(rawShows || [], activeOrg, abossStubArtistIds), [rawShows, activeOrg, abossStubArtistIds]);

  const { data: contracts = [] } = useQuery({
    queryKey: ['contracts'],
    queryFn: () => base44.entities.Contract.list('-created_date', 200),
  });

  const { data: tours = [] } = useQuery({
    queryKey: ['tours'],
    queryFn: () => base44.entities.Tour.list('-start_date', 100),
  });

  const { data: tasks = [] } = useQuery({
    queryKey: ['tasks'],
    queryFn: () => base44.entities.Task.list('-created_date', 200),
  });

  const logShowActivity = async (actionType, label, result, errorMessage = null) => {
    try {
      await base44.entities.ActivityLog.create({
        timestamp: new Date().toISOString(),
        action_type: actionType,
        actor_id: currentUser?.id || 'unknown',
        actor_label: currentUser?.full_name || currentUser?.email || 'Unknown',
        affected_entity_type: 'Show',
        affected_entity_id: 'pending',
        affected_entity_label: label,
        result,
        error_message: errorMessage,
      });
    } catch (logErr) {
      console.warn('[Shows] ActivityLog write failed:', logErr);
    }
  };

  const createMutation = useMutation({
    mutationFn: (data) => base44.entities.Show.create(data),
    onSuccess: (show) => {
      queryClient.invalidateQueries({ queryKey: ['shows'] });
      setShowForm(false);
      setEditingShow(null);
      toast.success('Show created successfully.');
      logShowActivity('Show Created', show?.event_name || 'New Show', 'Success');
    },
    onError: (err) => {
      const msg = err?.message || err?.toString() || 'Unknown error';
      toast.error(`Could not create show: ${msg}`);
      logShowActivity('Show Create Failed', 'New Show', 'Failure', msg);
    },
  });

  const updateMutation = useMutation({
    mutationFn: ({ id, data }) => base44.entities.Show.update(id, { ...data, ...auditStamp(currentUser, activeOrg) }),
    onSuccess: (show) => {
      queryClient.invalidateQueries({ queryKey: ['shows'] });
      setShowForm(false);
      setEditingShow(null);
      toast.success('Show updated successfully.');
      logShowActivity('Show Updated', show?.event_name || 'Show', 'Success');
    },
    onError: (err) => {
      const msg = err?.message || err?.toString() || 'Unknown error';
      toast.error(`Could not update show: ${msg}`);
      logShowActivity('Show Update Failed', editingShow?.event_name || 'Show', 'Failure', msg);
    },
  });

  const createTaskMutation = useMutation({
    mutationFn: (data) => base44.entities.Task.create(data),
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ['tasks'] });
      setTaskDialogOpen(false);
      setNewTask({ title: '', description: '', assigned_to: '', priority: 'medium' });
      setSelectedShow(null);
      toast.success('Task created successfully.');
    },
    onError: () => toast.error('Could not create the task. Please try again.'),
  });

  const handleSubmit = async (data) => {
    if (editingShow) {
      return updateMutation.mutateAsync({ id: editingShow.id, data });
    }
    return createMutation.mutateAsync(data);
  };

  const handleCreateTask = (show) => {
    setSelectedShow(show);
    const artist = artists.find((a) => a.id === show.artist_id);
    setNewTask({
      title: `Task for ${show.event_name}`,
      description: `Related to ${show.event_name} at ${show.venue}`,
      assigned_to: '',
      priority: 'medium',
      show_id: show.id,
      artist_id: show.artist_id,
    });
    setTaskDialogOpen(true);
  };

  const submitTask = () => {
    createTaskMutation.mutate(newTask);
  };

  const filteredShows = shows.filter((show) => {
    // Exclude shows hidden by Elevate (ShowOrgVisibility unlink)
    if (hiddenShowIds.has(show.id)) return false;
    const matchesSearch = show.event_name?.toLowerCase().includes(searchQuery.toLowerCase()) ||
      show.venue?.toLowerCase().includes(searchQuery.toLowerCase()) ||
      show.city?.toLowerCase().includes(searchQuery.toLowerCase());
    const matchesArtist = filterArtist === 'all' || show.artist_id === filterArtist;
    const matchesStatus = filterStatus === 'all' || show.status === filterStatus;
    return matchesSearch && matchesArtist && matchesStatus;
  });

  const now = new Date();
  const upcomingShows = filteredShows.filter((s) => { const d = new Date(s.date); return !isNaN(d) && d >= now; });
  const pastShows = filteredShows.filter((s) => { const d = new Date(s.date); return !isNaN(d) && d < now; });

  const getDaysInMonth = () => {
    const year = currentCalendarDate.getFullYear();
    const month = currentCalendarDate.getMonth();
    const firstDay = new Date(year, month, 1);
    const lastDay = new Date(year, month + 1, 0);
    const daysInMonth = lastDay.getDate();
    const startingDayOfWeek = firstDay.getDay();

    const days = [];
    for (let i = 0; i < startingDayOfWeek; i++) {
      days.push(null);
    }
    for (let i = 1; i <= daysInMonth; i++) {
      days.push(i);
    }
    return days;
  };

  const getShowsForDate = (day) => {
    if (!day) return [];
    const year = currentCalendarDate.getFullYear();
    const month = currentCalendarDate.getMonth();
    const dateStr = `${year}-${String(month + 1).padStart(2, '0')}-${String(day).padStart(2, '0')}`;
    return filteredShows.filter((show) => show.date === dateStr);
  };

  const calendarMonthLabel = currentCalendarDate.toLocaleDateString('en-US', {
    month: 'long',
    year: 'numeric',
  });

  const moveCalendarMonth = (offset) => {
    setCurrentCalendarDate((date) => new Date(date.getFullYear(), date.getMonth() + offset, 1));
    setSelectedDate(null);
  };

  const statusColors = {
    confirmed: 'bg-green-100 text-green-700 border-green-300',
    pending: 'bg-yellow-100 text-yellow-700 border-yellow-300',
    cancelled: 'bg-red-100 text-red-700 border-red-300',
    completed: 'bg-blue-100 text-blue-700 border-blue-300',
  };

  const renderShowCard = (show, idx) => {
    const artist = allArtists.find((a) => a.id === show.artist_id);
    const tour = tours.find((t) => t.id === show.tour_id);
    const showDate = !isNaN(new Date(show.date)) ? new Date(show.date) : null;
    const dayNum = showDate?.getDate();
    const monthStr = showDate?.toLocaleDateString('en-US', { month: 'short' });
    const weekdayStr = showDate?.toLocaleDateString('en-US', { weekday: 'short' });

    return (
      <Link
        key={show.id}
        to={createPageUrl('ShowDetail') + '?id=' + show.id}
        className="block"
        data-tour={idx === 0 ? 'shows-first-row' : undefined}
      >
        <div className="bg-white border border-gray-200 hover:border-yellow-400 rounded-2xl overflow-hidden transition-colors active:bg-gray-50">
          {/* Top row: date pill + main info + status */}
          <div className="flex items-start gap-3 p-4">
            {/* Date pill */}
            <div className="flex-shrink-0 w-12 bg-gray-50 rounded-xl text-center py-2 border border-gray-100">
              <p className="text-[10px] font-semibold text-gray-400 uppercase leading-none">{weekdayStr ?? '—'}</p>
              <p className="text-xl font-bold text-gray-900 leading-tight tabular-nums">{dayNum ?? '?'}</p>
              <p className="text-[10px] font-semibold text-gray-400 uppercase leading-none">{monthStr ?? ''}</p>
            </div>

            {/* Main info */}
            <div className="flex-1 min-w-0">
              <div className="flex items-start justify-between gap-2">
                <h3 className="font-bold text-gray-900 text-base leading-snug truncate flex items-center gap-1.5">
                  {show.event_name}
                  {show.is_locked && <Lock className="w-3 h-3 text-gray-400 flex-shrink-0" />}
                </h3>
                <ShowStatusChip show={show} />
              </div>

              <p className="text-sm text-gray-500 mt-0.5 truncate">
                {artist?._deleted ? <span className="italic">{artist?.name} (deleted)</span> : (artist?.name || 'Unknown Artist')}
              </p>

              <div className="flex items-center gap-1 mt-1.5 text-xs text-gray-400">
                <MapPin className="w-3 h-3 flex-shrink-0" aria-hidden="true" />
                <span className="truncate">{[show.venue, show.city, show.country].filter(Boolean).join(', ')}</span>
              </div>

              {show.time && (
                <p className="text-xs text-gray-400 mt-0.5">🕐 {show.time}</p>
              )}
            </div>
          </div>

          {/* Bottom row: stats strip */}
          <div className="flex items-center border-t border-gray-100 divide-x divide-gray-100">
            {show.guarantee_fee > 0 && canViewFinances && (
              <div className="flex-1 px-3 py-2 text-center">
                <p className="text-[10px] text-gray-400 uppercase tracking-wide">Fee</p>
                <p className="text-sm font-bold text-gray-900 tabular-nums leading-tight">
                  {show.guarantee_fee.toLocaleString()} <span className="text-[10px] font-normal text-gray-400">{show.currency || 'EUR'}</span>
                </p>
              </div>
            )}
            {show.capacity > 0 && (
              <div className="flex-1 px-3 py-2 text-center">
                <p className="text-[10px] text-gray-400 uppercase tracking-wide">Cap.</p>
                <p className="text-sm font-bold text-gray-900 tabular-nums leading-tight">{show.capacity.toLocaleString()}</p>
              </div>
            )}
            <div className="flex-1 px-3 py-2 text-center">
              <p className="text-[10px] text-gray-400 uppercase tracking-wide">Visa</p>
              <p className={`text-xs font-semibold leading-tight ${show.visa_required ? 'text-orange-600' : 'text-green-600'}`}>
                {show.visa_required ? (show.visa_status || 'Required') : '✓ OK'}
              </p>
            </div>
            <div className="flex-1 px-3 py-2 text-center">
              <p className="text-[10px] text-gray-400 uppercase tracking-wide">Flyer</p>
              <p className={`text-xs font-semibold leading-tight ${show.flyer_approved ? 'text-green-600' : 'text-gray-400'}`}>
                {show.flyer_approved ? '✓ Done' : 'Pending'}
              </p>
            </div>
            {/* Edit button */}
            {!show.is_locked && (
              <button
                className="px-3 py-2 text-gray-400 hover:text-gray-700 active:bg-gray-50 transition-colors"
                aria-label={`Edit ${show.event_name}`}
                onClick={(e) => {
                  e.preventDefault();
                  e.stopPropagation();
                  setEditingShow(show);
                  setShowForm(true);
                }}
              >
                <Pencil className="w-3.5 h-3.5" aria-hidden="true" />
              </button>
            )}
          </div>

          {/* Tour badge — only if assigned */}
          {tour && (
            <div className="px-4 pb-3">
              <span className="inline-flex items-center gap-1 text-[11px] bg-yellow-50 border border-yellow-200 text-yellow-700 px-2 py-0.5 rounded-full">
                🗺️ {tour.name}
              </span>
            </div>
          )}
        </div>
      </Link>
    );
  };

  const handleRefresh = async () => {
    await queryClient.invalidateQueries({ queryKey: ['shows'] });
  };

  return (
    <PullToRefresh onRefresh={handleRefresh}>
    <div className="space-y-6">
      {/* Header */}
      <div className="flex flex-col gap-3 border-b-4 border-yellow-400 pb-4">
        <div className="flex items-center justify-between">
          <div>
            <h1 className="text-2xl sm:text-4xl font-bold text-black">Shows & Tours</h1>
            <p className="text-gray-600 mt-0.5 text-sm">{upcomingShows.length} upcoming shows</p>
          </div>
          <Button
            data-tour="shows-create-btn"
            onClick={() => {
              setEditingShow(null);
              setShowForm(true);
            }}
            className="bg-black text-yellow-400 hover:bg-gray-900 font-semibold min-h-[44px]"
          >
            <Plus className="w-5 h-5 mr-1.5" aria-hidden="true" />
            <span className="hidden sm:inline">Add Show</span>
            <span className="sm:hidden">Add</span>
          </Button>
        </div>
        {/* View toggle — compact pill row */}
        <div className="flex items-center gap-1 bg-gray-100 rounded-xl p-1 self-start">
          {[
            { id: 'list',     icon: List,     label: 'Cards' },
            { id: 'table',    icon: Table,    label: 'Table' },
            { id: 'board',    icon: Kanban,   label: 'Board' },
            { id: 'calendar', icon: Calendar, label: 'Cal' },
          ].map(({ id, icon: Icon, label }) => (
            <button
              key={id}
              onClick={() => setView(id)}
              aria-pressed={view === id}
              className={`flex items-center gap-1 px-3 py-1.5 rounded-lg text-xs font-semibold transition-colors min-h-[36px] ${
                view === id ? 'bg-yellow-400 text-black shadow-sm' : 'text-gray-500 hover:text-gray-800'
              }`}
            >
              <Icon className="w-3.5 h-3.5" aria-hidden="true" />
              {label}
            </button>
          ))}
        </div>
      </div>

      {/* Search & Filters */}
      <div className="flex flex-col gap-2">
        <div className="relative flex-1">
          <Search className="absolute left-3 top-1/2 -translate-y-1/2 text-gray-400 w-4 h-4 pointer-events-none" aria-hidden="true" />
          <Input
            placeholder="Search shows, venues, cities…"
            value={searchQuery}
            onChange={(e) => setSearchQuery(e.target.value)}
            className="pl-9 border-2 border-gray-200 focus:border-yellow-400 h-10"
          />
        </div>
        <div className="flex gap-2">
          <Select value={filterArtist} onValueChange={setFilterArtist}>
            <SelectTrigger className="flex-1 border-2 border-gray-200 focus:border-yellow-400 h-10">
              <SelectValue placeholder="All Artists" />
            </SelectTrigger>
            <SelectContent>
              <SelectItem value="all">All Artists</SelectItem>
              {artists.map((artist) => (
                <SelectItem key={artist.id} value={artist.id}>{artist.name}</SelectItem>
              ))}
            </SelectContent>
          </Select>
          <Select value={filterStatus} onValueChange={setFilterStatus}>
            <SelectTrigger className="flex-1 border-2 border-gray-200 focus:border-yellow-400 h-10">
              <SelectValue placeholder="All Statuses" />
            </SelectTrigger>
            <SelectContent>
              <SelectItem value="all">All Statuses</SelectItem>
              <SelectItem value="confirmed">Confirmed</SelectItem>
              <SelectItem value="pending">Pending</SelectItem>
              <SelectItem value="cancelled">Cancelled</SelectItem>
              <SelectItem value="completed">Completed</SelectItem>
            </SelectContent>
          </Select>
        </div>
      </div>

      {/* Show Form Modal */}
      {showForm && (
        <div className="fixed inset-0 bg-black bg-opacity-50 z-50 flex items-center justify-center p-4">
          <div className="bg-white rounded-lg max-w-4xl w-full max-h-[90vh] overflow-y-auto">
            <ShowForm
              show={editingShow}
              artists={artists}
              contracts={contracts}
              tours={tours}
              onSubmit={handleSubmit}
              isSubmitting={createMutation.isPending || updateMutation.isPending}
              onCancel={() => {
                setShowForm(false);
                setEditingShow(null);
              }}
            />
          </div>
        </div>
      )}

      {/* Task Creation Dialog */}
      <Dialog open={taskDialogOpen} onOpenChange={setTaskDialogOpen}>
        <DialogContent>
          <DialogHeader>
            <DialogTitle>Create Task</DialogTitle>
          </DialogHeader>
          <div className="space-y-4 mt-4">
            <div>
              <Label>Task Title</Label>
              <Input
                value={newTask.title}
                onChange={(e) => setNewTask({ ...newTask, title: e.target.value })}
                className="mt-1"
              />
            </div>
            <div>
              <Label>Description</Label>
              <Textarea
                value={newTask.description}
                onChange={(e) => setNewTask({ ...newTask, description: e.target.value })}
                rows={3}
                className="mt-1"
              />
            </div>
            <div>
              <Label>Assign To</Label>
              <Select
                value={newTask.assigned_to}
                onValueChange={(value) => setNewTask({ ...newTask, assigned_to: value })}
              >
                <SelectTrigger className="mt-1">
                  <SelectValue placeholder="Select team member" />
                </SelectTrigger>
                <SelectContent>
                  {users.map((user) => (
                    <SelectItem key={user.id} value={user.id}>
                      {user.full_name}
                    </SelectItem>
                  ))}
                </SelectContent>
              </Select>
            </div>
            <div>
              <Label>Priority</Label>
              <Select
                value={newTask.priority}
                onValueChange={(value) => setNewTask({ ...newTask, priority: value })}
              >
                <SelectTrigger className="mt-1">
                  <SelectValue />
                </SelectTrigger>
                <SelectContent>
                  <SelectItem value="low">Low</SelectItem>
                  <SelectItem value="medium">Medium</SelectItem>
                  <SelectItem value="high">High</SelectItem>
                  <SelectItem value="urgent">Urgent</SelectItem>
                </SelectContent>
              </Select>
            </div>
            <div className="flex justify-end gap-2 pt-4">
              <Button variant="outline" onClick={() => setTaskDialogOpen(false)}>
                Cancel
              </Button>
              <Button 
                onClick={submitTask}
                disabled={createTaskMutation.isPending || !newTask.title?.trim()}
                className="bg-yellow-400 text-black hover:bg-yellow-500"
              >
                {createTaskMutation.isPending ? <><Loader2 className="w-4 h-4 mr-2 animate-spin" />Creating...</> : 'Create Task'}
              </Button>
            </div>
          </div>
        </DialogContent>
      </Dialog>

      {/* Shows List or Calendar */}
      {isLoading ? (
        <div className="text-center py-12">
          <p className="text-gray-500">Loading shows...</p>
        </div>
      ) : view === 'calendar' ? (
        /* Calendar View */
        <Card className="border-2 border-gray-200">
          <CardContent className="p-4">
            <div className="flex flex-col sm:flex-row sm:items-center justify-between gap-3 mb-4">
              <div>
                <h2 className="text-2xl font-bold text-black">{calendarMonthLabel}</h2>
                <p className="text-sm text-gray-500">{filteredShows.length} shows match your filters</p>
              </div>
              <div className="flex items-center gap-2">
                <Button variant="outline" size="sm" onClick={() => moveCalendarMonth(-1)}>
                  <ChevronLeft className="w-4 h-4" /> Previous
                </Button>
                <Button variant="outline" size="sm" onClick={() => { setCurrentCalendarDate(new Date()); setSelectedDate(null); }}>
                  Today
                </Button>
                <Button variant="outline" size="sm" onClick={() => moveCalendarMonth(1)}>
                  Next <ChevronRight className="w-4 h-4" />
                </Button>
              </div>
            </div>
            <div className="grid grid-cols-7 gap-2 mb-2">
              {['Sun', 'Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat'].map((day) => (
                <div key={day} className="text-center font-bold text-sm text-gray-600 py-2">
                  {day}
                </div>
              ))}
            </div>
            <div className="grid grid-cols-7 gap-2">
              {getDaysInMonth().map((day, index) => {
                const dayShows = getShowsForDate(day);
                return (
                  <div
                    key={index}
                    className={`min-h-24 p-2 border-2 rounded-lg ${
                      !day
                        ? 'bg-gray-50 border-gray-100'
                        : dayShows.length > 0
                        ? 'border-yellow-400 bg-yellow-50 cursor-pointer hover:bg-yellow-100'
                        : 'border-gray-200'
                    } transition`}
                    onClick={() => day && dayShows.length > 0 && setSelectedDate(day)}
                  >
                    {day && (
                      <>
                        <div className="text-sm font-semibold mb-1 text-gray-700">{day}</div>
                        <div className="space-y-1">
                          {dayShows.slice(0, 2).map((show) => {
                            const artist = allArtists.find((a) => a.id === show.artist_id);
                            return (
                              <div
                                key={show.id}
                                className="text-xs p-1 bg-black text-yellow-400 rounded truncate"
                              >
                                {artist?.name}
                              </div>
                            );
                          })}
                          {dayShows.length > 2 && (
                            <div className="text-xs text-gray-600">+{dayShows.length - 2} more</div>
                          )}
                        </div>
                      </>
                    )}
                  </div>
                );
              })}
            </div>
            {selectedDate && (
              <div className="mt-6 border-t-2 border-yellow-400 pt-4">
                <h3 className="text-xl font-bold mb-4">
                  Shows on {currentCalendarDate.toLocaleString('default', { month: 'long' })} {selectedDate}
                </h3>
                <div className="space-y-3">
                  {getShowsForDate(selectedDate).map((show) => {
                    const artist = allArtists.find((a) => a.id === show.artist_id);
                    return (
                      <div
                        key={show.id}
                        className="p-4 border-2 border-gray-200 rounded-lg hover:border-yellow-400 transition cursor-pointer"
                        onClick={() => {
                          setEditingShow(show);
                          setShowForm(true);
                        }}
                      >
                        <h4 className="font-bold text-black">{show.event_name}</h4>
                        <p className="text-sm text-gray-600">{artist?.name}{artist?._deleted ? ' (deleted)' : ''}</p>
                        <p className="text-sm text-gray-600">{show.venue}, {show.city}</p>
                      </div>
                    );
                  })}
                </div>
              </div>
            )}
          </CardContent>
        </Card>
      ) : view === 'board' ? (
        /* Pipeline Board View */
        <ShowsPipelineBoard
          shows={filteredShows}
          artistMap={Object.fromEntries(allArtists.map(a => [a.id, a]))}
          canViewFinances={canViewFinances}
          onCardClick={(show) => navigate(createPageUrl('ShowDetail') + '?id=' + show.id)}
          onStatusChange={(id, status) => {
            if (['confirmed', 'completed'].includes(status) && !canConfirmShow(currentUser)) {
              toast.error("You don't have permission to confirm or complete shows.");
              return;
            }
            updateMutation.mutate({ id, data: { status } });
          }}
        />
      ) : view === 'table' ? (
        /* Table View */
        <div className="border-2 border-gray-200 rounded-lg">
          <div className="overflow-x-auto scrollbar-thin rounded-lg">
            <table className="w-full min-w-[1100px]">
              <thead className="bg-black text-yellow-400">
                <tr>
                  <th className="text-left py-3 px-4 font-semibold text-sm border-r border-gray-700">Date</th>
                  <th className="text-left py-3 px-4 font-semibold text-sm border-r border-gray-700">Event</th>
                  <th className="text-left py-3 px-4 font-semibold text-sm border-r border-gray-700">Artist</th>
                  <th className="text-left py-3 px-4 font-semibold text-sm border-r border-gray-700">Tour</th>
                  <th className="text-left py-3 px-4 font-semibold text-sm border-r border-gray-700">Venue</th>
                  <th className="text-left py-3 px-4 font-semibold text-sm border-r border-gray-700">City</th>
                  <th className="text-left py-3 px-4 font-semibold text-sm border-r border-gray-700">Country</th>
                  <th className="text-right py-3 px-4 font-semibold text-sm border-r border-gray-700">Guarantee</th>
                  <th className="text-center py-3 px-4 font-semibold text-sm border-r border-gray-700">Status</th>
                  <th className="text-center py-3 px-4 font-semibold text-sm">Actions</th>
                </tr>
              </thead>
              <tbody>
                {filteredShows.map((show, idx) => {
                  const artist = allArtists.find((a) => a.id === show.artist_id);
                  const showTour = tours.find((t) => t.id === show.tour_id);
                  const isUpcoming = !isNaN(new Date(show.date)) && new Date(show.date) >= new Date();
                  return (
                    <tr 
                      key={show.id} 
                      className={`border-b border-gray-200 hover:bg-yellow-50 transition ${!isUpcoming ? 'opacity-50' : ''}`}
                    >
                      <td className="py-3 px-4 text-sm border-r border-gray-100 whitespace-nowrap">
                        {!isNaN(new Date(show.date)) ? new Date(show.date).toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' }) : 'TBD'}
                        {show.time && <div className="text-xs text-gray-500">{show.time}</div>}
                      </td>
                      <td className="py-3 px-4 text-sm font-medium border-r border-gray-100">
                        <Link 
                          to={createPageUrl('ShowDetail') + '?id=' + show.id}
                          className="hover:text-yellow-600 hover:underline inline-flex items-center gap-1"
                        >
                          {show.event_name}
                          {show.is_locked && <Lock className="w-3 h-3 text-gray-400 flex-shrink-0" />}
                        </Link>
                      </td>
                      <td className={`py-3 px-4 text-sm border-r border-gray-100 ${artist?._deleted ? 'italic text-gray-400' : ''}`}>{artist?.name || '-'}{artist?._deleted ? ' (deleted)' : ''}</td>
                      <td className="py-3 px-4 text-sm border-r border-gray-100">
                        {showTour ? (
                          <span className="text-xs bg-yellow-100 text-yellow-700 rounded px-2 py-0.5">{showTour.name}</span>
                        ) : '-'}
                      </td>
                      <td className="py-3 px-4 text-sm border-r border-gray-100">{show.venue || '-'}</td>
                      <td className="py-3 px-4 text-sm border-r border-gray-100">{show.city || '-'}</td>
                      <td className="py-3 px-4 text-sm border-r border-gray-100">{show.country || '-'}</td>
                      <td className="py-3 px-4 text-sm text-right font-semibold border-r border-gray-100">
                        {canViewFinances && show.guarantee_fee ? `${show.guarantee_fee.toLocaleString()} ${show.currency || 'EUR'}` : (canViewFinances ? '-' : '—')}
                      </td>
                      <td className="py-3 px-4 text-center border-r border-gray-100">
                        <ShowStatusChip show={show} />
                      </td>
                      <td className="py-3 px-4 text-center">
                        <div className="flex gap-1 justify-center">
                          {!show.is_locked && (
                            <Button
                              size="sm"
                              variant="ghost"
                              onClick={() => {
                                setEditingShow(show);
                                setShowForm(true);
                              }}
                              className="h-8 w-8 p-0 hover:bg-yellow-100"
                            >
                              <Pencil className="w-3.5 h-3.5" />
                            </Button>
                          )}
                          <Link to={createPageUrl('ShowDetail') + '?id=' + show.id}>
                            <Button size="sm" variant="ghost" className="h-8 w-8 p-0 hover:bg-yellow-100">
                              <Eye className="w-4 h-4" />
                            </Button>
                          </Link>
                        </div>
                      </td>
                    </tr>
                  );
                })}
              </tbody>
            </table>
          </div>
          {filteredShows.length === 0 && (
            <div className="text-center py-12 text-gray-500">
              No shows found
            </div>
          )}
        </div>
      ) : (
        /* List View */
        <div className="space-y-8">
          <div>
            <h2 className="text-base font-bold text-black mb-3 flex items-center gap-2">
              <div className="w-1.5 h-5 bg-yellow-400 rounded-full"></div>
              Upcoming Shows
            </h2>
            {upcomingShows.length === 0 ? (
              <div className="text-center py-10 bg-white rounded-2xl border border-gray-100">
                <Calendar className="w-10 h-10 text-gray-200 mx-auto mb-2" />
                <p className="text-sm text-gray-400">No upcoming shows</p>
              </div>
            ) : (
              <div className="space-y-3">{upcomingShows.map((show, idx) => renderShowCard(show, idx))}</div>
            )}
          </div>

          {pastShows.length > 0 && (
            <div>
              <h2 className="text-base font-bold text-gray-400 mb-3 flex items-center gap-2">
                <div className="w-1.5 h-5 bg-gray-300 rounded-full"></div>
                Past Shows
              </h2>
              <div className="space-y-3 opacity-60">{pastShows.slice(0, 10).map((show) => renderShowCard(show, -1))}</div>
            </div>
          )}
        </div>
      )}
    </div>
    </PullToRefresh>
  );
}
