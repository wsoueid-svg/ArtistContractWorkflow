import React from 'react';
import { useQuery } from '@tanstack/react-query';
import { base44 } from '@/api/base44Client';
import { Calendar, MapPin } from 'lucide-react';

const TYPE_META = {
  flight: { emoji: '✈️', label: 'Flight' },
  hotel: { emoji: '🏨', label: 'Hotel' },
  transfer: { emoji: '🚐', label: 'Transfer' },
  ground_transport: { emoji: '🚐', label: 'Ground transport' },
  other: { emoji: '📦', label: 'Other' },
};

const DETAIL_ROWS = [
  ['flight_number', 'Flight'],
  ['booking_reference', 'PNR'],
  ['departure_terminal', 'Departure terminal'],
  ['arrival_terminal', 'Arrival terminal'],
  ['confirmation_number', 'Reference'],
  ['transport_company', 'Company'],
  ['driver_name', 'Driver'],
  ['driver_phone', 'Driver phone'],
];

function moment(value) {
  if (!value) return null;
  const parsed = new Date(value);
  if (Number.isNaN(parsed.getTime())) return String(value);
  return parsed.toLocaleString(undefined, { weekday: 'short', day: 'numeric', month: 'short', hour: '2-digit', minute: '2-digit' });
}

/** Read-only, token-gated itinerary for artists, promoters and drivers. */
export default function PublicItinerary() {
  const token = new URLSearchParams(window.location.search).get('token');
  const { data, isLoading, error } = useQuery({
    queryKey: ['public-itinerary', token],
    queryFn: () => base44.functions.invoke('getPublicItineraryByToken', { token }).then(result => result.data),
    enabled: !!token,
    retry: false,
  });

  if (!token) return <div className="p-8 text-center text-sm text-slate-500">Missing itinerary link.</div>;
  if (isLoading) return <div className="p-8 text-center text-sm text-slate-500">Loading itinerary…</div>;
  if (error || data?.error) {
    return <div className="p-8 text-center text-sm text-slate-500">{data?.error || 'This itinerary link is no longer valid.'}</div>;
  }

  const itinerary = data?.itinerary;
  if (!itinerary) return <div className="p-8 text-center text-sm text-slate-500">Itinerary not found.</div>;
  const location = [itinerary.venue_name, itinerary.city, itinerary.country].filter(Boolean).join(', ');
  const scheduleRows = [
    ['load_in_time', 'Load in'],
    ['sound_check_time', 'Sound check'],
    ['doors_time', 'Doors'],
    ['stage_time', 'Stage'],
  ].filter(([key]) => itinerary.schedule?.[key]);

  return (
    <div className="min-h-screen bg-slate-50">
      <header className="bg-black px-5 py-6">
        <p className="text-[10px] font-bold uppercase tracking-widest text-yellow-400">Itinerary</p>
        <h1 className="mt-1 text-xl font-bold text-white">{itinerary.artist_name || 'Show'}</h1>
        {itinerary.event_name && <p className="mt-0.5 text-sm text-slate-300">{itinerary.event_name}</p>}
      </header>

      <div className="mx-auto max-w-xl space-y-4 p-4">
        <section className="rounded-2xl border border-slate-200 bg-white p-4">
          <p className="flex items-center gap-2 text-[13px] text-slate-700">
            <Calendar className="h-4 w-4 text-slate-400" />
            <span className="tabular-nums">{itinerary.show_date || 'Date TBD'}{itinerary.show_time ? ` · ${itinerary.show_time}` : ''}</span>
          </p>
          <p className="mt-2 flex items-center gap-2 text-[13px] text-slate-700">
            <MapPin className="h-4 w-4 text-slate-400" />{location || 'Venue TBD'}
          </p>
        </section>

        {scheduleRows.length > 0 && (
          <section className="rounded-2xl border border-slate-200 bg-white p-4">
            <h2 className="mb-2 text-sm font-semibold text-slate-900">Show day</h2>
            {scheduleRows.map(([key, label], index) => (
              <div key={key} className={`flex items-center justify-between py-2 text-[13px] ${index % 2 ? 'bg-slate-50' : ''}`}>
                <span className="text-slate-500">{label}</span>
                <span className="font-semibold tabular-nums text-slate-900">{itinerary.schedule[key]}</span>
              </div>
            ))}
            {itinerary.schedule?.ground_transport && (
              <p className="mt-2 border-t border-slate-200 pt-2 text-[13px] text-slate-700">🚐 {itinerary.schedule.ground_transport}</p>
            )}
          </section>
        )}

        {itinerary.events?.length > 0 && (
          <section className="space-y-3">
            <h2 className="text-sm font-semibold text-slate-900">Travel</h2>
            {itinerary.events.map((event, index) => {
              const meta = TYPE_META[event.type] || TYPE_META.other;
              const route = [event.from_location, event.to_location].filter(Boolean).join(' → ');
              return (
                <article key={index} className="rounded-2xl border border-slate-200 bg-white p-4">
                  <p className="text-sm font-semibold text-slate-900">{meta.emoji} {route || meta.label}</p>
                  {event.departure_datetime && <p className="mt-1 text-xs text-slate-500">Dep: {moment(event.departure_datetime)}</p>}
                  {event.arrival_datetime && <p className="text-xs text-slate-500">Arr: {moment(event.arrival_datetime)}</p>}
                  <dl className="mt-2 space-y-1">
                    {DETAIL_ROWS.filter(([key]) => event[key]).map(([key, label]) => (
                      <div key={key} className="flex items-center justify-between text-[13px]">
                        <dt className="text-slate-500">{label}</dt>
                        <dd className="font-semibold tabular-nums text-slate-900">
                          {key === 'driver_phone' ? <a href={`tel:${event[key]}`} className="underline">{event[key]}</a> : event[key]}
                        </dd>
                      </div>
                    ))}
                  </dl>
                </article>
              );
            })}
          </section>
        )}
      </div>
    </div>
  );
}
