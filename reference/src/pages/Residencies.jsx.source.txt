import React, { useState, useMemo } from 'react';
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
import { base44 } from '@/api/base44Client';
import { useOrg } from '../lib/OrgContext';
import { filterShowsForOrg, buildStubArtistIds } from '../lib/orgUtils';
import { Button } from '@/components/ui/button';
import { Input } from '@/components/ui/input';
import { Label } from '@/components/ui/label';
import { Badge } from '@/components/ui/badge';
import { Card, CardContent } from '@/components/ui/card';
import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from '@/components/ui/select';
import { Dialog, DialogContent, DialogHeader, DialogTitle } from '@/components/ui/dialog';
import { Textarea } from '@/components/ui/textarea';
import { Plus, Music2, Calendar, MapPin, X, Save, ChevronRight } from 'lucide-react';
import { toast } from 'sonner';
import { createPageUrl } from '../utils';
import { Link } from 'react-router-dom';
import VenueLink from '../components/shared/VenueLink';

const STATUS_COLORS = {
  active: 'bg-green-100 text-green-700',
  completed: 'bg-gray-100 text-gray-600',
  cancelled: 'bg-red-100 text-red-700',
};

function ResidencyForm({ onClose, editing = null }) {
  const { activeOrg } = useOrg();
  const queryClient = useQueryClient();
  const { data: allArtists = [] } = useQuery({
    queryKey: ['artists'],
    queryFn: () => base44.entities.Artist.list(),
  });
  const artists = useMemo(() => (allArtists || []).filter(a =>
    activeOrg === 'analog' ? a.is_aboss_import : !a.is_aboss_import
  ), [allArtists, activeOrg]);
  const abossStubArtistIds = useMemo(() => buildStubArtistIds(allArtists), [allArtists]);
  const { data: rawShows = [] } = useQuery({
    queryKey: ['all_shows'],
    queryFn: () => base44.entities.Show.list('-date', 200),
  });
  const allShows = useMemo(() => filterShowsForOrg(rawShows || [], activeOrg, abossStubArtistIds), [rawShows, activeOrg, abossStubArtistIds]);

  const [form, setForm] = useState({
    name: editing?.name || '',
    artist_id: editing?.artist_id || '',
    venue: editing?.venue || '',
    city: editing?.city || '',
    country: editing?.country || '',
    status: editing?.status || 'active',
    notes: editing?.notes || '',
    show_ids: editing?.show_ids || [],
  });

  const artistShows = allShows.filter(s => s.artist_id === form.artist_id);

  const mutation = useMutation({
    mutationFn: (data) =>
      editing
        ? base44.entities.Residency.update(editing.id, data)
        : base44.entities.Residency.create(data),
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ['residencies'] });
      toast.success(editing ? 'Residency updated' : 'Residency created');
      onClose();
    },
    onError: (err) => toast.error(err.message),
  });

  const toggleShow = (showId) => {
    setForm(f => ({
      ...f,
      show_ids: f.show_ids.includes(showId)
        ? f.show_ids.filter(id => id !== showId)
        : [...f.show_ids, showId],
    }));
  };

  const handleSubmit = (e) => {
    e.preventDefault();
    if (!form.name || !form.artist_id) {
      toast.error('Name and artist are required');
      return;
    }
    mutation.mutate(form);
  };

  return (
    <form onSubmit={handleSubmit} className="space-y-4">
      <div>
        <Label>Residency Name</Label>
        <Input
          value={form.name}
          onChange={e => setForm({ ...form, name: e.target.value })}
          placeholder="e.g. Hi-Ibiza Summer 2026"
          className="mt-1"
        />
      </div>

      <div className="grid grid-cols-2 gap-4">
        <div>
          <Label>Artist</Label>
          <Select value={form.artist_id} onValueChange={v => setForm({ ...form, artist_id: v })}>
            <SelectTrigger className="mt-1">
              <SelectValue placeholder="Select artist" />
            </SelectTrigger>
            <SelectContent>
              {artists.map(a => (
                <SelectItem key={a.id} value={a.id}>{a.name}</SelectItem>
              ))}
            </SelectContent>
          </Select>
        </div>
        <div>
          <Label>Status</Label>
          <Select value={form.status} onValueChange={v => setForm({ ...form, status: v })}>
            <SelectTrigger className="mt-1">
              <SelectValue />
            </SelectTrigger>
            <SelectContent>
              <SelectItem value="active">Active</SelectItem>
              <SelectItem value="completed">Completed</SelectItem>
              <SelectItem value="cancelled">Cancelled</SelectItem>
            </SelectContent>
          </Select>
        </div>
      </div>

      <div className="grid grid-cols-3 gap-3">
        <div>
          <Label>Venue</Label>
          <Input value={form.venue} onChange={e => setForm({ ...form, venue: e.target.value })} placeholder="Venue name" className="mt-1" />
        </div>
        <div>
          <Label>City</Label>
          <Input value={form.city} onChange={e => setForm({ ...form, city: e.target.value })} placeholder="City" className="mt-1" />
        </div>
        <div>
          <Label>Country</Label>
          <Input value={form.country} onChange={e => setForm({ ...form, country: e.target.value })} placeholder="Country" className="mt-1" />
        </div>
      </div>

      {form.artist_id && (
        <div>
          <Label>Link Show Dates ({form.show_ids.length} selected)</Label>
          <div className="mt-1 max-h-48 overflow-y-auto space-y-1 border rounded-lg p-2 bg-gray-50">
            {artistShows.length === 0 && (
              <p className="text-xs text-gray-400 italic p-1">No shows found for this artist.</p>
            )}
            {artistShows
              .sort((a, b) => (a.date || '').localeCompare(b.date || ''))
              .map(s => (
                <label key={s.id} className={`flex items-center gap-2 p-2 rounded-lg cursor-pointer hover:bg-white transition-colors ${
                  form.show_ids.includes(s.id) ? 'bg-yellow-50 border border-yellow-300' : ''
                }`}>
                  <input
                    type="checkbox"
                    checked={form.show_ids.includes(s.id)}
                    onChange={() => toggleShow(s.id)}
                    className="w-4 h-4 accent-yellow-400"
                  />
                  <span className="text-sm flex-1">
                    <span className="font-medium">{s.event_name}</span>
                    <span className="text-gray-400 ml-2 text-xs">{s.date} · {s.city}</span>
                  </span>
                  <Badge className={`text-[10px] ${s.status === 'confirmed' ? 'bg-green-100 text-green-700' : 'bg-gray-100 text-gray-500'}`}>
                    {s.status}
                  </Badge>
                </label>
              ))}
          </div>
        </div>
      )}

      <div>
        <Label>Notes (optional)</Label>
        <Textarea value={form.notes} onChange={e => setForm({ ...form, notes: e.target.value })} placeholder="Any notes about this residency..." className="mt-1 h-20" />
      </div>

      <div className="flex justify-end gap-2 pt-2">
        <Button type="button" variant="ghost" onClick={onClose}><X className="w-4 h-4 mr-1" /> Cancel</Button>
        <Button type="submit" className="bg-yellow-400 text-black hover:bg-yellow-500" disabled={mutation.isPending}>
          <Save className="w-4 h-4 mr-1" />{mutation.isPending ? 'Saving...' : editing ? 'Update' : 'Create Residency'}
        </Button>
      </div>
    </form>
  );
}

function ResidencyCard({ residency, shows, onClick }) {
  const linkedShows = shows.filter(s => residency.show_ids?.includes(s.id));

  return (
    <Card className="cursor-pointer hover:shadow-md transition-shadow" onClick={onClick}>
      <CardContent className="p-5">
        <div className="flex items-start justify-between mb-3">
          <div>
            <div className="flex items-center gap-2 mb-1">
              <span className="text-lg">🎪</span>
              <h3 className="font-bold text-gray-900">{residency.name}</h3>
            </div>
            <VenueLink venue={residency.venue} city={residency.city} country={residency.country} />
          </div>
          <Badge className={STATUS_COLORS[residency.status] || STATUS_COLORS.active}>
            {residency.status}
          </Badge>
        </div>

        <div className="flex items-center gap-2 mb-3">
          <Calendar className="w-3.5 h-3.5 text-gray-400" />
          <span className="text-xs text-gray-500">{linkedShows.length} show{linkedShows.length !== 1 ? 's' : ''}</span>
        </div>

        {linkedShows.length > 0 && (
          <div className="space-y-1">
            {linkedShows
              .sort((a, b) => (a.date || '').localeCompare(b.date || ''))
              .slice(0, 4)
              .map(s => (
                <div key={s.id} className="flex items-center justify-between text-xs py-1 border-b border-gray-50 last:border-0">
                  <span className="text-gray-700 font-medium">{s.date}</span>
                  <Badge className={`text-[10px] ${s.status === 'confirmed' ? 'bg-green-100 text-green-700' : 'bg-gray-100 text-gray-500'}`}>
                    {s.status}
                  </Badge>
                </div>
              ))}
            {linkedShows.length > 4 && (
              <p className="text-xs text-gray-400">+{linkedShows.length - 4} more</p>
            )}
          </div>
        )}

        <div className="flex justify-end mt-3">
          <ChevronRight className="w-4 h-4 text-gray-300" />
        </div>
      </CardContent>
    </Card>
  );
}

export default function Residencies() {
  const { activeOrg } = useOrg();
  const [formOpen, setFormOpen] = useState(false);
  const [selectedResidency, setSelectedResidency] = useState(null);

  const { data: allArtists = [] } = useQuery({
    queryKey: ['artists'],
    queryFn: () => base44.entities.Artist.list(),
  });
  const abossArtistIds = useMemo(() => new Set((allArtists || []).filter(a => a.is_aboss_import).map(a => a.id)), [allArtists]);

  const { data: rawResidencies = [], isLoading } = useQuery({
    queryKey: ['residencies'],
    queryFn: () => base44.entities.Residency.list('-created_date'),
  });
  const residencies = useMemo(() => (rawResidencies || []).filter(r =>
    activeOrg === 'analog' ? abossArtistIds.has(r.artist_id) : !abossArtistIds.has(r.artist_id)
  ), [rawResidencies, activeOrg, abossArtistIds]);

  const { data: rawShows = [] } = useQuery({
    queryKey: ['all_shows'],
    queryFn: () => base44.entities.Show.list('-date', 200),
  });
  const allShows = useMemo(() => filterShowsForOrg(rawShows || [], activeOrg, abossArtistIds), [rawShows, activeOrg, abossArtistIds]);

  if (isLoading) return <div className="text-center py-12 text-gray-400">Loading residencies...</div>;

  return (
    <div className="space-y-6">
      <div className="flex items-center justify-between">
        <div>
          <h1 className="text-2xl font-bold text-gray-900">Residencies</h1>
          <p className="text-sm text-gray-500 mt-1">Multi-date artist residencies grouped under one umbrella</p>
        </div>
        <Button onClick={() => { setSelectedResidency(null); setFormOpen(true); }} className="bg-yellow-400 text-black hover:bg-yellow-500 gap-2">
          <Plus className="w-4 h-4" /> New Residency
        </Button>
      </div>

      {residencies.length === 0 ? (
        <div className="text-center py-20 text-gray-400">
          <span className="text-5xl block mb-3">🎪</span>
          <p className="font-medium">No residencies yet</p>
          <p className="text-sm mt-1">Create one to group multi-date shows at the same venue</p>
        </div>
      ) : (
        <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
          {residencies.map(r => (
            <ResidencyCard
              key={r.id}
              residency={r}
              shows={allShows}
              onClick={() => { setSelectedResidency(r); setFormOpen(true); }}
            />
          ))}
        </div>
      )}

      <Dialog open={formOpen} onOpenChange={() => setFormOpen(false)}>
        <DialogContent className="max-w-xl max-h-[90vh] overflow-y-auto">
          <DialogHeader>
            <DialogTitle>{selectedResidency ? 'Edit Residency' : 'New Residency'}</DialogTitle>
          </DialogHeader>
          <ResidencyForm
            editing={selectedResidency}
            onClose={() => setFormOpen(false)}
          />
        </DialogContent>
      </Dialog>
    </div>
  );
}
