import React, { useMemo } from 'react';
import { useSearchParams } from 'react-router-dom';
import { CalendarDays, Lock } from 'lucide-react';
import { usePortal } from '@/lib/portal/PortalContext';
import PortalShowCard from '@/components/portal/PortalShowCard';

/** Chronological feed of the artist's upcoming shows. */
export default function PortalShows() {
  const { shows, permissions } = usePortal();
  const [searchParams] = useSearchParams();
  const artist = searchParams.get('artist');
  const search = artist ? `?artist=${artist}` : '';

  const today = new Date().toISOString().slice(0, 10);
  const { upcoming, recent } = useMemo(() => ({
    upcoming: shows.filter((s) => !s.date || s.date >= today),
    recent: shows.filter((s) => s.date && s.date < today),
  }), [shows, today]);

  if (!permissions.can_view_shows) {
    return (
      <div className="rounded-2xl border border-slate-200 bg-white p-6 text-center">
        <Lock className="mx-auto h-5 w-5 text-slate-300" aria-hidden="true" />
        <p className="mt-3 text-sm font-semibold text-slate-900">Shows aren’t shared with you</p>
        <p className="mt-1 text-xs text-slate-500">Ask your agent to enable show access.</p>
      </div>
    );
  }

  if (!shows.length) {
    return (
      <div className="rounded-2xl border border-slate-200 bg-white p-8 text-center">
        <CalendarDays className="mx-auto h-6 w-6 text-slate-300" aria-hidden="true" />
        <p className="mt-3 text-sm font-semibold text-slate-900">No shows scheduled</p>
        <p className="mt-1 text-xs text-slate-500">New confirmed shows appear here automatically.</p>
      </div>
    );
  }

  return (
    <div className="space-y-5">
      {upcoming.length > 0 && (
        <div className="space-y-2">
          <p className="px-1 text-[10px] font-bold uppercase tracking-[0.16em] text-slate-400">
            Upcoming · {upcoming.length}
          </p>
          {upcoming.map((show) => <PortalShowCard key={show.id} show={show} search={search} />)}
        </div>
      )}

      {recent.length > 0 && (
        <div className="space-y-2">
          <p className="px-1 text-[10px] font-bold uppercase tracking-[0.16em] text-slate-400">Just played</p>
          {recent.map((show) => <PortalShowCard key={show.id} show={show} search={search} />)}
        </div>
      )}
    </div>
  );
}
