import React from 'react';
import { Link, useParams, useSearchParams } from 'react-router-dom';
import {
  ArrowLeft, MapPin, Clock, Plane, Users, Wallet, StickyNote,
  Stamp, ListOrdered, FileText, Phone, Mail, BedDouble, Utensils, Megaphone,
} from 'lucide-react';
import { usePortal } from '@/lib/portal/PortalContext';
import PortalSection, { PortalRow } from '@/components/portal/PortalSection';
import PortalTravelCard from '@/components/portal/PortalTravelCard';
import PortalDocumentList from '@/components/portal/PortalDocumentList';
import { formatLocalClock, formatShowDate, resolveShowTimezone, zoneAbbreviation } from '@/lib/portal/timezone';

const TIMELINE_LABELS = {
  load_in: 'Load in',
  sound_check: 'Sound check',
  doors_open: 'Doors open',
  support_act: 'Support act',
  artist_on_stage: 'On stage',
  set_end: 'Set end',
  curfew: 'Curfew',
  travel: 'Travel',
  other: 'Other',
};

const VISA_STYLES = {
  approved: 'bg-green-100 text-green-700',
  pending: 'bg-yellow-100 text-yellow-700',
  denied: 'bg-red-100 text-red-700',
  not_needed: 'bg-slate-100 text-slate-600',
};

const money = (amount, currency) => {
  if (amount === null || amount === undefined) return null;
  try {
    return new Intl.NumberFormat('en-GB', { style: 'currency', currency: currency || 'EUR', maximumFractionDigits: 0 }).format(amount);
  } catch {
    return `${currency || ''} ${amount}`.trim();
  }
};

/** Full read-only show view — one scroll, stacked scannable sections. */
export default function PortalShowDetail() {
  const { showId } = useParams();
  const { shows, permissions } = usePortal();
  const [searchParams] = useSearchParams();
  const artist = searchParams.get('artist');
  const search = artist ? `?artist=${artist}` : '';

  const show = shows.find((s) => s.id === showId);

  if (!show) {
    return (
      <div className="rounded-2xl border border-slate-200 bg-white p-6 text-center">
        <p className="text-sm font-semibold text-slate-900">Show not found</p>
        <Link to={`/portal${search}`} className="mt-3 inline-block text-xs font-semibold text-yellow-600">
          Back to itinerary
        </Link>
      </div>
    );
  }

  const tz = resolveShowTimezone(show);
  const tzLabel = zoneAbbreviation(tz, show.date ? new Date(`${show.date}T12:00:00Z`) : new Date());
  const ops = show.operations || {};
  const timeline = (show.timeline_events || [])
    .slice()
    .sort((a, b) => String(a.time || '').localeCompare(String(b.time || '')));

  return (
    <div className="space-y-4">
      <Link
        to={`/portal${search}`}
        className="inline-flex min-h-[44px] items-center gap-1.5 text-xs font-semibold text-slate-500"
      >
        <ArrowLeft className="h-4 w-4" aria-hidden="true" /> Itinerary
      </Link>

      {/* Header */}
      <div className="rounded-2xl bg-slate-900 p-4">
        <p className="text-[10px] font-bold uppercase tracking-[0.16em] text-yellow-400">
          {formatShowDate(show.date)}
        </p>
        <h1 className="mt-1 text-xl font-bold leading-tight text-white">{show.event_name || 'Untitled show'}</h1>
        <p className="mt-1 flex items-center gap-1.5 text-xs text-slate-400">
          <MapPin className="h-3.5 w-3.5 shrink-0" aria-hidden="true" />
          {[show.venue, show.city, show.country].filter(Boolean).join(', ') || 'Location TBA'}
        </p>
        {tzLabel && (
          <p className="mt-2 inline-block rounded-full bg-white/10 px-2 py-0.5 text-[10px] font-semibold text-slate-300">
            All times {tzLabel} · venue local
          </p>
        )}
      </div>

      {show.artist_facing_notes && (
        <div className="rounded-2xl border-2 border-yellow-400 bg-yellow-50 p-4">
          <p className="flex items-center gap-1.5 text-[10px] font-bold uppercase tracking-[0.16em] text-yellow-700">
            <Megaphone className="h-3.5 w-3.5" aria-hidden="true" /> Notes From Your Team
          </p>
          <p className="mt-2 whitespace-pre-wrap text-sm font-medium text-slate-900">{show.artist_facing_notes}</p>
        </div>
      )}

      {/* Artist fee — permission gated, hidden entirely when not allowed */}
      {permissions.can_view_finances && show.fee && (
        <PortalSection icon={Wallet} title="Artist Fee" accent>
          <PortalRow label="Guarantee" value={money(show.fee.guarantee_fee, show.fee.currency)} mono />
          <PortalRow label="Due to artist" value={money(show.fee.due_to_artist, show.fee.currency)} mono />
          <PortalRow label="Deal type" value={show.fee.deal_type?.replace(/_/g, ' ')} />
        </PortalSection>
      )}

      {/* Venue & schedule */}
      <PortalSection icon={MapPin} title="Venue & Schedule">
        <PortalRow label="Venue" value={show.venue} />
        <PortalRow label="City" value={[show.city, show.country].filter(Boolean).join(', ')} />
        <PortalRow label="Capacity" value={show.capacity ? show.capacity.toLocaleString() : null} mono />
        <PortalRow label="Show time" value={formatLocalClock(show.time, show)} mono />
        {show.map_url && (
          <a
            href={show.map_url}
            target="_blank"
            rel="noopener noreferrer"
            className="mt-2 flex min-h-[44px] items-center justify-center gap-2 rounded-xl border border-slate-200 text-sm font-semibold text-slate-700 active:bg-slate-50"
          >
            <MapPin className="h-4 w-4" aria-hidden="true" /> Open in Maps
          </a>
        )}
      </PortalSection>

      {/* Operations */}
      {(ops.load_in_time || ops.sound_check_time || ops.doors_time || ops.stage_time || ops.ground_transport || ops.ground_transport_notes) && (
        <PortalSection icon={Clock} title="Operations">
          <PortalRow label="Load in" value={formatLocalClock(ops.load_in_time, show)} mono />
          <PortalRow label="Sound check" value={formatLocalClock(ops.sound_check_time, show)} mono />
          <PortalRow label="Doors" value={formatLocalClock(ops.doors_time, show)} mono />
          <PortalRow label="On stage" value={formatLocalClock(ops.stage_time, show)} mono />
          <PortalRow label="Ground transport" value={ops.ground_transport} />
          {ops.ground_transport_notes && (
            <p className="whitespace-pre-wrap pt-2 text-sm text-slate-700">{ops.ground_transport_notes}</p>
          )}
        </PortalSection>
      )}

      {/* Logistics */}
      {permissions.can_view_logistics && (show.logistics_events?.length > 0 || show.hotel?.hotel_name) && (
        <PortalSection icon={Plane} title="Logistics">
          <div className="space-y-2">
            {show.hotel?.hotel_name && (
              <div className="rounded-xl border border-slate-200 bg-white p-3">
                <div className="flex items-center gap-2">
                  <BedDouble className="h-4 w-4 text-slate-500" aria-hidden="true" />
                  <p className="flex-1 text-xs font-bold uppercase tracking-wider text-slate-600">Hotel</p>
                </div>
                <p className="mt-1.5 text-sm font-semibold text-slate-900">{show.hotel.hotel_name}</p>
                {show.hotel.address && <p className="text-xs text-slate-500">{show.hotel.address}</p>}
                <div className="mt-1 text-xs text-slate-600">
                  {show.hotel.check_in_date && <p>Check in {show.hotel.check_in_date}</p>}
                  {show.hotel.check_out_date && <p>Check out {show.hotel.check_out_date}</p>}
                </div>
              </div>
            )}
            {(show.logistics_events || []).map((event) => (
              <PortalTravelCard key={event.id} event={event} show={show} />
            ))}
          </div>
        </PortalSection>
      )}

      {/* Day-of timeline */}
      {timeline.length > 0 && (
        <PortalSection icon={ListOrdered} title="Day-of Timeline">
          <ol className="space-y-0">
            {timeline.map((event, idx) => (
              <li key={event.id || idx} className="flex gap-3 border-b border-slate-100 py-2 last:border-0">
                <span className="w-[86px] shrink-0 text-sm font-bold tabular-nums text-slate-900">
                  {formatLocalClock(event.time, show) || 'TBA'}
                </span>
                <span className="min-w-0 flex-1">
                  <span className="block text-sm font-medium text-slate-800">
                    {TIMELINE_LABELS[event.type] || 'Other'}
                  </span>
                  {event.notes && <span className="block text-xs text-slate-500">{event.notes}</span>}
                </span>
              </li>
            ))}
          </ol>
        </PortalSection>
      )}

      {/* Points of contact */}
      {show.contacts?.length > 0 && (
        <PortalSection icon={Users} title="Points of Contact">
          <div className="space-y-2">
            {show.contacts.map((c, idx) => (
              <div key={idx} className="rounded-xl border border-slate-200 bg-white p-3">
                <p className="text-[10px] font-bold uppercase tracking-wider text-slate-400">{c.label}</p>
                {c.name && <p className="text-sm font-semibold text-slate-900">{c.name}</p>}
                <div className="mt-1 flex flex-wrap gap-3">
                  {c.phone && (
                    <a href={`tel:${c.phone}`} className="inline-flex min-h-[44px] items-center gap-1.5 text-sm font-semibold text-yellow-600">
                      <Phone className="h-4 w-4" aria-hidden="true" /> {c.phone}
                    </a>
                  )}
                  {c.email && (
                    <a href={`mailto:${c.email}`} className="inline-flex min-h-[44px] items-center gap-1.5 text-sm font-semibold text-slate-600">
                      <Mail className="h-4 w-4" aria-hidden="true" /> {c.email}
                    </a>
                  )}
                </div>
              </div>
            ))}
          </div>
        </PortalSection>
      )}

      {/* Visa */}
      {(show.visa_required || show.visa_status !== 'not_needed') && (
        <PortalSection icon={Stamp} title="Visa Status">
          <div className="flex items-center justify-between">
            <span className="text-sm text-slate-600">
              {show.visa_required ? 'Visa required for this show' : 'No visa required'}
            </span>
            <span className={`rounded-full px-2.5 py-1 text-[10px] font-bold uppercase ${VISA_STYLES[show.visa_status] || 'bg-slate-100 text-slate-600'}`}>
              {String(show.visa_status || 'not_needed').replace(/_/g, ' ')}
            </span>
          </div>
        </PortalSection>
      )}

      {/* Show-specific preferences */}
      {show.preferences && Object.values(show.preferences).some(Boolean) && (
        <PortalSection icon={Utensils} title="Preferences For This Show">
          <PortalRow label="Dietary" value={show.preferences.dietary} />
          <PortalRow label="Hotel" value={show.preferences.hotel_preferences} />
          <PortalRow label="Transport" value={show.preferences.transportation_notes} />
          <PortalRow label="Rider" value={show.preferences.technical_rider} />
        </PortalSection>
      )}

      {/* Documents */}
      {permissions.can_view_documents && show.documents?.length > 0 && (
        <PortalSection icon={FileText} title="Documents">
          <PortalDocumentList documents={show.documents} />
        </PortalSection>
      )}

      {show.notes && (
        <PortalSection icon={StickyNote} title="Notes">
          <p className="whitespace-pre-wrap text-sm text-slate-700">{show.notes}</p>
        </PortalSection>
      )}
    </div>
  );
}
