import React, { useState } from 'react';
import { base44 } from '@/api/base44Client';
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
import { Button } from '@/components/ui/button';
import { Badge } from '@/components/ui/badge';
import {
  AlertDialog, AlertDialogAction, AlertDialogCancel, AlertDialogContent,
  AlertDialogDescription, AlertDialogFooter, AlertDialogHeader, AlertDialogTitle,
} from '@/components/ui/alert-dialog';
import { Trash2, RotateCcw, Clock, AlertTriangle } from 'lucide-react';

const ENTITY_ICONS = {
  Artist: '🎵', Show: '📅', Contract: '📄', Invoice: '🧾',
  FinancialRecord: '💰', Release: '💿', Tour: '🗺️', Task: '✅',
  Contact: '👤', Document: '📁', SocialMediaPost: '📱',
};

export default function Trash() {
  const [purgeTarget, setPurgeTarget] = useState(null);
  const queryClient = useQueryClient();

  const { data: items = [], isLoading } = useQuery({
    queryKey: ['deleted-items'],
    queryFn: () => base44.entities.DeletedItem.list('-created_date'),
  });

  const restoreMutation = useMutation({
    mutationFn: (deletedItemId) => base44.functions.invoke('restoreDeletedItem', { deleted_item_id: deletedItemId }),
    onSuccess: () => queryClient.invalidateQueries(),
  });

  const permanentDeleteMutation = useMutation({
    mutationFn: (id) => base44.entities.DeletedItem.delete(id),
    onSuccess: () => queryClient.invalidateQueries({ queryKey: ['deleted-items'] }),
  });

  const daysLeft = (expiresAt) => {
    if (!expiresAt) return 0;
    return Math.max(0, Math.ceil((new Date(expiresAt) - new Date()) / (1000 * 60 * 60 * 24)));
  };

  return (
    <div className="space-y-6">
      <div className="flex flex-col sm:flex-row justify-between items-start sm:items-center gap-4 border-b-4 border-yellow-400 pb-4">
        <div>
          <h1 className="text-4xl font-bold text-black">Trash</h1>
          <p className="text-gray-500 mt-1">{items.length} deleted items · Auto-purged after 30 days</p>
        </div>
      </div>

      {isLoading ? (
        <div className="text-center py-12 text-gray-400">Loading...</div>
      ) : items.length === 0 ? (
        <div className="text-center py-16">
          <Trash2 className="w-16 h-16 text-gray-200 mx-auto mb-4" />
          <p className="text-gray-400 text-lg">Trash is empty</p>
          <p className="text-gray-300 text-sm mt-1">Deleted items will appear here for 30 days</p>
        </div>
      ) : (
        <div className="space-y-2">
          {items.map((item) => {
            const days = daysLeft(item.expires_at);
            const isRestoring = restoreMutation.isPending && restoreMutation.variables === item.id;
            return (
              <div key={item.id} className="flex items-center gap-4 bg-white border border-gray-200 rounded-xl px-4 py-3 hover:border-gray-300 transition-colors">
                <span className="text-2xl flex-shrink-0">{ENTITY_ICONS[item.entity_type] || '📦'}</span>
                <div className="flex-1 min-w-0">
                  <div className="flex items-center gap-2">
                    <p className="text-sm font-semibold text-gray-900 truncate">{item.display_name || 'Unnamed'}</p>
                    <Badge variant="outline" className="text-[10px] flex-shrink-0">{item.entity_type}</Badge>
                  </div>
                  <div className="flex items-center gap-3 mt-0.5 text-[11px] text-gray-400">
                    <span>Deleted by {item.deleted_by || 'unknown'}</span>
                    <span>·</span>
                    <span>{new Date(item.created_date).toLocaleDateString('en-US', { month: 'short', day: 'numeric', year: 'numeric' })}</span>
                  </div>
                </div>
                <div className="flex items-center gap-2 flex-shrink-0">
                  <div className={`flex items-center gap-1 text-xs px-2 py-1 rounded-full ${days <= 5 ? 'bg-red-50 text-red-600' : 'bg-gray-100 text-gray-500'}`}>
                    <Clock className="w-3 h-3" />
                    {days}d left
                  </div>
                  <Button
                    size="sm"
                    variant="outline"
                    className="border-green-200 text-green-700 hover:bg-green-50 hover:border-green-300"
                    onClick={() => restoreMutation.mutate(item.id)}
                    disabled={isRestoring}
                  >
                    <RotateCcw className="w-3.5 h-3.5 mr-1" />
                    {isRestoring ? 'Restoring…' : 'Restore'}
                  </Button>
                  <Button
                    size="sm"
                    variant="outline"
                    className="border-red-200 text-red-600 hover:bg-red-50 hover:border-red-300"
                    onClick={() => setPurgeTarget(item)}
                  >
                    <Trash2 className="w-3.5 h-3.5" />
                  </Button>
                </div>
              </div>
            );
          })}
        </div>
      )}

      {/* Permanent Delete Confirmation */}
      <AlertDialog open={!!purgeTarget} onOpenChange={() => setPurgeTarget(null)}>
        <AlertDialogContent>
          <AlertDialogHeader>
            <div className="flex items-center gap-3 mb-1">
              <div className="w-10 h-10 rounded-full bg-red-100 flex items-center justify-center">
                <AlertTriangle className="w-5 h-5 text-red-600" />
              </div>
              <AlertDialogTitle>Permanently Delete</AlertDialogTitle>
            </div>
            <AlertDialogDescription>
              This will permanently delete <strong className="text-gray-900">{purgeTarget?.display_name}</strong>. This cannot be undone.
            </AlertDialogDescription>
          </AlertDialogHeader>
          <AlertDialogFooter>
            <AlertDialogCancel>Cancel</AlertDialogCancel>
            <AlertDialogAction
              className="bg-red-600 hover:bg-red-700 text-white"
              onClick={() => { permanentDeleteMutation.mutate(purgeTarget.id); setPurgeTarget(null); }}
            >
              Delete Forever
            </AlertDialogAction>
          </AlertDialogFooter>
        </AlertDialogContent>
      </AlertDialog>
    </div>
  );
}
