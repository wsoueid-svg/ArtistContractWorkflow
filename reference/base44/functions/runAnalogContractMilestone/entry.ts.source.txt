import { createClientFromRequest } from 'npm:@base44/sdk@0.8.44';
import { secrets } from 'base44:runtime';

export default async function(req) {
  try {
    const base44 = createClientFromRequest(req);
    const authenticated = await base44.auth.isAuthenticated();
    if (!authenticated) return Response.json({ error: 'Unauthorized' }, { status: 401 });

    let caller = null;
    try { caller = await base44.auth.me(); } catch { caller = null; }
    if (caller) return Response.json({ error: 'Workflow-only function' }, { status: 403 });

    const { show_id } = await req.json();
    if (!show_id) return Response.json({ error: 'show_id is required' }, { status: 400 });
    const response = await base44.asServiceRole.functions.invoke('generateAnalogBookingFeeDraft', {
      show_id,
      system_token: secrets.get('ANALOG_SYSTEM_TOKEN'),
    });
    return Response.json(response.data || response);
  } catch (error) {
    return Response.json({ error: error.message }, { status: 500 });
  }
}
