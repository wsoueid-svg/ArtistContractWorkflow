import { createClientFromRequest } from 'npm:@base44/sdk@0.8.44';
import { header, parseAddresses, getMailboxEmail } from '../../shared/gmailThread.ts';
import { getUserGmailToken } from '../../shared/gmailUser.ts';

/**
 * getGmailThreadMetadata
 *
 * Reads an existing Gmail thread (the deal spine) so the Dispatch SlideOver can
 * pre-fill reply-all: latest To / Cc, the thread Subject, and the most recent
 * Message-ID (required for the In-Reply-To / References headers on the reply).
 *
 * POST { thread_id }
 * -> { connected, subject, to: [], cc: [], from, message_id, references }
 */
export default async function(req) {
  try {
    const base44 = createClientFromRequest(req);
    const user = await base44.auth.me();
    if (!user) return Response.json({ error: 'Unauthorized' }, { status: 401 });

    const { thread_id } = await req.json();
    if (!thread_id) return Response.json({ error: 'thread_id is required' }, { status: 400 });

    const accessToken = await getUserGmailToken(base44);
    if (!accessToken) {
      return Response.json({ connected: false, error: 'Gmail is not connected.' });
    }

    const res = await fetch(
      `https://gmail.googleapis.com/gmail/v1/users/me/threads/${encodeURIComponent(thread_id)}?format=metadata&metadataHeaders=Subject&metadataHeaders=From&metadataHeaders=Reply-To&metadataHeaders=To&metadataHeaders=Cc&metadataHeaders=Message-ID&metadataHeaders=References&metadataHeaders=In-Reply-To`,
      { headers: { Authorization: `Bearer ${accessToken}` } }
    );

    if (!res.ok) {
      const text = await res.text();
      return Response.json(
        { connected: true, error: `Could not read thread: ${text.slice(0, 200)}` },
        { status: res.status === 404 ? 404 : 502 }
      );
    }

    const thread = await res.json();
    const messages = thread.messages || [];
    if (!messages.length) {
      return Response.json({ connected: true, error: 'Thread has no messages' }, { status: 404 });
    }

    const last = messages[messages.length - 1];
    const first = messages[0];

    const mailboxEmail = await getMailboxEmail(accessToken);
    const ownEmails = new Set([String(user.email || '').toLowerCase(), mailboxEmail].filter(Boolean));
    const replyTo = parseAddresses(header(last, 'Reply-To'));
    const from = parseAddresses(header(last, 'From'));
    const latestTo = parseAddresses(header(last, 'To')).filter((email) => !ownEmails.has(email));
    const latestCc = parseAddresses(header(last, 'Cc')).filter((email) => !ownEmails.has(email));
    const primary = [...replyTo, ...from].find((email) => !ownEmails.has(email)) || latestTo[0] || '';
    const cc = Array.from(new Set([...latestTo, ...latestCc].filter((email) => email !== primary)));

    return Response.json({
      connected: true,
      thread_id,
      subject: header(last, 'Subject') || header(first, 'Subject'),
      from: from[0] || '',
      reply_to: replyTo[0] || '',
      to: primary ? [primary] : [],
      cc,
      participants: [primary, ...cc].filter(Boolean),
      message_id: header(last, 'Message-ID'),
      references: [header(last, 'References'), header(last, 'Message-ID')].filter(Boolean).join(' '),
      message_count: messages.length,
    });
  } catch (error) {
    return Response.json({ error: error.message }, { status: 500 });
  }
}
