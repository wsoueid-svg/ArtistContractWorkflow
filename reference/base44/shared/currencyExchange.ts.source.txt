export async function fetchEurPerCurrencyUnit(date, currency) {
  if (!currency || currency === 'EUR') return 1;
  const rateDate = date || new Date().toISOString().slice(0, 10);
  const requestRate = async (path) => {
    const response = await fetch(`https://api.frankfurter.dev/v1/${path}?base=${currency}&symbols=EUR`);
    if (!response.ok) throw new Error(`Frankfurter API error: ${response.status}`);
    const data = await response.json();
    const rate = Number(data.rates?.EUR);
    if (!Number.isFinite(rate) || rate <= 0) throw new Error(`No EUR exchange rate available for ${currency}`);
    return rate;
  };
  try {
    return await requestRate(rateDate);
  } catch {
    return requestRate('latest');
  }
}

export async function convertCentsViaEur({ amountCents, fromCurrency, toCurrency, date, toEurRate }) {
  const amount = Number(amountCents);
  if (!Number.isSafeInteger(amount)) throw new Error('Currency conversion requires an integer-cent amount.');
  if (fromCurrency === toCurrency) return amount;
  const fromRate = await fetchEurPerCurrencyUnit(date, fromCurrency);
  const targetRate = toCurrency === 'EUR' ? 1 : Number(toEurRate);
  if (!Number.isFinite(targetRate) || targetRate <= 0) throw new Error(`Invoice has no valid locked EUR rate for ${toCurrency}.`);
  const eurCents = Math.round(amount * fromRate);
  return Math.round(eurCents / targetRate);
}
