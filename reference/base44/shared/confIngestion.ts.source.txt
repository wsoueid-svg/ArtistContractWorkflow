import sanitizeHtml from 'npm:sanitize-html@2.13.1';
import { header, parseAddresses } from './gmailThread.ts';

const MAX_ATTACHMENT_BYTES = 15 * 1024 * 1024;
const MAX_BODY_CHARS = 250000;
const MIME_BY_EXTENSION = {
  pdf: 'application/pdf',
  doc: 'application/msword',
  docx: 'application/vnd.openxmlformats-officedocument.wordprocessingml.document',
  xls: 'application/vnd.ms-excel',
  xlsx: 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet',
  ppt: 'application/vnd.ms-powerpoint',
  pptx: 'application/vnd.openxmlformats-officedocument.presentationml.presentation',
  txt: 'text/plain',
  csv: 'text/csv',
};

const decodeBytes = (data) => {
  const normalized = String(data || '').replace(/-/g, '+').replace(/_/g, '/');
  const binary = atob(normalized);
  return Uint8Array.from(binary, (char) => char.charCodeAt(0));
};

const decodeText = (data) => new TextDecoder().decode(decodeBytes(data)).slice(0, MAX_BODY_CHARS);
const partHeader = (part, name) => (part?.headers || []).find((item) => String(item.name || '').toLowerCase() === name.toLowerCase())?.value || '';
const extensionOf = (name) => String(name || '').split('.').pop()?.toLowerCase() || '';
const isPdf = (bytes) => bytes.length >= 5 && new TextDecoder().decode(bytes.slice(0, 5)) === '%PDF-';
const isZip = (bytes) => bytes[0] === 0x50 && bytes[1] === 0x4b && [0x03, 0x05, 0x07].includes(bytes[2]);
const isOle = (bytes) => [0xd0, 0xcf, 0x11, 0xe0, 0xa1, 0xb1, 0x1a, 0xe1].every((value, index) => bytes[index] === value);
const isText = (bytes) => !bytes.slice(0, Math.min(bytes.length, 4096)).some((value) => value === 0);

function validatedMime(fileName, declaredMime, bytes) {
  const extension = extensionOf(fileName);
  const expected = MIME_BY_EXTENSION[extension];
  if (!expected || bytes.length === 0 || bytes.length > MAX_ATTACHMENT_BYTES) return null;
  const declared = String(declaredMime || '').toLowerCase();
  if (declared.startsWith('image/')) return null;
  if (extension === 'pdf' && !isPdf(bytes)) return null;
  if (['docx', 'xlsx', 'pptx'].includes(extension) && !isZip(bytes)) return null;
  if (['doc', 'xls', 'ppt'].includes(extension) && !isOle(bytes)) return null;
  if (['txt', 'csv'].includes(extension) && !isText(bytes)) return null;
  return expected;
}

function collectParts(payload) {
  const bodies = { plain: [], html: [] };
  const attachments = [];
  const walk = (part) => {
    if (!part) return;
    const disposition = partHeader(part, 'Content-Disposition').toLowerCase();
    const contentId = partHeader(part, 'Content-ID');
    const fileName = String(part.filename || '').trim();
    if (fileName || part.body?.attachmentId) {
      if (!disposition.includes('inline') && !contentId && !String(part.mimeType || '').startsWith('image/')) attachments.push(part);
      return;
    }
    if (part.body?.data && part.mimeType === 'text/plain') bodies.plain.push(decodeText(part.body.data));
    if (part.body?.data && part.mimeType === 'text/html') bodies.html.push(decodeText(part.body.data));
    (part.parts || []).forEach(walk);
  };
  walk(payload);
  return { bodies, attachments };
}

function sanitizedEmailHtml(value) {
  return sanitizeHtml(String(value || '').slice(0, MAX_BODY_CHARS), {
    allowedTags: ['p', 'br', 'div', 'span', 'strong', 'b', 'em', 'i', 'u', 'ul', 'ol', 'li', 'blockquote', 'table', 'thead', 'tbody', 'tr', 'th', 'td'],
    allowedAttributes: { td: ['colspan', 'rowspan'], th: ['colspan', 'rowspan'] },
    disallowedTagsMode: 'discard',
  });
}

async function uploadPrivate(base44, bytes, name, type) {
  const file = new File([new Blob([bytes], { type })], name, { type });
  const result = await base44.asServiceRole.integrations.Core.UploadPrivateFile({ file });
  return result?.file_uri || null;
}

async function attachmentBytes(accessToken, gmailMessageId, part) {
  if (part.body?.data) return decodeBytes(part.body.data);
  const attachmentId = part.body?.attachmentId;
  if (!attachmentId) return new Uint8Array();
  const response = await fetch(`https://gmail.googleapis.com/gmail/v1/users/me/messages/${encodeURIComponent(gmailMessageId)}/attachments/${encodeURIComponent(attachmentId)}`, { headers: { Authorization: `Bearer ${accessToken}` } });
  if (!response.ok) return new Uint8Array();
  const payload = await response.json();
  return decodeBytes(payload.data);
}

export function isCommunicationFrozen(show, now = new Date()) {
  if (!show) return false;
  if (show.status === 'cancelled') return true;
  if (!show.date) return false;
  const parts = new Intl.DateTimeFormat('en-CA', { timeZone: 'Europe/Madrid', year: 'numeric', month: '2-digit', day: '2-digit' }).formatToParts(now);
  const today = Object.fromEntries(parts.map((part) => [part.type, part.value]));
  const todayKey = `${today.year}-${today.month}-${today.day}`;
  const cutoff = new Date(`${show.date.slice(0, 10)}T00:00:00Z`);
  cutoff.setUTCDate(cutoff.getUTCDate() + 30);
  return todayKey > cutoff.toISOString().slice(0, 10);
}

export async function ingestConfThread({ base44, accessToken, thread, confThread, mailboxOwnerUserId, linkedShow }) {
  if (isCommunicationFrozen(linkedShow)) return { frozen: true, messages_created: 0, attachments_created: 0 };
  const entities = base44.asServiceRole.entities;
  let messagesCreated = 0;
  let attachmentsCreated = 0;
  for (const message of thread.messages || []) {
    const dedupeKey = `${mailboxOwnerUserId}:${message.id}`;
    const existing = await entities.ConfMessage.filter({ dedupe_key: dedupeKey }, '-created_date', 1);
    if (existing[0]) continue;
    const { bodies, attachments } = collectParts(message.payload);
    const plain = bodies.plain.join('\n\n').trim();
    const html = sanitizedEmailHtml(bodies.html.join('\n'));
    const bodyTextUri = plain ? await uploadPrivate(base44, new TextEncoder().encode(plain), `${crypto.randomUUID()}.txt`, 'text/plain') : null;
    const bodyHtmlUri = html ? await uploadPrivate(base44, new TextEncoder().encode(html), `${crypto.randomUUID()}.html`, 'text/html') : null;
    const stored = await entities.ConfMessage.create({
      dedupe_key: dedupeKey,
      conf_thread_id: confThread.id,
      gmail_thread_id: thread.id,
      gmail_message_id: message.id,
      rfc_message_id: header(message, 'Message-ID') || undefined,
      mailbox_owner_user_id: mailboxOwnerUserId,
      from_email: parseAddresses(header(message, 'From'))[0] || '',
      to_emails: parseAddresses(header(message, 'To')),
      cc_emails: parseAddresses(header(message, 'Cc')),
      subject: header(message, 'Subject') || '',
      sent_at: message.internalDate ? new Date(Number(message.internalDate)).toISOString() : new Date().toISOString(),
      snippet: message.snippet || '',
      body_text_uri: bodyTextUri || undefined,
      body_html_uri: bodyHtmlUri || undefined,
      has_html: Boolean(bodyHtmlUri)
    });
    messagesCreated++;
    for (const part of attachments) {
      const displayName = String(part.filename || 'attachment').slice(0, 180);
      const bytes = await attachmentBytes(accessToken, message.id, part);
      const mimeType = validatedMime(displayName, part.mimeType, bytes);
      if (!mimeType) continue;
      const attachmentKey = `${dedupeKey}:${part.body?.attachmentId || displayName}`;
      const prior = await entities.ConfAttachment.filter({ dedupe_key: attachmentKey }, '-created_date', 1);
      if (prior[0]) continue;
      const fileUri = await uploadPrivate(base44, bytes, `${crypto.randomUUID()}.${extensionOf(displayName)}`, mimeType);
      if (!fileUri) continue;
      await entities.ConfAttachment.create({
        dedupe_key: attachmentKey,
        conf_thread_id: confThread.id,
        conf_message_id: stored.id,
        mailbox_owner_user_id: mailboxOwnerUserId,
        display_name: displayName,
        mime_type: mimeType,
        size_bytes: bytes.byteLength,
        file_uri: fileUri
      });
      attachmentsCreated++;
    }
  }
  return { frozen: false, messages_created: messagesCreated, attachments_created: attachmentsCreated };
}
