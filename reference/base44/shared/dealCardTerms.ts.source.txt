/**
 * Shared Deal Card financial math + deal-type vocabulary (Deno / backend copy).
 * Kept in sync with src/lib/dealCardTerms.js — the two runtimes cannot share a module.
 */

export const DEAL_TYPE_LABELS: Record<string, string> = {
  flat_fee: 'Flat Fee',
  guarantee: 'Guarantee',
  guarantee_plus: 'Guarantee Plus',
  guarantee_vs: 'Guarantee vs. Percentage',
  door_deal: 'Door Deal',
  landed: 'Landed Deal',
};

export const DEAL_TYPE_PROVISIONS: Record<string, string> = {
  flat_fee:
    'The Artist fee is a fixed flat amount, payable in full regardless of ticket sales or door revenue.',
  guarantee:
    'The Artist fee is a guaranteed amount, payable in full regardless of ticket sales or door revenue.',
  guarantee_plus:
    'The Artist receives the guaranteed amount plus the agreed additional participation on top of the guarantee.',
  guarantee_vs:
    'The Artist receives the greater of the guaranteed amount or the agreed percentage of net box office receipts.',
  door_deal:
    'The Artist fee is derived from door revenue on the agreed split. Settlement is performed on the night of the event.',
  landed:
    'The fee is landed — net and free of any withholding tax, bank charges, or local deductions, all of which are borne by the Promoter.',
};

export const dealTypeLabel = (t?: string) => (t && DEAL_TYPE_LABELS[t]) || (t ? String(t) : '—');
export const dealTypeProvision = (t?: string) => (t && DEAL_TYPE_PROVISIONS[t]) || '';

export const toCents = (v: unknown) => Math.round(Number(v || 0) * 100);
export const fromCents = (c: unknown) => Number(c || 0) / 100;

export function formatMoney(cents: number, currency = 'EUR') {
  return `${currency} ${fromCents(cents).toLocaleString('en-IE', {
    minimumFractionDigits: 2,
    maximumFractionDigits: 2,
  })}`;
}

export function computeDealTerms(opts: {
  guarantee?: unknown;
  bookingFee?: unknown;
  bookingFeeRate?: unknown;
  bookingFeeBase?: unknown;
  bookingFeeType?: string;
  depositPct?: unknown;
}) {
  const guaranteeCents = toCents(opts.guarantee);
  const fixedBookingFeeCents = toCents(opts.bookingFee);
  const bookingFeeRateBps = Math.max(0, Math.round(Number(opts.bookingFeeRate || 0) * 100));
  const bookingFeeBaseCents = toCents(opts.bookingFeeBase ?? opts.guarantee);
  const bookingFeeCents = fixedBookingFeeCents > 0
    ? fixedBookingFeeCents
    : Math.round(bookingFeeBaseCents * bookingFeeRateBps / 10000);

  const dealValueCents =
    (opts.bookingFeeType || 'exclusive') === 'exclusive'
      ? guaranteeCents + bookingFeeCents
      : guaranteeCents;

  const pct = Math.min(100, Math.max(0, Number(opts.depositPct || 0)));
  const depositCents = Math.round((dealValueCents * pct) / 100);
  const balanceCents = dealValueCents - depositCents;

  return { guaranteeCents, bookingFeeCents, dealValueCents, depositCents, balanceCents, depositPct: pct };
}
