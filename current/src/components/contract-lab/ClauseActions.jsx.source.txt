import { ArrowUp, ArrowDown, Plus, RotateCcw, Trash2, MoreHorizontal } from 'lucide-react';
import { Button } from '@/components/ui/button';
import { DropdownMenu, DropdownMenuTrigger, DropdownMenuContent, DropdownMenuItem } from '@/components/ui/dropdown-menu';
export default function ClauseActions({block,onAction,canUp,canDown,appendix=false}) {
  const actions=[...(!appendix?[['above','Insert above',Plus,false],['below','Insert below',Plus,false]]:[]),['up','Move up',ArrowUp,!canUp],['down','Move down',ArrowDown,!canDown],...(block.master?[['restore','Restore to master',RotateCcw,false]]:[]),['remove','Remove',Trash2,false]];
  return <div className="clause-actions"><DropdownMenu><DropdownMenuTrigger asChild><Button size="icon" variant="ghost" aria-label={`Clause options: ${block.id}`} title="Clause options"><MoreHorizontal className="h-4 w-4"/></Button></DropdownMenuTrigger><DropdownMenuContent align="end">{actions.map(([action,label,Icon,disabled])=><DropdownMenuItem key={action} disabled={disabled} className="min-h-11 gap-2" onSelect={()=>onAction(action,block)}><Icon className="h-4 w-4"/>{label}</DropdownMenuItem>)}</DropdownMenuContent></DropdownMenu></div>;
}
