import { Button } from '@/components/ui/button';
import { textOf, formatTime } from '@/components/contract-lab/documentText';
export default function DraftCompare({comparison,onClose,onLoadServer,onKeepBrowser}) {
  const text=d=>d.blocks.map(b=>`${b.id}\n${textOf(b.rich_text)}`).join('\n\n');
  return <section className="rounded-lg border border-border bg-card p-4" aria-label="Compare browser and server copies"><h3 className="font-semibold">Compare copies • nothing applied automatically</h3><div className="mt-3 grid gap-4 md:grid-cols-2"><div><h4 className="text-sm font-semibold">Browser changes</h4><p className="my-2 text-xs text-muted-foreground">{comparison.browserAt?formatTime(new Date(comparison.browserAt).toISOString()):'Your unsaved wording'}</p><pre className="max-h-64 overflow-y-auto whitespace-pre-wrap break-words text-xs">{text(comparison.browser)}</pre></div><div><h4 className="text-sm font-semibold">Server • save {comparison.server.revision_no}</h4><p className="my-2 text-xs text-muted-foreground">{formatTime(comparison.server.approved_at)}</p><pre className="max-h-64 overflow-y-auto whitespace-pre-wrap break-words text-xs">{text(comparison.server)}</pre></div></div><div className="mt-3 flex flex-wrap gap-2"><Button variant="outline" onClick={onClose}>Close comparison</Button>{onKeepBrowser&&<Button variant="outline" onClick={onKeepBrowser}>Keep browser wording, review current terms</Button>}{onLoadServer&&<Button onClick={onLoadServer}>Use server copy</Button>}</div></section>;
}
