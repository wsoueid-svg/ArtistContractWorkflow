export const recoveryKey = (userId, showId, draftId) => `contract-draft:${userId}:${showId}:${draftId}`;
export function readRecovery(key) {
  try { const text=localStorage.getItem(key); if(!text) return null; const entry=JSON.parse(text); if(!entry || !Array.isArray(entry.document?.blocks) || !Number.isFinite(entry.savedAt) || !entry.document.show_id) return {malformed:true}; return entry; }
  catch { return {malformed:true}; }
}
export function writeRecovery(key,document) { const savedAt=Date.now();localStorage.setItem(key,JSON.stringify({savedAt,document}));return savedAt; }
export function blockContent(blocks) { return JSON.stringify(blocks.map(({id,kind,slot,rich_text})=>({id,kind,slot,rich_text}))); }
export function hasDocumentEdits(document) { return blockContent(document.blocks)!==blockContent(document.master_blocks||document.source_master_blocks||document.blocks); }
