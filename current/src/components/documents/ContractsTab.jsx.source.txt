import React, { useState } from 'react';
import contractPreviewUrl from '@/components/contracts/contractPreviewUrl';
import { base44 } from '@/api/base44Client';
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
import { Button } from '@/components/ui/button';
import { Input } from '@/components/ui/input';
import { Badge } from '@/components/ui/badge';
import { Card, CardContent } from '@/components/ui/card';
import { Plus, Search, FileText, ExternalLink, FolderOpen, Eye, Table, List } from 'lucide-react';
import ContractForm from '../contracts/ContractForm';
import { saveContractRecord, deleteContractRecord } from '@/components/contracts/contractRecordApi';
import ContractAlerts from '../contracts/ContractAlerts';
import StageBadge from '../shared/StageBadge';
import SkeletonBlock from '../shared/SkeletonBlock';

const typeColors = {
  show_agreement: 'bg-purple-100 text-purple-700',
  management: 'bg-blue-100 text-blue-700',
  label: 'bg-green-100 text-green-700',
  sponsorship: 'bg-yellow-100 text-yellow-700',
  merchandise: 'bg-pink-100 text-pink-700',
  other: 'bg-gray-100 text-gray-700',
};

export default function ContractsTab({ artists }) {
  const [searchQuery, setSearchQuery] = useState('');
  const [showForm, setShowForm] = useState(false);
  const [editingContract, setEditingContract] = useState(null);
  const [filterStatus, setFilterStatus] = useState('all');
  const [filterArtist, setFilterArtist] = useState('all');
  const [view, setView] = useState('cards');
  const queryClient = useQueryClient();

  const { data: contracts = [], isLoading } = useQuery({
    queryKey: ['contracts'],
    queryFn: () => base44.entities.Contract.list('-created_date'),
  });

  const { data: contacts = [] } = useQuery({
    queryKey: ['contacts'],
    queryFn: () => base44.entities.Contact.list(),
  });

  const createMutation = useMutation({
    mutationFn: (data) => saveContractRecord(data),
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ['contracts'] });
      setShowForm(false);
      setEditingContract(null);
    },
  });

  const updateMutation = useMutation({
    mutationFn: ({ id, data }) => saveContractRecord(data, id),
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ['contracts'] });
      setShowForm(false);
      setEditingContract(null);
    },
  });

  const deleteMutation = useMutation({
    mutationFn: (id) => deleteContractRecord(id),
    onSuccess: () => queryClient.invalidateQueries({ queryKey: ['contracts'] }),
  });

  const handleSubmit = (data) => {
    if (editingContract) return updateMutation.mutateAsync({ id: editingContract.id, data });
    return createMutation.mutateAsync(data);
  };

  // Include archived artists for lookups (historical data)
  const allArtists = artists;

  const filtered = contracts.filter((c) => {
    const matchesSearch = c.title?.toLowerCase().includes(searchQuery.toLowerCase());
    const matchesStatus = filterStatus === 'all' || c.status === filterStatus;
    const matchesArtist = filterArtist === 'all' || c.artist_id === filterArtist;
    return matchesSearch && matchesStatus && matchesArtist;
  });

  return (
    <div className="space-y-5">
      {/* Toolbar */}
      <div className="flex flex-col sm:flex-row gap-3 items-start sm:items-center justify-between">
        <div className="flex flex-wrap gap-2 items-center">
          <div className="relative">
            <Search className="absolute left-3 top-1/2 -translate-y-1/2 text-gray-400 w-4 h-4" />
            <Input
              placeholder="Search contracts..."
              value={searchQuery}
              onChange={(e) => setSearchQuery(e.target.value)}
              className="pl-9 h-8 text-sm border-2 border-gray-200 focus:border-yellow-400 w-48"
            />
          </div>
          <select
            value={filterArtist}
            onChange={(e) => setFilterArtist(e.target.value)}
            className="px-3 py-1.5 text-sm border-2 border-gray-200 rounded-md focus:border-yellow-400 focus:outline-none h-8"
          >
            <option value="all">All Artists</option>
            {allArtists.map((a) => (
              <option key={a.id} value={a.id}>
                {a.stage_name || a.name}{a.status === 'archived' ? ' (Archived)' : ''}
              </option>
            ))}
          </select>
          {['all', 'draft', 'READY', 'DISPATCHING', 'SENT', 'RETURNED', 'SIGNED'].map((s) => (
            <Button
              key={s}
              variant={filterStatus === s ? 'default' : 'outline'}
              size="sm"
              onClick={() => setFilterStatus(s)}
              className={filterStatus === s ? 'bg-yellow-400 text-black hover:bg-yellow-500 h-8' : 'h-8'}
            >
              {s === 'all' ? 'All' : s}
            </Button>
          ))}
        </div>
        <div className="flex gap-2 flex-shrink-0">
          <Button
            variant={view === 'cards' ? 'default' : 'outline'}
            size="sm"
            onClick={() => setView('cards')}
            className={view === 'cards' ? 'bg-yellow-400 text-black hover:bg-yellow-500 h-8' : 'h-8'}
          >
            <List className="w-4 h-4" />
          </Button>
          <Button
            variant={view === 'table' ? 'default' : 'outline'}
            size="sm"
            onClick={() => setView('table')}
            className={view === 'table' ? 'bg-yellow-400 text-black hover:bg-yellow-500 h-8' : 'h-8'}
          >
            <Table className="w-4 h-4" />
          </Button>
          <Button
            onClick={() => { setEditingContract(null); setShowForm(true); }}
            className="bg-black text-yellow-400 hover:bg-gray-900 font-semibold h-8 text-sm"
          >
            <Plus className="w-4 h-4 mr-1" /> New Contract
          </Button>
        </div>
      </div>

      {/* Alerts */}
      <ContractAlerts contracts={contracts} artists={allArtists} />

      {/* Content */}
      {isLoading ? (
        <div className="space-y-3">{[1, 2, 3].map(i => <SkeletonBlock key={i} type="card" />)}</div>
      ) : filtered.length === 0 ? (
        <div className="text-center py-16">
          <FileText className="w-12 h-12 text-gray-200 mx-auto mb-3" />
          <p className="text-gray-400">No contracts found</p>
        </div>
      ) : view === 'table' ? (
        <div className="border-2 border-gray-200 rounded-xl overflow-hidden">
          <div className="overflow-x-auto">
            <table className="w-full">
              <thead className="bg-black text-yellow-400">
                <tr>
                  <th className="text-left py-3 px-4 font-semibold text-sm">Title</th>
                  <th className="text-left py-3 px-4 font-semibold text-sm">Artist</th>
                  <th className="text-left py-3 px-4 font-semibold text-sm">Type</th>
                  <th className="text-right py-3 px-4 font-semibold text-sm">Value</th>
                  <th className="text-center py-3 px-4 font-semibold text-sm">Status</th>
                  <th className="text-center py-3 px-4 font-semibold text-sm">Actions</th>
                </tr>
              </thead>
              <tbody>
                {filtered.map((contract) => {
                  const artist = allArtists.find(a => a.id === contract.artist_id);
                  return (
                    <tr key={contract.id} className="border-b border-gray-100 hover:bg-yellow-50 transition">
                      <td className="py-3 px-4 text-sm font-medium">{contract.title}</td>
                      <td className={`py-3 px-4 text-sm ${artist?.status === 'archived' ? 'italic text-gray-400' : ''}`}>
                        {artist?.stage_name || artist?.name || '-'}
                        {artist?.status === 'archived' && <span className="ml-1 text-xs text-amber-600">(Archived)</span>}
                      </td>
                      <td className="py-3 px-4 text-sm">
                        <Badge className={`text-xs ${typeColors[contract.type || 'other']}`}>
                          {contract.type?.replace('_', ' ') || 'other'}
                        </Badge>
                      </td>
                      <td className="py-3 px-4 text-sm text-right font-semibold">
                        {contract.value ? `${contract.currency || 'USD'} ${contract.value.toLocaleString()}` : '-'}
                      </td>
                      <td className="py-3 px-4 text-center">
                        <StageBadge stage={contract.status || 'draft'} />
                      </td>
                      <td className="py-3 px-4 text-center">
                        <div className="flex gap-1 justify-center">
                          <Button size="sm" variant="ghost" className="h-7 px-2 hover:bg-yellow-100"
                            onClick={() => { setEditingContract(contract); setShowForm(true); }}>
                            Edit
                          </Button>
                          {contract.contract_file_url && (
                            <a href={contractPreviewUrl(contract.id)} target="_blank" rel="noopener noreferrer">
                              <Button size="sm" variant="ghost" className="h-7 px-2 hover:bg-yellow-100">
                                <Eye className="w-3.5 h-3.5" />
                              </Button>
                            </a>
                          )}
                        </div>
                      </td>
                    </tr>
                  );
                })}
              </tbody>
            </table>
          </div>
        </div>
      ) : (
        <div className="space-y-3">
          {filtered.map((contract) => {
            const artist = allArtists.find(a => a.id === contract.artist_id);
            const contact = contacts.find(c => c.id === contract.contact_id);
            const isArchivedArtist = artist?.status === 'archived';
            return (
              <Card
                key={contract.id}
                className="border-2 border-gray-200 hover:border-yellow-400 transition-all cursor-pointer"
                onClick={() => { setEditingContract(contract); setShowForm(true); }}
              >
                <CardContent className="p-5">
                  <div className="flex items-start gap-4">
                    <div className="p-2.5 bg-black rounded-lg flex-shrink-0">
                      <FileText className="w-5 h-5 text-yellow-400" />
                    </div>
                    <div className="flex-1 min-w-0">
                      <div className="flex items-center gap-2 mb-1.5 flex-wrap">
                        <Badge className={`text-xs ${typeColors[contract.type || 'other']}`}>
                          {contract.type?.replace('_', ' ') || 'other'}
                        </Badge>
                        <StageBadge stage={contract.status || 'draft'} />
                      </div>
                      <h3 className="font-bold text-black text-base mb-0.5">{contract.title}</h3>
                      <p className={`text-sm ${isArchivedArtist ? 'text-gray-400 italic' : 'text-gray-600'}`}>
                        {artist?.stage_name || artist?.name || 'Unknown Artist'}
                        {isArchivedArtist && <span className="ml-1 text-xs text-amber-600">(Archived)</span>}
                        {contact && ` · ${contact.name}`}
                      </p>
                      <div className="flex gap-4 mt-3">
                        {contract.value && (
                          <div>
                            <p className="text-xs text-gray-400">Value</p>
                            <p className="text-sm font-semibold">{contract.currency} {contract.value.toLocaleString()}</p>
                          </div>
                        )}
                        {contract.start_date && (
                          <div>
                            <p className="text-xs text-gray-400">Start</p>
                            <p className="text-sm font-semibold">{new Date(contract.start_date).toLocaleDateString()}</p>
                          </div>
                        )}
                        {contract.end_date && (
                          <div>
                            <p className="text-xs text-gray-400">End</p>
                            <p className="text-sm font-semibold">{new Date(contract.end_date).toLocaleDateString()}</p>
                          </div>
                        )}
                      </div>
                      {(contract.contract_file_url || contract.dropbox_folder_url) && (
                        <div className="flex gap-2 mt-3">
                          {contract.contract_file_url && (
                            <a href={contractPreviewUrl(contract.id)} target="_blank" rel="noopener noreferrer"
                              onClick={e => e.stopPropagation()}
                              className="text-xs px-2.5 py-1 bg-gray-100 rounded flex items-center gap-1 hover:bg-gray-200">
                              Contract <ExternalLink className="w-3 h-3" />
                            </a>
                          )}
                          {contract.signed_contract_url && (
                            <a href={`${contractPreviewUrl(contract.id)}&variant=signed`} target="_blank" rel="noopener noreferrer"
                              onClick={e => e.stopPropagation()}
                              className="text-xs px-2.5 py-1 bg-green-100 rounded flex items-center gap-1 hover:bg-green-200">
                              Signed <ExternalLink className="w-3 h-3" />
                            </a>
                          )}
                          {contract.dropbox_folder_url && (
                            <a href={contract.dropbox_folder_url} target="_blank" rel="noopener noreferrer"
                              onClick={e => e.stopPropagation()}
                              className="text-xs px-2.5 py-1 bg-blue-50 text-blue-700 rounded flex items-center gap-1 hover:bg-blue-100">
                              <FolderOpen className="w-3 h-3" /> Dropbox
                            </a>
                          )}
                        </div>
                      )}
                    </div>
                  </div>
                </CardContent>
              </Card>
            );
          })}
        </div>
      )}

      {/* Contract Form Modal */}
      {showForm && (
        <div className="fixed inset-0 bg-black bg-opacity-50 z-50 flex items-center justify-center p-4">
          <div className="bg-white rounded-lg max-w-4xl w-full max-h-[90vh] overflow-y-auto">
            <ContractForm
              contract={editingContract}
              artists={allArtists}
              contacts={contacts}
              onSubmit={handleSubmit}
              onDelete={async id => { await deleteMutation.mutateAsync(id); setShowForm(false); setEditingContract(null); }}
              onCancel={() => { setShowForm(false); setEditingContract(null); }}
            />
          </div>
        </div>
      )}
    </div>
  );
}
