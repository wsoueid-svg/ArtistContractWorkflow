import DealStatusBadge from '@/components/shared/DealStatusBadge';
import contractPreviewUrl from '@/components/contracts/contractPreviewUrl';
import { Link } from 'react-router-dom';
import { Button } from '@/components/ui/button';
import { formatTime } from '@/components/contract-lab/documentText';
export default function ContractHistory({contracts=[],currentId,onView,hasMore,onLoadMore,loadingMore}) {
  if(!contracts.length)return <p className="text-sm text-muted-foreground">No generated revisions yet. Save and generate the current draft first.</p>;
  return <div><ol className="divide-y divide-border">{contracts.map(contract=><li key={contract.id} data-testid={`contract-history-${contract.id}`} className="flex min-h-20 flex-wrap items-start justify-between gap-3 py-4"><div className="min-w-0"><div className="flex flex-wrap items-center gap-2"><h3 className="text-sm font-semibold">Version {contract.version_number||1}</h3><DealStatusBadge kind="contract" status={contract.status} size="sm"/><span className="text-xs text-muted-foreground">{contract.id===currentId?(['SENT','RETURNED','SIGNED'].includes(contract.status)?'Current issued revision':'Current generated revision'):'Historical • read-only'}</span></div><p className="mt-2 text-xs leading-relaxed text-muted-foreground">Created by {contract.created_by_name} • {formatTime(contract.created_date)}<br/>{contract.approved_at?`Approved by ${contract.approved_by_name} • ${formatTime(contract.approved_at)}`:'Approval not recorded for this uploaded revision'}<br/>{contract.generated_at?`Generated ${formatTime(contract.generated_at)}`:'Generation timestamp not recorded'}</p></div><div className="flex flex-wrap gap-2">{contract.contract_file_url&&<Button variant="outline" asChild><Link to={contractPreviewUrl(contract.id)}>Open PDF</Link></Button>}<Button variant="ghost" onClick={()=>onView(contract)} aria-label={`View document version ${contract.version_number||1}`}>View document</Button></div></li>)}</ol>{hasMore&&<Button variant="outline" disabled={loadingMore} onClick={onLoadMore}>{loadingMore?'Loading…':'Load older revisions'}</Button>}</div>;
}
