import { useState } from 'react';
import { useNavigate } from 'react-router-dom';
import contractPreviewUrl from '@/components/contracts/contractPreviewUrl';
import { useInfiniteQuery, useQueryClient } from '@tanstack/react-query';
import { Upload } from 'lucide-react';
import { base44 } from '@/api/base44Client';
import { Button } from '@/components/ui/button';
import UploadContractModal from '@/components/shows/UploadContractModal';
import GenerateDealCardButton from '@/components/shows/GenerateDealCardButton';
import ContractSignatureControl from '@/components/contracts/ContractSignatureControl';
import ContractHistory from '@/components/analog/show-detail/ContractHistory';
import DetailPanel from '@/components/analog/show-detail/DetailPanel';
import { getContractualSetWindow } from '@/lib/showTiming';
import CurrentRevisionCard from '@/components/contract-lab/CurrentRevisionCard';
import ContractEditorModal from '@/components/contract-lab/ContractEditorModal';
import RevisionDocumentView from '@/components/contract-lab/RevisionDocumentView';
import documentValidation from '@/components/contract-lab/documentValidation';
import contractError from '@/components/contract-lab/contractErrors';
import ContractRedlines from '@/components/contracts/redlines/ContractRedlines';
import refreshContractWorkspace from '@/components/contracts/refreshContractWorkspace';
export default function AnalogContractTab({show,artist}) {
  const [uploadOpen,setUploadOpen]=useState(false),[modal,setModal]=useState(null),[view,setView]=useState(null),[busy,setBusy]=useState(false),[actionError,setActionError]=useState('');
  const queryClient=useQueryClient(),navigate=useNavigate();
  const revisions=useInfiniteQuery({queryKey:['contract-revisions',show.id],initialPageParam:null,queryFn:async({pageParam})=>(await base44.functions.invoke('getContractRevisionView',{show_id:show.id,cursor:pageParam})).data,getNextPageParam:last=>last.has_more?last.next_cursor:undefined,enabled:!!show.id,retry:false});
  const first=revisions.data?.pages[0],current=first?.current,draft=first?.draft,contracts=revisions.data?.pages.flatMap(p=>p.items)||[];
  const refresh=()=>refreshContractWorkspace(queryClient,show.id);
  const enriched={...show,artist_name:artist?.stage_name||artist?.name};
  const edit=d=>{setActionError('');setModal({draftId:d?.id||null,forceAmendment:!!d?.parent_draft_id});};
  const generate=async d=>{
    setBusy(true);setActionError('');
    try{const {data}=await base44.functions.invoke('getContractDraft',{show_id:show.id,draft_id:d.id,force_amendment:!!d.parent_draft_id});if(data.draft.status!=='approved'||data.draft.browser_dirty||documentValidation(data.draft).length){edit(d);return;}const result=await base44.functions.invoke('generateSmartContract',{show_id:show.id,draft_id:d.id,contract_template_id:data.draft.template_id,expected_content_hash:data.draft.content_hash,force_amendment:!!data.draft.parent_draft_id});refresh();navigate(contractPreviewUrl(result.data.contract_id,show.id));}
    catch(e){setActionError(contractError(e));}finally{setBusy(false);}
  };
  return <div className="space-y-4">{revisions.isLoading?<p role="status" className="p-5 text-sm text-muted-foreground">Loading revisions…</p>:revisions.error?<p role="alert" className="p-5 text-sm text-destructive">{contractError(revisions.error)}</p>:<CurrentRevisionCard show={enriched} current={current} draft={draft} onEdit={edit} onGenerate={generate} onAmend={c=>setModal({cloneId:c.id})} onChanged={refresh} busy={busy} error={actionError}/>}
    <DetailPanel><div className="flex flex-wrap items-center justify-between gap-3"><div><h2 className="text-sm font-semibold">Contractual set window</h2><p className="mt-1 text-sm tabular-nums text-muted-foreground">{getContractualSetWindow(show)}</p></div><div className="flex flex-wrap gap-2">{!current&&<Button variant="outline" onClick={()=>setUploadOpen(true)}><Upload className="h-4 w-4"/>Upload existing PDF</Button>}<GenerateDealCardButton show={enriched}/></div></div>{current&&<details className="mt-4 border-t border-border"><summary className="flex min-h-11 cursor-pointer items-center text-sm text-muted-foreground">Record a received signature</summary><ContractSignatureControl contract={current} onChange={refresh}/></details>}</DetailPanel>
    {current&&<ContractRedlines showId={show.id} contractId={current.id} readOnly={current.status==='SIGNED'}/>}
    <DetailPanel><h2 className="mb-2 text-base font-semibold">Revision history</h2><ContractHistory contracts={contracts} currentId={current?.id} onView={setView} hasMore={revisions.hasNextPage} onLoadMore={()=>revisions.fetchNextPage()} loadingMore={revisions.isFetchingNextPage}/></DetailPanel>
    {modal&&<ContractEditorModal key={`${modal.draftId||'new'}-${modal.cloneId||''}`} show={enriched} open onClose={()=>setModal(null)} draftId={modal.draftId} cloneId={modal.cloneId} forceAmendment={modal.forceAmendment} onGenerated={refresh}/>}
    {view&&<RevisionDocumentView showId={show.id} contract={view} onClose={()=>setView(null)}/>}
    <UploadContractModal show={show} open={uploadOpen} onClose={()=>setUploadOpen(false)} onSaved={refresh}/>
  </div>;
}
