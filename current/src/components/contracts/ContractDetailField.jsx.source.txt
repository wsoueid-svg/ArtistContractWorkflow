import { Input } from '@/components/ui/input';
import { Label } from '@/components/ui/label';

export default function ContractDetailField({ label, name, value, onChange, type = 'text', placeholder, required = false }) {
  return <div className="space-y-2" data-contract-field={name}>
    <Label htmlFor={`contract-field-${name}`} className="text-sm">{label}{required && ' *'}</Label>
    <Input id={`contract-field-${name}`} name={name} type={type} value={value ?? ''} required={required} placeholder={placeholder} onChange={event => onChange(name, event.target.value)} className="min-h-11" />
  </div>;
}
