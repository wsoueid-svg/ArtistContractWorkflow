import { Button } from '@/components/ui/button';
import ContractDetailField from '@/components/contracts/ContractDetailField';
import ContractSourceSummary from '@/components/contracts/ContractSourceSummary';
import ContractPromoterField from '@/components/contracts/ContractPromoterField';
import ContractTemplateSelect from '@/components/contracts/ContractTemplateSelect';
import MonthYearPicker from '@/components/contracts/MonthYearPicker';
import ContractNotesField from '@/components/contracts/ContractNotesField';

export default function ContractSetupForm({ show, fields, onFieldChange, promoter, templateId, onTemplateSelect, entity, entityLocked, bookingFee, onReview, onClose, busy, error }) {
  const field = (label, name, extra = {}) => <ContractDetailField label={label} name={name} value={fields[name]} onChange={onFieldChange} {...extra} />;
  return <form className="flex min-h-0 flex-1 flex-col bg-card text-card-foreground" onSubmit={event => { if (event.target !== event.currentTarget) return; event.preventDefault(); onReview(); }}>
    <header className="border-b border-border p-4 pr-16 sm:p-6 sm:pr-16">
      <h2 className="text-lg font-semibold">Contract details</h2>
      <p className="mt-2 text-sm text-muted-foreground">Confirm the details, then review, save and generate your document.</p>
    </header>
    <div className="min-h-0 flex-1 space-y-6 overflow-y-auto p-4 sm:p-6">
      <ContractSourceSummary show={show} fields={fields} bookingFee={bookingFee} />
      <section className="space-y-4" aria-label="Promoter and company">
        <h3 className="text-base font-semibold">Promoter and company</h3>
        <ContractPromoterField fields={fields} onTextChange={value => onFieldChange('company_name', value)} promoter={promoter} />
        {field('Company address', 'company_address', { required: true, placeholder: 'Full billing address' })}
        {field('VAT / Tax ID', 'company_vat', { required: true })}
        {field('Promoter signatory name', 'company_signatory', { required: true, placeholder: 'Person signing for the promoter' })}
      </section>
      <section className="space-y-4 border-t border-border pt-6" aria-label="Contract template">
        <h3 className="text-base font-semibold">Template and agreement</h3>
        <ContractTemplateSelect value={templateId} onSelect={onTemplateSelect} disabled={entityLocked} />
        {entity && <details className="text-sm"><summary className="cursor-pointer font-medium">Contracting entity: {entity.short_name || entity.name}</summary><dl className="mt-3 space-y-3 text-muted-foreground"><div><dt>Legal name</dt><dd className="text-foreground">{fields.agency_name}</dd></div><div><dt>Address</dt><dd className="break-words text-foreground">{fields.agency_address}</dd></div><div><dt>Tax ID</dt><dd className="text-foreground">{fields.agency_rut}</dd></div><div><dt>Agency signatory</dt><dd className="text-foreground">{fields.agency_signatory}</dd></div></dl></details>}
        {field('Contract date', 'contract_date', { type: 'date', required: true })}
      </section>
      <details className="border-t border-border pt-4"><summary className="cursor-pointer text-sm font-medium">Additional contract terms</summary><div className="mt-4 space-y-4">
        {field('Billing requirements', 'billing_requirements')}
        {field('Billing position on artwork', 'billing_position')}
        <MonthYearPicker label="Deposit month (for contract text)" value={fields.deposit_month} onChange={value => onFieldChange('deposit_month', value)} hint="Use when required by your payment clause." />
        <ContractNotesField value={fields.contract_notes} onChange={value => onFieldChange('contract_notes', value)} />
      </div></details>
      {error && <p role="alert" className="text-sm text-destructive">{error}</p>}
      {!templateId && <p className="text-sm text-muted-foreground">Select a template to continue to document review.</p>}
    </div>
    <footer className="flex shrink-0 items-center justify-between gap-3 border-t border-border bg-card p-4 sm:px-6">
      <Button type="button" variant="outline" className="min-h-11" onClick={onClose} disabled={busy}>Cancel</Button>
      <Button type="submit" className="min-h-11" disabled={busy || promoter.saving || !promoter.ready || !templateId}>{busy ? 'Building document…' : 'Review document'}</Button>
    </footer>
  </form>;
}
