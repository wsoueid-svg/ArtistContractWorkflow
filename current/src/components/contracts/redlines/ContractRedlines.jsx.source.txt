import { useState } from 'react';
import { useInfiniteQuery, useQuery, useQueryClient } from '@tanstack/react-query';
import { base44 } from '@/api/base44Client';
import { Button } from '@/components/ui/button';
import { Input } from '@/components/ui/input';
import RedlineReviewRow from '@/components/contracts/redlines/RedlineReviewRow';
import contractError from '@/components/contract-lab/contractErrors';
export default function ContractRedlines({showId,contractId,readOnly=false}) {
  const [file,setFile]=useState(null),[busy,setBusy]=useState(false),[error,setError]=useState(''),[inputKey,setInputKey]=useState(0),client=useQueryClient();
  const query=useInfiniteQuery({queryKey:['contract-redlines',showId],initialPageParam:null,queryFn:async({pageParam})=>(await base44.functions.invoke('contractRedlines',{action:'list',show_id:showId,cursor:pageParam})).data,getNextPageParam:p=>p.has_more?p.next_cursor:undefined,retry:false});
  const context=useQuery({queryKey:['contract-redline-context',showId],queryFn:async()=>(await base44.functions.invoke('getContractRevisionView',{show_id:showId})).data,retry:false});
  const current=context.data?.current,locked=readOnly||!current||['SIGNED','DISPATCHING'].includes(current.status)||!!current.generation_lock||!!current.dispatch_lock,canUpload=!locked&&current.id===contractId;
  const refresh=()=>client.invalidateQueries({queryKey:['contract-redlines',showId]});
  const upload=async e=>{e.preventDefault();if(!file)return;setBusy(true);setError('');try{if(file.size>10*1024*1024||!file.size||!/^.+\.(pdf|docx)$/i.test(file.name))throw new Error('Choose a PDF or DOCX file up to 10 MB.');await base44.functions.invoke('contractRedlines',{action:'upload',show_id:showId,contract_id:contractId,file});setFile(null);setInputKey(k=>k+1);await refresh();}catch(e){setError(contractError(e));}finally{setBusy(false);}};
  return <section className="space-y-4 rounded-lg border border-border bg-card p-4 sm:p-5" data-testid="contract-redlines"><h2 className="text-base font-semibold">Promoter redlines</h2><p className="text-sm text-muted-foreground">Returned files are stored privately against their source version. Review them here and incorporate agreed changes in an editable revision.</p>{canUpload&&<form onSubmit={upload} className="space-y-3"><label htmlFor={`redline-upload-${contractId}`} className="block text-sm font-medium">Attach returned PDF or DOCX • version {current.version_number||1}</label><Input key={inputKey} id={`redline-upload-${contractId}`} type="file" accept=".pdf,.docx" disabled={busy} onChange={e=>setFile(e.target.files?.[0]||null)} className="min-h-11 max-w-full"/><p className="text-xs text-muted-foreground">Up to 10 MB. Uploading does not approve, sign, or send a contract.</p><Button type="submit" variant="outline" disabled={!file||busy}>{busy?'Uploading redline…':'Upload promoter redline'}</Button></form>}{error&&<p role="alert" className="text-sm text-destructive">{error}</p>}{query.isLoading?<p role="status" className="text-sm">Loading returned files…</p>:query.error?<p role="alert" className="text-sm text-destructive">{contractError(query.error)}</p>:<><ul>{query.data?.pages.flatMap(p=>p.items).map(record=><RedlineReviewRow key={`${record.id}-${record.review_revision}`} record={record} readOnly={locked} onChanged={refresh}/>)}</ul>{query.data?.pages[0]?.items.length===0&&<p className="text-sm text-muted-foreground">No promoter redlines attached yet.</p>}{query.hasNextPage&&<Button variant="outline" disabled={query.isFetchingNextPage} onClick={()=>query.fetchNextPage()}>{query.isFetchingNextPage?'Loading…':'Load older redlines'}</Button>}</>}</section>;
}
