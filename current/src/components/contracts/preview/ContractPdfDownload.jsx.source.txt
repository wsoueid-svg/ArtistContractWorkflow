import { useState } from 'react';
import { Download, Loader2 } from 'lucide-react';
import { Button } from '@/components/ui/button';

export default function ContractPdfDownload({ file, url, filename }) {
  const [busy, setBusy] = useState(false), [message, setMessage] = useState(''), [error, setError] = useState('');
  const download = async () => {
    setBusy(true); setError(''); setMessage('');
    try {
      if (!file || !url || file.type !== 'application/pdf') throw new Error('PDF unavailable');
      const link = document.createElement('a');
      link.href = url; link.download = filename; document.body.appendChild(link); link.click(); link.remove();
      setMessage('Download requested. Check your browser downloads or device Files app.');
    } catch { setError('The download could not start. Try again, or reopen this preview.'); }
    finally { setBusy(false); }
  };
  return <div className="space-y-2"><Button className="min-h-11 w-full sm:w-auto" onClick={download} disabled={!file || busy} data-testid="download-contract-pdf">{busy ? <Loader2 className="h-4 w-4 animate-spin" /> : <Download className="h-4 w-4" />}Download PDF</Button>{message && <p role="status" className="max-w-sm text-sm text-muted-foreground">{message}</p>}{error && <p role="alert" className="max-w-sm text-sm text-destructive">{error}</p>}</div>;
}
