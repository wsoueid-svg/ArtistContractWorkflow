import { useId, useState } from 'react';
import { base44 } from '@/api/base44Client';
import { Input } from '@/components/ui/input';
import { Label } from '@/components/ui/label';
export default function ContractPrivateUpload({value,onChange,label}) {
  const id=useId(),[busy,setBusy]=useState(false),[error,setError]=useState('');
  const upload=async e=>{
    const file=e.target.files?.[0];if(!file)return;setBusy(true);setError('');
    try{
      if(file.size>10*1024*1024||!file.size||!/\.pdf$/i.test(file.name)||new TextDecoder().decode((await file.arrayBuffer()).slice(0,5))!=='%PDF-')throw new Error('Choose a valid PDF up to 10 MB.');
      const {file_uri}=await base44.integrations.Core.UploadPrivateFile({file});onChange(file_uri);
    }catch(e){setError(e.message);}finally{setBusy(false);e.target.value='';}
  };
  return <div className="space-y-2"><Label htmlFor={id}>{label}</Label><Input id={id} type="file" accept=".pdf,application/pdf" disabled={busy} onChange={upload}/><p className="text-xs text-muted-foreground" role="status">{busy?'Uploading…':value?'Document attached (stored privately).':'PDF up to 10 MB (stored privately).'}</p>{error&&<p role="alert" className="text-sm text-destructive">{error}</p>}</div>;
}
