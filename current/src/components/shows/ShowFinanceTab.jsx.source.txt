import React, { useState, useEffect } from 'react';
import { useQuery, useQueryClient } from '@tanstack/react-query';
import { base44 } from '@/api/base44Client';
import { Lock } from 'lucide-react';
import { Button } from '@/components/ui/button';
import { Card, CardContent } from '@/components/ui/card';
import { FileText, Receipt, TrendingUp, Upload, ExternalLink, RefreshCw, ShieldAlert } from 'lucide-react';
import { LastModifiedAttribution } from './OrgAttributionBadge';
import { toast } from 'sonner';
import { formatCurrency, getCurrencySymbol } from '../shared/CurrencyUtils';
import UploadContractModal from './UploadContractModal';
import UploadInvoiceModal from './UploadInvoiceModal';
import DealStatusBadge from '../shared/DealStatusBadge';
import SplitInvoiceCard from './SplitInvoiceCard';
import { getInvoicePaymentStatus } from '@/lib/invoiceChase';

// FinancialRecord amounts remain raw integer cents until formatCurrency renders them.
const cents = (r) => (r.amount_base_eur ?? r.amount ?? 0);

const PAYMENT_STATUSES = {
  not_invoiced: undefined,
  invoiced: 'Sent',
  partially_paid: 'Partially Paid',
  paid: 'Paid',
  cancelled: 'Cancelled',
};

const getLivePaymentStatus = (show, invoices) => {
  const today = new Date().toISOString().slice(0, 10);
  const overdue = invoices.some(invoice => {
    const status = String(invoice.status || '').toUpperCase();
    const outstanding = !['PAID', 'CANCELLED', 'DRAFT'].includes(status);
    return status === 'OVERDUE' || (outstanding && invoice.due_date?.slice(0, 10) < today);
  });
  return overdue ? 'Overdue' : PAYMENT_STATUSES[getInvoicePaymentStatus(show, invoices)];
};

export default function ShowFinanceTab({ show }) {
  const [contractModalOpen, setContractModalOpen] = useState(false);
  const [invoiceModalOpen, setInvoiceModalOpen] = useState(false);
  const [currentUser, setCurrentUser] = useState(null);
  const [syncingDrafts, setSyncingDrafts] = useState(false);
  const [draftError, setDraftError] = useState('');
  const queryClient = useQueryClient();

  useEffect(() => {
    let mounted = true;
    base44.auth.me().then(u => { if (mounted) setCurrentUser(u); }).catch(() => {});
    return () => { mounted = false; };
  }, []);

  const canViewFinances = currentUser?.role === 'admin' ||
    currentUser?.finance_role === 'Finance Manager' ||
    currentUser?.finance_role === 'Finance Viewer';

  // Fetch linked contract
  const { data: contract } = useQuery({
    queryKey: ['contract', show?.contract_id],
    queryFn: () => base44.entities.Contract.get(show.contract_id),
    enabled: !!show?.contract_id,
  });

  // Fetch every active invoice linked to this show, including split billing.
  const { data: invoices = [] } = useQuery({
    queryKey: ['invoices-show', show?.id],
    queryFn: async () => {
      const [direct, parent] = await Promise.all([
        base44.entities.Invoice.filter({ show_id: show.id, is_deleted: false }),
        base44.entities.Invoice.filter({ parent_show_id: show.id, is_deleted: false }),
      ]);
      return [...new Map([...direct, ...parent].map(item => [item.id, item])).values()];
    },
    enabled: !!show?.id,
  });

  // Fetch financial records for this show
  const { data: financialRecords = [] } = useQuery({
    queryKey: ['financial-records-show', show?.id],
    queryFn: () => base44.entities.FinancialRecord.filter({ show_id: show.id }),
    enabled: !!show?.id,
  });

  const refreshShow = () => {
    queryClient.invalidateQueries({ queryKey: ['show', show.id] });
    queryClient.invalidateQueries({ queryKey: ['contract', show.contract_id] });
    queryClient.invalidateQueries({ queryKey: ['invoices-show', show.id] });
    queryClient.invalidateQueries({ queryKey: ['financial-records-show', show.id] });
  };

  const totalIncome = financialRecords.filter(r => r.type === 'income').reduce((s, r) => s + cents(r), 0);
  const totalExpenses = financialRecords.filter(r => r.type === 'expense').reduce((s, r) => s + cents(r), 0);
  const invoiceTypeOrder = { DEPOSIT: 0, BALANCE: 1, FULL_FEE: 2 };
  const sortedInvoices = [...invoices].sort((a, b) =>
    (invoiceTypeOrder[a.invoice_type || 'FULL_FEE'] ?? 3) - (invoiceTypeOrder[b.invoice_type || 'FULL_FEE'] ?? 3)
  );
  const immutableStatuses = ['Sent', 'Viewed', 'Partially Paid', 'Paid', 'Overdue', 'Disputed', 'Cancelled'];
  const financialsLocked = invoices.some(invoice => immutableStatuses.includes(invoice.status));
  const syncDrafts = async () => {
    setSyncingDrafts(true);
    setDraftError('');
    try {
      await base44.functions.invoke('generateShowFinancialDrafts', { show_id: show.id, generate: 'invoice' });
      toast.success(invoices.length ? 'Financial drafts synchronized' : 'Financial drafts generated');
      refreshShow();
    } catch (error) {
      setDraftError(error.response?.data?.error || error.message || 'Financial draft sync failed');
    } finally {
      setSyncingDrafts(false);
    }
  };

  if (currentUser && !canViewFinances) {
    return (
      <div className="flex flex-col items-center justify-center py-16 text-center">
        <Lock className="w-8 h-8 text-gray-300 mb-3" />
        <p className="text-sm font-semibold text-gray-500">Finance access required</p>
        <p className="text-xs text-gray-400 mt-1">Contact an administrator to gain Finance Manager access.</p>
      </div>
    );
  }

  return (
    <div className="space-y-4">
      <UploadContractModal show={show} open={contractModalOpen} onClose={() => setContractModalOpen(false)} onSaved={refreshShow} />
      <UploadInvoiceModal show={show} open={invoiceModalOpen} onClose={() => setInvoiceModalOpen(false)} onSaved={refreshShow} />

      {/* Attribution line */}
      {(show.last_modified_by_org_id || show.last_modified_at) && (
        <LastModifiedAttribution
          orgId={show.last_modified_by_org_id}
          timestamp={show.last_modified_at}
        />
      )}

        {/* ── Derived Status Row ─────────────────────────────────────────────── */}
        <Card>
          <CardContent className="p-4">
            <p className="text-xs font-semibold text-gray-500 uppercase tracking-wider mb-3">Live Financial Status</p>
            <div className="flex flex-wrap items-center gap-2">
              <DealStatusBadge kind="invoice" status={getLivePaymentStatus(show, invoices)} size="sm" />
              <span className="text-xs text-gray-500">Across {invoices.length} active invoice{invoices.length === 1 ? '' : 's'}</span>
            </div>
            <p className="text-[10px] text-gray-400 mt-3">These statuses update automatically when invoices or payments change.</p>
          </CardContent>
        </Card>

        {/* ── Contract Section ──────────────────────────────────────────────── */}
        <Card>
          <CardContent className="p-5">
            <div className="flex items-center justify-between mb-3">
              <div className="flex items-center gap-2">
                <FileText className="w-4 h-4 text-gray-500" />
                <h3 className="font-semibold text-black">Contract</h3>
              </div>
              {!show.contract_id && (
                <Button
                  size="sm"
                  variant="outline"
                  onClick={() => setContractModalOpen(true)}
                  className="h-8 text-xs gap-1.5"
                >
                  <Upload className="w-3.5 h-3.5" />
                  Add Contract
                </Button>
              )}
            </div>
            {contract ? (
              <div className="space-y-2">
                <div className="flex items-center gap-2 flex-wrap">
                  <p className="text-sm font-medium text-black">{contract.title}</p>
                  <DealStatusBadge kind="contract" status={contract.status} />
                </div>
                <p className="text-xs text-gray-500">
                  Value: {getCurrencySymbol(contract.currency)}{(contract.value || 0).toLocaleString()} {contract.currency}
                  {contract.value_base_eur && contract.currency !== 'EUR' && (
                    <span className="ml-1 text-gray-400">(≈ €{contract.value_base_eur.toLocaleString()})</span>
                  )}
                </p>
                {contract.contract_file_url && (
                  <a href={`/contract-preview?contract=${encodeURIComponent(contract.id)}`} target="_blank" rel="noopener noreferrer"
                    className="inline-flex items-center gap-1 text-xs text-blue-600 hover:underline">
                    <ExternalLink className="w-3 h-3" /> View Contract PDF
                  </a>
                )}
              </div>
            ) : (
              <p className="text-sm text-gray-400">No contract linked. Click "Add Contract" to upload or enter details manually.</p>
            )}
          </CardContent>
        </Card>

        {(financialsLocked || draftError) && (
          <div className="flex items-start gap-3 rounded-xl border border-red-200 bg-red-50 p-4 text-red-800">
            <ShieldAlert className="mt-0.5 h-5 w-5 shrink-0" />
            <div><p className="text-sm font-semibold">{financialsLocked ? 'Financials already dispatched. Revisions require Finance approval.' : draftError}</p><p className="mt-1 text-xs text-red-600">Existing ledger entries have not been changed.</p></div>
          </div>
        )}

        {/* ── Invoice Section ───────────────────────────────────────────────── */}
        <Card>
          <CardContent className="p-5">
            <div className="flex items-center justify-between mb-3">
              <div className="flex items-center gap-2">
                <Receipt className="w-4 h-4 text-gray-500" />
                <h3 className="font-semibold text-black">Invoice</h3>
              </div>
              <div className="flex flex-wrap gap-2">
                <Button size="sm" variant="outline" onClick={syncDrafts} disabled={financialsLocked || syncingDrafts || !['confirmed', 'completed'].includes(show.status)} className="min-h-[44px] gap-1.5 text-xs">
                  <RefreshCw className={`h-3.5 w-3.5 ${syncingDrafts ? 'animate-spin' : ''}`} />{invoices.length ? 'Sync Drafts' : 'Generate Drafts'}
                </Button>
                {invoices.length === 0 && <Button size="sm" variant="outline" onClick={() => setInvoiceModalOpen(true)} className="min-h-[44px] gap-1.5 text-xs"><Upload className="w-3.5 h-3.5" />Add Invoice</Button>}
              </div>
            </div>
            {sortedInvoices.length > 0 ? (
              <div className="grid gap-3 md:grid-cols-2">
                {sortedInvoices.map(invoice => <SplitInvoiceCard key={invoice.id} invoice={invoice} />)}
              </div>
            ) : (
              <p className="text-sm text-gray-400">No invoice linked. Click "Add Invoice" to upload or enter details manually.</p>
            )}
          </CardContent>
        </Card>

        {/* ── Financial Records Summary ─────────────────────────────────────── */}
        <Card>
          <CardContent className="p-5">
            <div className="flex items-center gap-2 mb-3">
              <TrendingUp className="w-4 h-4 text-gray-500" />
              <h3 className="font-semibold text-black">Financial Records</h3>
            </div>
            {financialRecords.length === 0 ? (
              <p className="text-sm text-gray-400">No financial records yet. They are created automatically when a draft invoice is generated.</p>
            ) : (
              <>
                <div className="grid grid-cols-2 gap-3 mb-3">
                  <div className="bg-green-50 rounded-lg p-3">
                    <p className="text-xs text-green-600 font-medium">Total Income</p>
                    <p className="text-lg font-bold text-green-700">{formatCurrency(totalIncome, 'EUR')}</p>
                  </div>
                  <div className="bg-red-50 rounded-lg p-3">
                    <p className="text-xs text-red-600 font-medium">Total Expenses</p>
                    <p className="text-lg font-bold text-red-700">{formatCurrency(totalExpenses, 'EUR')}</p>
                  </div>
                </div>
                <div className="space-y-1.5">
                  {financialRecords.map(r => (
                    <div key={r.id} className="flex items-center justify-between text-xs py-1.5 border-b border-gray-50 last:border-0">
                      <div className="flex items-center gap-2">
                        <span className={`w-1.5 h-1.5 rounded-full flex-shrink-0 ${r.type === 'income' ? 'bg-green-500' : 'bg-red-400'}`} />
                        <span className="text-gray-700 truncate max-w-[200px]">{r.description || r.category}</span>
                      </div>
                      <span className={`font-semibold tabular-nums ${r.type === 'income' ? 'text-green-700' : 'text-red-600'}`}>
                        {r.type === 'income' ? '+' : '-'}{formatCurrency(cents(r), 'EUR')}
                      </span>
                    </div>
                  ))}
                </div>
              </>
            )}
          </CardContent>
        </Card>

    </div>
  );
}
