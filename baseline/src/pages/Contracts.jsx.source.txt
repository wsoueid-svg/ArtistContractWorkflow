import React, { useState } from 'react';
import { base44 } from '@/api/base44Client';
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query';
import { Button } from '@/components/ui/button';
import { Input } from '@/components/ui/input';
import { Card, CardContent } from '@/components/ui/card';
import { Badge } from '@/components/ui/badge';
import { Plus, Search, FileText, ExternalLink, List, Table, Eye, FolderOpen } from 'lucide-react';
import { toast } from 'sonner';
import ContractForm from '../components/contracts/ContractForm';
import ContractAlerts from '../components/contracts/ContractAlerts';
import StageBadge from '../components/shared/StageBadge';
import SkeletonBlock from '../components/shared/SkeletonBlock';
import useArtistLookup from '../hooks/useArtistLookup';

export default function Contracts() {
  const [searchQuery, setSearchQuery] = useState('');
  const [showForm, setShowForm] = useState(false);
  const [editingContract, setEditingContract] = useState(null);
  const [filterStatus, setFilterStatus] = useState('all');
  const [filterArtist, setFilterArtist] = useState('all');
  const [view, setView] = useState('cards');
  const queryClient = useQueryClient();

  const { data: contracts = [], isLoading } = useQuery({
    queryKey: ['contracts'],
    queryFn: () => base44.entities.Contract.list('-created_date'),
  });

  const { artists, allArtists } = useArtistLookup();

  const { data: contacts = [] } = useQuery({
    queryKey: ['contacts'],
    queryFn: () => base44.entities.Contact.list(),
  });

  const createMutation = useMutation({
    mutationFn: (data) => base44.entities.Contract.create(data),
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ['contracts'] });
      setShowForm(false);
      setEditingContract(null);
      toast.success('Contract created successfully.');
    },
    onError: () => toast.error('Could not create the contract. Please try again.'),
  });

  const updateMutation = useMutation({
    mutationFn: ({ id, data }) => base44.entities.Contract.update(id, data),
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ['contracts'] });
      setShowForm(false);
      setEditingContract(null);
      toast.success('Contract updated successfully.');
    },
    onError: () => toast.error('Could not update the contract. Please try again.'),
  });

  const deleteMutation = useMutation({
    mutationFn: (id) => base44.entities.Contract.delete(id),
    onSuccess: () => {
      queryClient.invalidateQueries({ queryKey: ['contracts'] });
      toast.success('Contract deleted.');
    },
    onError: () => toast.error('Could not delete the contract. Please try again.'),
  });

  const handleSubmit = (data) => {
    if (editingContract) {
      updateMutation.mutate({ id: editingContract.id, data });
    } else {
      createMutation.mutate(data);
    }
  };

  const filteredContracts = contracts.filter((contract) => {
    const matchesSearch =
      contract.title?.toLowerCase().includes(searchQuery.toLowerCase());
    const matchesStatus = filterStatus === 'all' || contract.status === filterStatus;
    const matchesArtist = filterArtist === 'all' || contract.artist_id === filterArtist;
    return matchesSearch && matchesStatus && matchesArtist;
  });

  const statusColors = {
    draft: 'bg-gray-100 text-gray-700 border-gray-300',
    requested: 'bg-blue-100 text-blue-700 border-blue-300',
    ready_to_send: 'bg-purple-100 text-purple-700 border-purple-300',
    sent: 'bg-yellow-100 text-yellow-700 border-yellow-300',
    signed: 'bg-green-100 text-green-700 border-green-300',
    paid: 'bg-green-100 text-green-700 border-green-300',
    cancelled: 'bg-red-100 text-red-700 border-red-300',
  };

  const typeColors = {
    show_agreement: 'bg-purple-100 text-purple-700',
    management: 'bg-blue-100 text-blue-700',
    label: 'bg-green-100 text-green-700',
    sponsorship: 'bg-yellow-100 text-yellow-700',
    merchandise: 'bg-pink-100 text-pink-700',
    other: 'bg-gray-100 text-gray-700',
  };

  return (
    <div className="space-y-6">
      {/* Header */}
      <div className="flex flex-col sm:flex-row justify-between items-start sm:items-center gap-4 border-b-4 border-yellow-400 pb-4">
        <div>
          <h1 className="text-4xl font-bold text-black">Contracts</h1>
          <p className="text-gray-600 mt-1">{contracts.length} contracts</p>
        </div>
        <div className="flex gap-2">
          <Button
            variant={view === 'cards' ? 'default' : 'outline'}
            onClick={() => setView('cards')}
            size="sm"
            className={view === 'cards' ? 'bg-yellow-400 text-black hover:bg-yellow-500' : ''}
          >
            <List className="w-4 h-4 mr-1" />
            Cards
          </Button>
          <Button
            variant={view === 'table' ? 'default' : 'outline'}
            onClick={() => setView('table')}
            size="sm"
            className={view === 'table' ? 'bg-yellow-400 text-black hover:bg-yellow-500' : ''}
          >
            <Table className="w-4 h-4 mr-1" />
            Table
          </Button>
          <Button
            onClick={() => {
              setEditingContract(null);
              setShowForm(true);
            }}
            className="bg-black text-yellow-400 hover:bg-gray-900 font-semibold"
          >
            <Plus className="w-5 h-5 mr-2" />
            New Contract
          </Button>
        </div>
      </div>

      {/* Filters */}
      <div className="flex flex-col sm:flex-row gap-4">
        <div className="relative flex-1 max-w-md">
          <Search className="absolute left-3 top-1/2 transform -translate-y-1/2 text-gray-400 w-5 h-5" />
          <Input
            placeholder="Search contracts..."
            value={searchQuery}
            onChange={(e) => setSearchQuery(e.target.value)}
            className="pl-10 border-2 border-gray-200 focus:border-yellow-400"
          />
        </div>
        <div className="flex gap-2 flex-wrap">
          <select
            value={filterArtist}
            onChange={(e) => setFilterArtist(e.target.value)}
            className="px-3 py-1.5 text-sm border-2 border-gray-200 rounded-md focus:border-yellow-400 focus:outline-none"
          >
            <option value="all">All Artists</option>
            {artists.map((artist) => (
              <option key={artist.id} value={artist.id}>
                {artist.name}
              </option>
            ))}
          </select>
          {['all', 'draft', 'sent', 'signed', 'paid'].map((status) => (
            <Button
              key={status}
              variant={filterStatus === status ? 'default' : 'outline'}
              onClick={() => setFilterStatus(status)}
              size="sm"
              className={
                filterStatus === status
                  ? 'bg-yellow-400 text-black hover:bg-yellow-500'
                  : ''
              }
            >
              {status === 'all' ? 'All' : status.replace('_', ' ')}
            </Button>
          ))}
        </div>
      </div>

      {/* Contract Form Modal */}
      {showForm && (
        <div className="fixed inset-0 bg-black bg-opacity-50 z-50 flex items-center justify-center p-4">
          <div className="bg-white rounded-lg max-w-4xl w-full max-h-[90vh] overflow-y-auto">
            <ContractForm
              contract={editingContract}
              artists={artists}
              contacts={contacts}
              onSubmit={handleSubmit}
              onDelete={(id) => {
                deleteMutation.mutate(id);
                setShowForm(false);
                setEditingContract(null);
              }}
              onCancel={() => {
                setShowForm(false);
                setEditingContract(null);
              }}
            />
          </div>
        </div>
      )}

      {/* Contract Renewal / Expiration Alerts */}
      <ContractAlerts contracts={contracts} artists={artists} />

      {/* Contracts List */}
      {isLoading ? (
        <div className="space-y-4">
          {[1, 2, 3].map(i => <SkeletonBlock key={i} type="card" />)}
        </div>
      ) : filteredContracts.length === 0 ? (
        <div className="text-center py-12">
          <FileText className="w-16 h-16 text-gray-300 mx-auto mb-4" />
          <p className="text-gray-500">No contracts found</p>
        </div>
      ) : view === 'table' ? (
        <div className="border-2 border-gray-200 rounded-lg overflow-hidden">
          <div className="overflow-x-auto">
            <table className="w-full">
              <thead className="bg-black text-yellow-400">
                <tr>
                  <th className="text-left py-3 px-4 font-semibold text-sm border-r border-gray-700">Title</th>
                  <th className="text-left py-3 px-4 font-semibold text-sm border-r border-gray-700">Artist</th>
                  <th className="text-left py-3 px-4 font-semibold text-sm border-r border-gray-700">Type</th>
                  <th className="text-left py-3 px-4 font-semibold text-sm border-r border-gray-700">Contact</th>
                  <th className="text-right py-3 px-4 font-semibold text-sm border-r border-gray-700">Value</th>
                  <th className="text-left py-3 px-4 font-semibold text-sm border-r border-gray-700">Start Date</th>
                  <th className="text-center py-3 px-4 font-semibold text-sm border-r border-gray-700">Status</th>
                  <th className="text-center py-3 px-4 font-semibold text-sm">Actions</th>
                </tr>
              </thead>
              <tbody>
                {filteredContracts.map((contract) => {
                  const artist = allArtists.find((a) => a.id === contract.artist_id);
                  const contact = contacts.find((c) => c.id === contract.contact_id);
                  return (
                    <tr key={contract.id} className="border-b border-gray-200 hover:bg-yellow-50 transition">
                      <td className="py-3 px-4 text-sm font-medium border-r border-gray-100">
                        {contract.title}
                      </td>
                      <td className={`py-3 px-4 text-sm border-r border-gray-100 ${artist?._deleted ? 'italic text-gray-400' : ''}`}>{artist?.name || '-'}{artist?._deleted ? ' (del)' : ''}</td>
                      <td className="py-3 px-4 text-sm border-r border-gray-100">
                        <Badge className={`text-xs ${typeColors[contract.type || 'other']}`}>
                          {contract.type?.replace('_', ' ') || 'other'}
                        </Badge>
                      </td>
                      <td className="py-3 px-4 text-sm border-r border-gray-100">{contact?.name || '-'}</td>
                      <td className="py-3 px-4 text-sm text-right font-semibold border-r border-gray-100">
                        {contract.value ? `${contract.currency || 'USD'} ${contract.value.toLocaleString()}` : '-'}
                      </td>
                      <td className="py-3 px-4 text-sm border-r border-gray-100 whitespace-nowrap">
                        {contract.start_date ? new Date(contract.start_date).toLocaleDateString() : '-'}
                      </td>
                      <td className="py-3 px-4 text-center border-r border-gray-100">
                        <StageBadge stage={contract.status || 'draft'} />
                      </td>
                      <td className="py-3 px-4 text-center">
                        <div className="flex gap-1 justify-center">
                          <Button
                            size="sm"
                            variant="ghost"
                            onClick={() => {
                              setEditingContract(contract);
                              setShowForm(true);
                            }}
                            className="h-8 px-2 hover:bg-yellow-100"
                          >
                            Edit
                          </Button>
                          {contract.contract_file_url && (
                            <a href={contract.contract_file_url} target="_blank" rel="noopener noreferrer">
                              <Button size="sm" variant="ghost" className="h-8 px-2 hover:bg-yellow-100">
                                <Eye className="w-4 h-4" />
                              </Button>
                            </a>
                          )}
                        </div>
                      </td>
                    </tr>
                  );
                })}
              </tbody>
            </table>
          </div>
        </div>
      ) : (
        <div className="space-y-4">
          {filteredContracts.map((contract) => {
            const artist = allArtists.find((a) => a.id === contract.artist_id);
            const contact = contacts.find((c) => c.id === contract.contact_id);
            return (
              <Card
                key={contract.id}
                className="border-2 border-gray-200 hover:border-yellow-400 transition-all cursor-pointer"
                onClick={() => {
                  setEditingContract(contract);
                  setShowForm(true);
                }}
              >
                <CardContent className="p-6">
                  <div className="flex items-start gap-4">
                    <div className="p-3 bg-black rounded-lg">
                      <FileText className="w-6 h-6 text-yellow-400" />
                    </div>

                    <div className="flex-1 min-w-0">
                      <div className="flex items-center gap-2 mb-2">
                        <Badge className={typeColors[contract.type || 'other']}>
                          {contract.type?.replace('_', ' ') || 'other'}
                        </Badge>
                        <StageBadge stage={contract.status || 'draft'} />
                      </div>

                      <h3 className="font-bold text-black text-xl mb-1">
                        {contract.title}
                      </h3>
                      <p className={artist?._deleted ? 'text-gray-400 italic' : 'text-gray-600'}>
                        {artist?.name || 'Unknown Artist'}{artist?._deleted ? ' (deleted)' : ''}
                        {contact && ` • ${contact.name}`}
                      </p>

                      <div className="grid grid-cols-2 md:grid-cols-4 gap-4 mt-4">
                        {contract.value && (
                          <div>
                            <p className="text-xs text-gray-500">Value</p>
                            <p className="font-semibold text-black">
                              {contract.currency} {contract.value.toLocaleString()}
                            </p>
                          </div>
                        )}
                        {contract.commission_amount && (
                          <div>
                            <p className="text-xs text-gray-500">Commission</p>
                            <p className="font-semibold text-black">
                              {contract.currency} {contract.commission_amount.toLocaleString()}
                            </p>
                          </div>
                        )}
                        {contract.start_date && (
                          <div>
                            <p className="text-xs text-gray-500">Start Date</p>
                            <p className="font-semibold text-black">
                              {new Date(contract.start_date).toLocaleDateString()}
                            </p>
                          </div>
                        )}
                        {contract.end_date && (
                          <div>
                            <p className="text-xs text-gray-500">End Date</p>
                            <p className="font-semibold text-black">
                              {new Date(contract.end_date).toLocaleDateString()}
                            </p>
                          </div>
                        )}
                      </div>

                      {(contract.contract_file_url || contract.signed_contract_url || contract.dropbox_folder_url) && (
                        <div className="flex gap-2 mt-4 flex-wrap">
                          {contract.contract_file_url && (
                            <a
                              href={contract.contract_file_url}
                              target="_blank"
                              rel="noopener noreferrer"
                              onClick={(e) => e.stopPropagation()}
                              className="text-xs px-3 py-1 bg-gray-100 rounded flex items-center gap-1 hover:bg-gray-200"
                            >
                              Contract <ExternalLink className="w-3 h-3" />
                            </a>
                          )}
                          {contract.signed_contract_url && (
                            <a
                              href={contract.signed_contract_url}
                              target="_blank"
                              rel="noopener noreferrer"
                              onClick={(e) => e.stopPropagation()}
                              className="text-xs px-3 py-1 bg-green-100 rounded flex items-center gap-1 hover:bg-green-200"
                            >
                              Signed <ExternalLink className="w-3 h-3" />
                            </a>
                          )}
                          {contract.dropbox_folder_url && (
                            <a
                              href={contract.dropbox_folder_url}
                              target="_blank"
                              rel="noopener noreferrer"
                              onClick={(e) => e.stopPropagation()}
                              className="text-xs px-3 py-1 bg-blue-50 text-blue-700 rounded flex items-center gap-1 hover:bg-blue-100"
                            >
                              <FolderOpen className="w-3 h-3" /> Dropbox
                            </a>
                          )}
                        </div>
                      )}
                    </div>

                    <Button
                      variant="ghost"
                      size="sm"
                      onClick={(e) => {
                        e.stopPropagation();
                        setEditingContract(contract);
                        setShowForm(true);
                      }}
                      className="text-black hover:bg-yellow-100"
                    >
                      Edit
                    </Button>
                  </div>
                </CardContent>
              </Card>
            );
          })}
        </div>
      )}
    </div>
  );
}
