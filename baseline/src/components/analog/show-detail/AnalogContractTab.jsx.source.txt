import React, { useState } from 'react';
import { useQuery, useQueryClient } from '@tanstack/react-query';
import { Upload } from 'lucide-react';
import { base44 } from '@/api/base44Client';
import { Button } from '@/components/ui/button';
import GenerateContractButton from '@/components/shows/GenerateContractButton';
import UploadContractModal from '@/components/shows/UploadContractModal';
import ContractSignatureControl from '@/components/contracts/ContractSignatureControl';
import DealStatusBadge from '@/components/shared/DealStatusBadge';
import ContractHistory from './ContractHistory';
import DetailPanel from './DetailPanel';
import { getContractualSetWindow } from '@/lib/showTiming';

export default function AnalogContractTab({ show, artist }) {
  const [uploadOpen, setUploadOpen] = useState(false);
  const queryClient = useQueryClient();
  const { data: contracts = [] } = useQuery({ queryKey: ['show-contracts', show.id], queryFn: () => base44.entities.Contract.filter({ show_id: show.id }, '-created_date'), enabled: !!show.id });
  const current = contracts.find(item => item.id === show.contract_id) || contracts[0] || null;
  const refresh = () => { queryClient.invalidateQueries({ queryKey: ['show', show.id] }); queryClient.invalidateQueries({ queryKey: ['show-contracts', show.id] }); };
  return (
    <div className="space-y-0 md:space-y-4">
      <DetailPanel>
        <div className="flex flex-col gap-3 sm:flex-row sm:items-start sm:justify-between">
          <div>
            <h2 className="font-semibold text-slate-900">Contract workflow</h2>
            <p className="mt-1 text-[13px] text-slate-500 sm:text-sm">Generate, dispatch, upload, and sign the legal show agreement.</p>
          </div>
          <div className="flex flex-wrap gap-2">
            <Button variant="outline" onClick={() => setUploadOpen(true)} className="min-h-[44px]"><Upload className="h-4 w-4" />Upload version</Button>
            <GenerateContractButton show={{ ...show, artist_name: artist?.stage_name || artist?.name }} contractId={show.contract_id} contract={current} />
          </div>
        </div>
        <div className="mt-4 border-t border-slate-200 pt-4">
          <p className="text-xs font-semibold uppercase tracking-wide text-slate-400">Contractual Set Window</p>
          <p className="mt-1 text-[13px] tabular-nums text-slate-700 sm:text-sm">{getContractualSetWindow(show)}</p>
        </div>
        {current && (
          <div className="mt-5 flex flex-wrap items-center gap-3 border-t border-slate-200 pt-4">
            <DealStatusBadge kind="contract" status={current.status} size="sm" />
            <span className="text-[13px] text-slate-700 sm:text-sm">{current.title}</span>
            {current.contract_file_url && <a href={current.contract_file_url} target="_blank" rel="noreferrer" className="inline-flex min-h-[44px] items-center text-[13px] text-blue-700 underline sm:text-sm">View current PDF</a>}
            <ContractSignatureControl contract={current} onChange={refresh} />
          </div>
        )}
      </DetailPanel>
      <DetailPanel>
        <h2 className="mb-4 font-semibold text-slate-900">Version &amp; status history</h2>
        <ContractHistory contracts={contracts} />
      </DetailPanel>
      <UploadContractModal show={show} open={uploadOpen} onClose={() => setUploadOpen(false)} onSaved={refresh} />
    </div>
  );
}
