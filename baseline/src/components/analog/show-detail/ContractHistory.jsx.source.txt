import React from 'react';
import DealStatusBadge from '@/components/shared/DealStatusBadge';

export default function ContractHistory({ contracts = [] }) {
  if (!contracts.length) return <p className="text-sm text-slate-400">No contract versions recorded yet.</p>;
  return <div className="divide-y divide-slate-200 border-y border-slate-200">{contracts.map((contract, index) => <div key={contract.id} className="flex min-h-[58px] flex-wrap items-center justify-between gap-2 py-3"><div><p className="text-sm font-medium text-slate-800">{contract.title || `Contract version ${contracts.length - index}`}</p><p className="text-xs text-slate-400">{new Date(contract.created_date).toLocaleDateString()}</p></div><div className="flex items-center gap-2"><DealStatusBadge kind="contract" status={contract.status} size="sm" />{contract.contract_file_url && <a href={contract.contract_file_url} target="_blank" rel="noreferrer" className="text-xs font-medium text-blue-700 underline">View PDF</a>}</div></div>)}</div>;
}
