import React from 'react';
import { useQueryClient } from '@tanstack/react-query';
import { base44 } from '@/api/base44Client';
import { Dialog, DialogContent, DialogHeader, DialogTitle } from '@/components/ui/dialog';
import ContactForm from '@/components/crm/ContactForm';
import { CONTACT_DIRECTORY_KEY } from '@/hooks/useContactSearch';

// Full CRM contact form in a modal; returns the created contact to the caller.
export default function NewContactDialog({ open, initialName, onClose, onCreated }) {
  const queryClient = useQueryClient();

  const handleSubmit = async (data) => {
    const created = await base44.entities.Contact.create(data);
    await queryClient.invalidateQueries({ queryKey: CONTACT_DIRECTORY_KEY });
    onCreated?.(created);
    onClose();
  };

  return (
    <Dialog open={open} onOpenChange={(v) => !v && onClose()}>
      <DialogContent className="max-w-2xl max-h-[90vh] overflow-y-auto">
        <DialogHeader><DialogTitle>New contact</DialogTitle></DialogHeader>
        {open && (
          <ContactForm
            key={initialName}
            contact={{ name: initialName || '', type: 'promoter', social_media: {}, tags: [], default_deal_terms: {} }}
            isNew
            onSubmit={handleSubmit}
            onCancel={onClose}
          />
        )}
      </DialogContent>
    </Dialog>
  );
}
