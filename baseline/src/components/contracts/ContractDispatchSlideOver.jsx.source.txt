import React, { useState, useEffect, useRef } from 'react';
import { useQueryClient } from '@tanstack/react-query';
import { invalidateShowQueries } from '@/lib/showEdit';
import { base44 } from '@/api/base44Client';
import { Sheet, SheetContent, SheetTitle, SheetDescription } from '@/components/ui/sheet';
import { Button } from '@/components/ui/button';
import { Input } from '@/components/ui/input';
import { Textarea } from '@/components/ui/textarea';
import {
  Send, Loader2, FileText, CheckCircle2, AlertTriangle, CornerUpLeft, Mail,
} from 'lucide-react';
import EmailChipInput from '@/components/contracts/EmailChipInput';
import PersonalTemplatePicker from '@/components/contracts/PersonalTemplatePicker';
import ContractReviewEditor from '@/components/contracts/ContractReviewEditor';
import RiderAttachmentPicker from '@/components/contracts/RiderAttachmentPicker';

const fmtDate = (d) => {
  if (!d) return '';
  try { return new Date(d).toLocaleDateString('en-GB', { day: '2-digit', month: 'short', year: 'numeric' }); }
  catch { return d; }
};

export default function ContractDispatchSlideOver({
  open, onClose, show, contractId, contractUrl, onSent,
}) {
  const queryClient = useQueryClient();
  const [loading, setLoading] = useState(true);
  const [sending, setSending] = useState(false);
  const [error, setError] = useState(null);
  const [sent, setSent] = useState(null);
  const [gmailBlocked, setGmailBlocked] = useState(false);
  const [contractState, setContractState] = useState(null);
  const [resetting, setResetting] = useState(false);
  const [clock, setClock] = useState(Date.now());
  const inFlightRef = useRef(false);
  const dispatchKeyRef = useRef(crypto.randomUUID());

  const [to, setTo] = useState('');
  const [cc, setCc] = useState([]);
  const [subject, setSubject] = useState('');
  const [emailBody, setEmailBody] = useState('');
  const [thread, setThread] = useState(null);
  const [me, setMe] = useState(null);
  const [reviewing, setReviewing] = useState(false);
  const [pdfUrl, setPdfUrl] = useState(contractUrl);
  const [riders, setRiders] = useState([]);
  useEffect(() => { setPdfUrl(contractUrl); }, [contractUrl]);
  useEffect(() => { if (open) { setReviewing(false); setRiders([]); } }, [open]);

  const handleRegenerated = async () => {
    const fresh = await base44.entities.Contract.get(contractId);
    setContractState(fresh);
    setPdfUrl(fresh?.contract_file_url || pdfUrl);
    await invalidateShowQueries(queryClient, show.id);
    setReviewing(false);
  };

  useEffect(() => {
    let alive = true;
    base44.auth.me().then(u => { if (alive) setMe(u); }).catch(() => {});
    return () => { alive = false; };
  }, []);

  const threadId = show?.active_thread_id || show?.gmail_thread_id || null;
  const fileName = (pdfUrl || '').split('/').pop()?.split('?')[0] || 'contract.pdf';
  const heartbeatAt = contractState?.dispatch_heartbeat_at
    ? new Date(contractState.dispatch_heartbeat_at).getTime()
    : 0;
  const isDispatching = contractState?.status === 'DISPATCHING';
  const isStale = isDispatching && heartbeatAt > 0 && clock - heartbeatAt > 5 * 60 * 1000;

  useEffect(() => {
    if (!open || !isDispatching) return undefined;
    const timer = window.setInterval(() => setClock(Date.now()), 1000);
    return () => window.clearInterval(timer);
  }, [open, isDispatching]);

  useEffect(() => {
    if (!open || !show?.id) return;
    let alive = true;

    (async () => {
      setLoading(true);
      setError(null);
      dispatchKeyRef.current = crypto.randomUUID();
      setSent(null);
      setGmailBlocked(false);

      try {
        // Promoter contact + artist for the default subject/body
        const contactId = show.contract_contact_id || show.client_contact_id || show.contact_id;
        const [contact, artist, currentContract] = await Promise.all([
          contactId ? base44.entities.Contact.get(contactId).catch(() => null) : null,
          show.artist_id ? base44.entities.Artist.get(show.artist_id).catch(() => null) : null,
          contractId ? base44.entities.Contract.get(contractId).catch(() => null) : null,
        ]);
        setContractState(currentContract);

        const promoterEmail =
          contact?.email || contact?.contact_persons?.[0]?.email || show.contact_email || '';
        const artistName = artist?.stage_name || artist?.name || '';
        const defaultSubject = `CONTRACT — ${[artistName, show.event_name, fmtDate(show.date)].filter(Boolean).join(' — ')}`;

        let nextTo = promoterEmail;
        let nextCc = [];
        let nextSubject = defaultSubject;
        let meta = null;

        if (threadId) {
          const res = await base44.functions.invoke('getGmailThreadMetadata', { thread_id: threadId });
          const d = res.data || {};
          if (d.connected === false) setGmailBlocked(true);
          if (d.subject) {
            meta = d;
            nextSubject = d.subject.replace(/^((re|fwd|fw)\s*:\s*)+/i, '');
            nextTo = promoterEmail || d.to?.[0] || d.from || '';
            // Reply-all preserve: every thread participant except me and the promoter
            nextCc = (d.participants || []).filter(
              (e) => e && e.toLowerCase() !== String(nextTo).toLowerCase()
            );
          }
        }

        // Union in the show's assigned agents
        const agentIds = show.responsible_agent_ids || [];
        if (agentIds.length) {
          const agents = await Promise.all(
            agentIds.map((id) => base44.entities.BookingAgent.get(id).catch(() => null))
          );
          agents.forEach((a) => {
            const e = a?.email?.toLowerCase();
            if (e && e !== String(nextTo).toLowerCase() && !nextCc.includes(e)) nextCc.push(e);
          });
        }

        if (!alive) return;
        setThread(meta);
        setTo(nextTo);
        setCc(nextCc);
        setSubject(nextSubject);
        setEmailBody(
          `Hi${contact?.contact_persons?.[0]?.name ? ` ${contact.contact_persons[0].name}` : ''},\n\n` +
          `Please find attached the contract for ${artistName || 'the artist'}${show.event_name ? ` at ${show.event_name}` : ''}${show.date ? ` on ${fmtDate(show.date)}` : ''}.\n\n` +
          `Kindly review, sign and return it at your earliest convenience. Once we receive the signed copy we will issue the corresponding invoice.\n\n` +
          `Any questions, just reply to this email.\n\nBest regards`
        );
      } catch (e) {
        if (alive) setError(e?.response?.data?.error || e.message || 'Could not prepare the dispatch');
      } finally {
        if (alive) setLoading(false);
      }
    })();

    return () => { alive = false; };
  }, [open, show?.id, threadId, contractId]);

  const handleSend = async () => {
    if (inFlightRef.current) return;
    inFlightRef.current = true;
    setSending(true);
    setError(null);
    const idempotencyKey = dispatchKeyRef.current;
    try {
      const res = await base44.functions.invoke('dispatchContract', {
        show_id: show.id,
        contract_id: contractId || undefined,
        to,
        cc,
        subject: threadId ? `Re: ${subject}` : subject,
        body: emailBody,
        attachment_name: fileName,
        idempotency_key: idempotencyKey,
        rider_attachments: riders.map(({ source, id, file_uri, filename }) => ({ source, id, file_uri, filename })),
      });
      setContractState((current) => ({ ...current, status: 'SENT', dispatch_lock: null, dispatch_heartbeat_at: null }));
      setSent(res.data);
      await invalidateShowQueries(queryClient, show.id);
      onSent?.(res.data);
    } catch (e) {
      const msg = e?.response?.data?.error || e.message || 'Failed to send';
      if (/not connected/i.test(msg)) setGmailBlocked(true);
      setError(msg);
    } finally {
      inFlightRef.current = false;
      setSending(false);
    }
  };

  const handleReset = async () => {
    setResetting(true);
    setError(null);
    try {
      await base44.functions.invoke('resetDispatchLock', { show_id: show.id, contract_id: contractId });
      setContractState((current) => ({ ...current, status: 'READY', dispatch_lock: null, dispatch_heartbeat_at: null }));
    } catch (e) {
      setError(e?.response?.data?.error || e.message || 'Could not reset the dispatch lock');
    } finally {
      setResetting(false);
    }
  };

  const canReview = ['draft', 'READY'].includes(contractState?.status) && !!contractState?.contract_template_id;
  const canSend = !!to && !!subject && !!pdfUrl && !!contractId && !gmailBlocked && !loading && contractState?.status === 'READY';

  return (
    <Sheet open={open} onOpenChange={(v) => !v && onClose?.()}>
      <SheetContent side="right" className="flex w-full flex-col gap-0 border-l border-slate-200 bg-white p-0 text-slate-900 sm:max-w-xl">
        {/* Header */}
        <div className="flex items-start justify-between gap-3 border-b border-slate-200 px-5 py-4 pr-14">
          <div className="flex min-w-0 items-center gap-3">
            <div className="flex h-9 w-9 flex-shrink-0 items-center justify-center rounded-xl bg-black text-yellow-400">
              <Mail className="h-4 w-4" />
            </div>
            <div className="min-w-0">
            <SheetTitle className="text-base font-semibold text-slate-900">
              Dispatch to Promoter
            </SheetTitle>
            <SheetDescription className="mt-0.5 flex items-center gap-1.5 text-xs text-slate-500">
              {threadId ? (
                <><CornerUpLeft className="h-3 w-3" /> Replying on the existing deal thread</>
              ) : (
                'Starting a new email thread for this deal'
              )}
            </SheetDescription>
            </div>
          </div>
        </div>

        {sent ? (
          <div className="flex flex-1 flex-col items-center justify-center gap-4 px-6 text-center">
            <div className="flex h-14 w-14 items-center justify-center rounded-full bg-green-50">
              <CheckCircle2 className="h-7 w-7 text-green-600" />
            </div>
            <div>
              <p className="text-base font-semibold text-slate-900">Contract Sent</p>
              <p className="mt-1 text-sm text-slate-500">
                {sent.was_reply ? 'Sent as a reply on the deal thread.' : 'New thread started and bound to this show.'}
                {' '}Contract status is now SENT.
              </p>
            </div>
            <Button onClick={onClose} className="h-11 bg-slate-900 text-white hover:bg-slate-800">Done</Button>
          </div>
        ) : reviewing ? (
          <ContractReviewEditor show={show} contract={contractState} onBack={() => setReviewing(false)} onUpdated={handleRegenerated} />
        ) : (
          <>
            <div className="flex-1 space-y-4 overflow-y-auto px-5 py-4">
              {gmailBlocked && (
                <div className="flex items-start gap-2.5 rounded-lg border border-slate-200 bg-amber-50 p-3">
                  <AlertTriangle className="mt-0.5 h-4 w-4 flex-shrink-0 text-amber-600" />
                  <div className="text-xs text-slate-700">
                    <p className="font-semibold">Gmail is not connected</p>
                    <p className="mt-0.5">Connect your Gmail account to send contracts as yourself.</p>
                    <a href="/settings" className="mt-1.5 inline-block font-semibold text-slate-900 underline">
                      Connect Gmail →
                    </a>
                  </div>
                </div>
              )}

              {loading ? (
                <div className="flex items-center gap-2 py-10 text-sm text-slate-500">
                  <Loader2 className="h-4 w-4 animate-spin" /> Preparing the email…
                </div>
              ) : (
                <>
                  <EmailChipInput
                    label="To"
                    values={to ? [to] : []}
                    onChange={(v) => setTo(v[v.length - 1] || '')}
                    placeholder="promoter@example.com"
                  />

                  <EmailChipInput label="Cc" values={cc} onChange={setCc} placeholder="Add an email and press Enter" />

                  <div className="space-y-1.5">
                    <p className="text-[11px] font-semibold uppercase tracking-wide text-slate-500">Subject</p>
                    <div className="flex items-center rounded-lg border border-input bg-background focus-within:ring-1 focus-within:ring-ring">
                      {threadId && <span className="pl-3 text-sm text-slate-500">Re:</span>}
                      <Input
                        value={subject}
                        onChange={(e) => setSubject(e.target.value)}
                        className="h-11 border-0 text-sm shadow-none focus-visible:ring-0"
                      />
                    </div>
                    {threadId && <p className="text-[11px] text-slate-500">Stays in the same email thread with the promoter.</p>}
                  </div>

                  <PersonalTemplatePicker user={me} currentBody={emailBody} subject={subject} onApply={setEmailBody} />

                  <div className="space-y-1.5">
                    <p className="text-[11px] font-semibold uppercase tracking-wide text-slate-500">Message</p>
                    <Textarea
                      value={emailBody}
                      onChange={(e) => setEmailBody(e.target.value)}
                      rows={12}
                      className="text-sm leading-relaxed"
                    />
                  </div>

                  {contractState && !['READY', 'DISPATCHING'].includes(contractState.status) && (
                    <p className="rounded-lg border border-slate-200 bg-slate-50 p-3 text-xs text-slate-700">This contract is {contractState.status}. Create an amendment before dispatching a new version.</p>
                  )}
                  {error && (
                    <p className="rounded-lg border border-slate-200 bg-red-50 p-3 text-xs text-red-700">{error}</p>
                  )}
                </>
              )}
            </div>

            {/* Attachment + send */}
            <div className="space-y-3 border-t border-slate-200 bg-slate-50 px-5 py-4">
              {isStale && (
                <div className="rounded-xl border border-slate-200 bg-white p-4" role="alert">
                  <div className="flex items-start gap-3">
                    <AlertTriangle className="mt-0.5 h-5 w-5 flex-shrink-0 text-amber-600" />
                    <div>
                      <p className="text-xs font-bold uppercase tracking-wider text-amber-700">Dispatch timed out</p>
                      <p className="mt-1 text-xs leading-relaxed text-slate-700">Check your Gmail Sent folder to ensure it didn't already send before retrying.</p>
                    </div>
                  </div>
                  <Button
                    onClick={handleReset}
                    disabled={resetting}
                    className="mt-3 h-11 w-full gap-2 bg-red-600 text-white hover:bg-red-500"
                  >
                    {resetting ? <><Loader2 className="h-4 w-4 animate-spin" /> Resetting…</> : <><AlertTriangle className="h-4 w-4" /> Reset & Retry Lock</>}
                  </Button>
                </div>
              )}
              {isDispatching && !isStale && (
                <div className="rounded-xl border border-slate-200 bg-white p-3 text-xs text-amber-700">
                  Dispatch is in progress. This lock becomes reviewable after five minutes.
                </div>
              )}
              <div className="flex items-center gap-3 rounded-xl border border-slate-200 bg-white px-3 py-2.5">
                <div className="flex h-9 w-9 flex-shrink-0 items-center justify-center rounded-lg bg-slate-100"><FileText className="h-4 w-4 text-slate-600" /></div>
                <div className="min-w-0 flex-1">
                  <p className="truncate text-xs font-semibold text-slate-800">{fileName}</p>
                  <p className="text-[11px] text-slate-500">Generated contract PDF</p>
                </div>
                {canReview && (
                  <button onClick={() => setReviewing(true)} className="min-h-[44px] px-1 text-xs font-semibold text-slate-900 hover:underline">
                    Review &amp; Edit
                  </button>
                )}
                {pdfUrl && (
                  <a
                    href={pdfUrl}
                    target="_blank"
                    rel="noopener noreferrer"
                    className="inline-flex min-h-[44px] items-center px-1 text-xs font-semibold text-slate-500 hover:text-slate-900 hover:underline"
                  >
                    View
                  </a>
                )}
              </div>

              <RiderAttachmentPicker show={show} value={riders} onChange={setRiders} />

              {!isStale && (
                <Button
                  onClick={handleSend}
                  disabled={!canSend || sending}
                  className="h-11 w-full gap-2 bg-black text-yellow-400 hover:bg-slate-800 disabled:bg-slate-200 disabled:text-slate-400 disabled:opacity-100"
                >
                  {sending
                    ? <><Loader2 className="h-4 w-4 animate-spin" /> Sending…</>
                    : <><Send className="h-4 w-4" /> {threadId ? 'Send Reply' : 'Send Email'}</>}
                </Button>
              )}
            </div>
          </>
        )}
      </SheetContent>
    </Sheet>
  );
}
