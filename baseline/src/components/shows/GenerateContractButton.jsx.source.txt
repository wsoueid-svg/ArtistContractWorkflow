import React, { useEffect, useState } from 'react';
import { Button } from '@/components/ui/button';
import { Dialog, DialogContent, DialogHeader, DialogTitle } from '@/components/ui/dialog';
import { FileText, Download, Check, Send } from 'lucide-react';
import ContractFillForm from '@/components/contracts/ContractFillForm';
import GenerateDealCardButton from '@/components/shows/GenerateDealCardButton';
import ContractDispatchSlideOver from '@/components/contracts/ContractDispatchSlideOver';
import { base44 } from '@/api/base44Client';

export default function GenerateContractButton({ show, showId, contractId: initialContractId, contract: initialContract = null }) {
  const [open, setOpen] = useState(false);
  const [generatedUrl, setGeneratedUrl] = useState(show?.contract_url || null);
  const [contractId, setContractId] = useState(initialContractId || show?.contract_id || null);
  const [dispatchOpen, setDispatchOpen] = useState(false);
  const [dispatched, setDispatched] = useState(false);
  const [threadId, setThreadId] = useState(show?.gmail_thread_id || null);
  const [contract, setContract] = useState(initialContract);
  const [amendmentMode, setAmendmentMode] = useState(false);

  useEffect(() => {
    if (!contractId) return;
    if (initialContract?.id === contractId) { setContract(initialContract); return; }
    base44.entities.Contract.get(contractId).then(setContract).catch(() => setContract(null));
  }, [contractId, initialContract]);

  const immutable = ['SENT', 'RETURNED', 'SIGNED'].includes(contract?.status);

  // Support both `show` object and legacy `showId` prop
  const showObj = show || (showId ? { id: showId } : null);

  const dispatchSlideOver = showObj && generatedUrl ? (
    <ContractDispatchSlideOver
      open={dispatchOpen}
      onClose={() => setDispatchOpen(false)}
      show={{ ...showObj, gmail_thread_id: threadId }}
      contractId={contractId}
      contractUrl={generatedUrl}
      onSent={(res) => {
        setDispatched(true);
        setContract((current) => ({ ...current, status: 'SENT' }));
        if (res?.gmail_thread_id) setThreadId(res.gmail_thread_id);
      }}
    />
  ) : null;

  if (generatedUrl) {
    return (
      <>
        <div className="flex flex-wrap items-center gap-2">
          <Check className="h-4 w-4 text-green-600" />
          <a
            href={generatedUrl}
            target="_blank"
            rel="noopener noreferrer"
            className="flex items-center gap-1 text-sm font-medium text-blue-600 hover:underline"
          >
            <Download className="h-3.5 w-3.5" /> Download PDF
          </a>
          {!immutable && !dispatched && <Button
            size="sm"
            onClick={() => setDispatchOpen(true)}
            className="h-8 gap-1.5 bg-slate-900 text-xs text-white hover:bg-slate-800"
          >
            <Send className="h-3 w-3" /> Dispatch to Promoter
          </Button>}
          {dispatched && (
            <span className="rounded-full bg-violet-100 px-2 py-0.5 text-[10px] font-semibold uppercase tracking-wide text-violet-700">
              Sent
            </span>
          )}
          <Button size="sm" variant="ghost" onClick={() => { setAmendmentMode(immutable); setGeneratedUrl(null); }} className="h-7 text-xs">
            {immutable ? 'Create Amendment' : 'Regenerate'}
          </Button>
          <GenerateDealCardButton show={showObj} />
        </div>
        {dispatchSlideOver}
      </>
    );
  }

  return (
    <>
      <div className="flex flex-wrap items-center gap-2">
        <Button
          size="sm"
          onClick={() => setOpen(true)}
          className="gap-1.5 bg-black text-xs text-white hover:bg-gray-800"
        >
          <FileText className="h-3 w-3" /> {amendmentMode ? 'Fill & Generate Amendment' : 'Fill & Generate Contract'}
        </Button>
        <GenerateDealCardButton show={showObj} />
      </div>

      <Dialog open={open} onOpenChange={setOpen}>
        <DialogContent className="max-w-2xl overflow-hidden p-0">
          <DialogHeader className="px-5 pb-0 pt-5">
            <DialogTitle className="flex items-start gap-2 pr-8 text-left text-base leading-tight">
              <FileText className="mt-0.5 h-4 w-4 shrink-0" />
              <span>Confirmation of Artist — Contract Generator</span>
            </DialogTitle>
            <p className="mt-1 pr-8 text-left text-xs text-gray-500">Fill in the promoter details to generate the standard Analog contract PDF.</p>
          </DialogHeader>
          <ContractFillForm
            show={showObj}
            forceAmendment={amendmentMode}
            onGenerated={(url, newContractId) => {
              setGeneratedUrl(url);
              setAmendmentMode(false);
              setContract((current) => ({ ...current, id: newContractId || current?.id, status: 'READY' }));
              if (newContractId) setContractId(newContractId);
              setOpen(false);
              setDispatchOpen(true);
            }}
            onClose={() => setOpen(false)}
          />
        </DialogContent>
      </Dialog>
    </>
  );
}
