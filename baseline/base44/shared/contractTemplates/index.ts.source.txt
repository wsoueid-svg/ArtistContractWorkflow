/** Brand family for a legal entity, from its explicit key or its legal name. */
function brandFamily(legalEntity: any): 'cidony' | 'nysm' | 'aesl' {
  const explicit = String(legalEntity?.contract_template_key || '');
  if (explicit.startsWith('nysm')) return 'nysm';
  if (explicit.startsWith('aesl')) return 'aesl';
  if (explicit === 'cidony') return 'cidony';

  const name = `${legalEntity?.name || ''} ${legalEntity?.short_name || ''}`.toUpperCase();
  if (name.includes('NOW YOU SEE ME') || name.includes('NYSM')) return 'nysm';
  if (name.includes('ANALOG ENTERTAINMENT')) return 'aesl';
  return 'cidony';
}

/**
 * Resolve the template key for a generation run.
 * Spanish is chosen when the promoter Contact prefers Spanish (or the entity
 * defaults to Spanish) AND the brand has a Spanish edition.
 */
export function resolveTemplateKey(legalEntity: any, contact: any, override?: string): TemplateKey {
  if (override && TEMPLATE_KEYS.includes(override as TemplateKey)) return override as TemplateKey;

  const family = brandFamily(legalEntity);
  const wantsSpanish =
    String(contact?.preferred_language || '').toLowerCase() === 'es' ||
    (!contact?.preferred_language && String(legalEntity?.invoice_language_default || '').toLowerCase() === 'es');

  if (family === 'nysm') return wantsSpanish ? 'nysm-es' : 'nysm';
  if (family === 'aesl') return wantsSpanish ? 'aesl-es' : 'aesl-en';
  return 'cidony';
}

/** Payment-schedule clauses — language-aware, percentage-interpolated. */
export function buildPaymentClauses(
  lang: 'en' | 'es',
  opts: { depositPct: number; balancePct: number; depositMonth?: string; artistName: string; templateKey?: TemplateKey },
) {
  const { depositPct, balancePct, artistName } = opts;
  const month = opts.depositMonth || '';
  const paymentDays = opts.templateKey === 'cidony' ? 10 : 3;

  if (lang === 'es') {
    return [
      `- El ${depositPct}% del caché del Artista se pagará al artista. Este pago se realizará${month ? ` en ${month}` : ''} y a más tardar en ${paymentDays} días en concepto de DEPÓSITO.`,
      `- El ${balancePct}% restante del cachet del artista se abonará antes de la fecha acordada entre la agencia y el promotor.`,
      `- Si no se acuerda una fecha para el ${balancePct}% del saldo del cachet, éste se abonará un (1) mes antes de la fecha del evento.`,
      '- El pago de la comisión de contratación y del DEPÓSITO deberá realizarse antes del anuncio del artista. El promotor dispondrá de un máximo de tres (3) días desde la emisión del contrato y de la factura para formalizar dichos pagos. Condición sine qua non para el uso del nombre del artista en cualquier tipo de publicidad del evento.',
      '- El incumplimiento por parte de la Compañía de este calendario de pagos constituirá un incumplimiento sustancial del presente Compromiso y facultará a la Agencia y al Artista para resolverlo de forma inmediata y sin previo aviso.',
      '- Se añadirá el IVA a todos los cachets acordados del artista en los shows del Reino Unido.',
      '- La compañía se compromete a contactar con ian@analog-a.com / alex@analog-a.com en caso de que sea necesario negociar las condiciones de pago.',
      `- El cachet del Artista será pagadero a ${artistName}: la mitad de forma inmediata en la fecha de firma del presente ("el Depósito") y el saldo a más tardar catorce (14) días antes de la Fecha de la Actuación. Todos los pagos se realizarán en la divisa acordada en el contrato (f).`,
    ];
  }

  return [
    `- The ${depositPct}% of the Artist fee will be paid to the artist. This payment will be made${month ? ` in ${month}` : ''} and ${paymentDays} days the latest in the concept of DEPOSIT.`,
    `- The ${balancePct}% Balance of the Artist fee shall be paid before the agreed date between agency and promoter.`,
    `- If not a date will be agreed for the ${balancePct}% balance of the artist fee to one (1) month before the event date.`,
    "- The Booking fee payment and DEPOSIT should be made before artist announcement. The promoter will have a maximum of three (3) days from contract and invoice issue to formalize said payments. Sine qua non condition, for the use of the artist's name in any type of publicity of the event.",
    '- The breach from the Company of this payment schedule will constitute a substantial breach of this Commitment, and authorizes the Agency and the Artist to terminate it immediately without prior notice.',
    '- A VAT will be added in addition to all agreed artist rates for UK shows.',
    '- The company agrees to contact ian@analog-a.com / alex@analog-a.com in case there needs to be a negotiation of payment terms.',
    `- The Artist Fee shall be payable ${artistName} and shall be payable as to one half promptly on the date of signature hereof ("the Deposit") and the balance by no later than fourteen (14) days prior to the Date of the Performance. All Payments are to be made in the currency agreed as per the contract (f).`,
  ];
}

/** Row labels for the payment table, per language. */
export function rowLabel(lang: 'en' | 'es', type: string, opts: { pct?: number; party: string }) {
  const labels = lang === 'es' ? ES.LABELS_ES : EN.LABELS_EN;
  const pct = opts.pct != null ? ` ${opts.pct}%` : '';
  if (type === 'BOOKING_FEE') return `${labels.bookingFeeRow} — ${opts.party}`;
  if (type === 'DEPOSIT') return `${labels.depositRow.replace('(', `${pct.trim()} (`)} — ${opts.party}`;
  if (type === 'BALANCE') return `${labels.balanceRow.replace('(', `${pct.trim()} (`)} — ${opts.party}`;
  return `${labels.fullFeeRow} — ${opts.party}`;
}

export function getTemplate(key: TemplateKey) {
  return REGISTRY[key];
}

/** Render a template key to { html, footerHtml } ready for PDFShift. */
export function renderContractDocument(key: TemplateKey, ctx: ContractContext) {
  const entry = REGISTRY[key];
  if (!entry) {
    const err: any = new Error(`Unknown contract template: ${key}`);
    err.status = 422;
    throw err;
  }
  return {
    html: buildContractHtml({ brand: entry.brand, pack: entry.pack, ctx }),
    footerHtml: buildFooterHtml(entry.brand),
    lang: entry.pack.lang,
    brandKey: entry.brand.key,
  };
}
